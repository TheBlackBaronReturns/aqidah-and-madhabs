# =====================================================================
# AQIDAH SCHOOLS OF CREED - the Fatwa of Jihad.
#
# Fired by on_action/00_aqd_game_start.txt's aqd_jihad_fatwa_pulse
# (monthly_country_pulse) on any Muslim realm currently party to a jihad,
# via EITHER of two participation paths (scripted_triggers/z_aqd_jihad_
# fatwa_triggers.txt): OFFENSIVE - leader, IO joiner, or a subject/ally
# fighting under casus_belli:cb_jihad without formal IO membership alike
# (aqd_is_jihad_participant) - or DEFENSIVE - a heathen (non-Muslim) power
# has declared a religious war on root's own territory (aqd_is_defensive_
# jihad_participant; heretic-on-heretic wars, e.g. Shia declaring on
# Sunni, never qualify - see that trigger's own header for why). One of
# the realm's own eligible jurist-seat provinces (aqd_jihad_seat_eligible
# - already-displeased seats and seats already carrying an unexpired
# Fatwa of Jihad of their own madhhab are always excluded; HETERODOX seats
# are excluded only from the offensive path, and are eligible to answer a
# defensive call) issues a fatwa calling all able-bodied believers to the
# jihad. Both paths share this one event, its flavor text, and its
# modifier keys - "defensive jihad" is itself a standard classical-fiqh
# concept, so the existing "called to take up arms in the jihad against
# [enemy]" desc reads correctly for either case with no wording changes.
#
# SILENT MISS BY DESIGN: unlike z_aqd_fatwa.1, this event never shows a
# "the jurists declined" popup. The per-madhhab approval roll (scripted_
# effects/z_aqd_madhhab_approval_effects.txt, fed by script_values/
# z_aqd_madhhab_approval.txt) happens entirely inside the pulse's own
# effect (scripted_effects/z_aqd_jihad_fatwa_effects.txt's aqd_jihad_
# fatwa_attempt), BEFORE this event ever fires - a declining roll simply
# produces no popup that month. This event only ever fires once a seat
# has already agreed, so it never needs a decline desc variant.
#
# Saved scopes do not survive trigger_event, so this event's own
# immediate re-derives its target seat via the shared pending-pick marker
# (main_menu/common/static_modifiers/z_aqd_fatwa_pending_pick.txt) that
# aqd_jihad_fatwa_attempt stamped just before calling trigger_event, and
# consumes it immediately - mirrors z_aqd_fatwa.2's own "re-find via a
# stable anchor, not a fresh random pick" idiom exactly.
# =====================================================================

namespace = z_aqd_jfatwa

z_aqd_jfatwa.1 = {
	type = country_event
	title = z_aqd_jfatwa.1.title
	desc = {
		first_valid = {
			# 9 concrete branches, NOT $aqd_seat_madhhab_name$ - that
			# customizable localization (common/customizable_localization/
			# z_aqd_seat_madhhab_name.txt, since deleted) checked
			# has_building, a LOCATION-scoped trigger, but was being invoked
			# bare from a single shared desc key, which renders at this
			# event's default scope (root) rather than scope:target_location
			# - every branch silently failed to match and the seat's
			# madhhab name came out blank.
			#
			# Checking var:aqd_jihad_fatwa_seat_madhhab (NOT scope:target_
			# location's has_building either) is the fix: a location can
			# legitimately hold more than one seat_of_<X>_jurists at once
			# (building_types/z_aqd_jurist_seats.txt's location_potential
			# only ever excludes a duplicate of the SAME madhhab), so even a
			# location-scoped has_building re-check here could name the
			# WRONG co-located seat. scripted_effects/z_aqd_jihad_fatwa_
			# effects.txt's aqd_jihad_fatwa_pick_seat already resolved
			# exactly which madhhab was picked and stamped it into this
			# variable on root; reading it back is unambiguous. scope:
			# target_location and scope:enemy are only ever set inside
			# their own exists = guards below - under a real fire both will
			# resolve, but the final fallback branch keeps a debug
			# force-fire (bypassing the normal pending-pick marker) from
			# rendering an unresolved custom scope name, mirroring
			# z_aqd_fatwa.1's own precedent for this exact failure class.
			triggered_desc = {
				trigger = { exists = scope:target_location  exists = scope:enemy  var:aqd_jihad_fatwa_seat_madhhab = 1 }
				desc = z_aqd_jfatwa.1.desc.hanafi
			}
			triggered_desc = {
				trigger = { exists = scope:target_location  exists = scope:enemy  var:aqd_jihad_fatwa_seat_madhhab = 2 }
				desc = z_aqd_jfatwa.1.desc.hanbali
			}
			triggered_desc = {
				trigger = { exists = scope:target_location  exists = scope:enemy  var:aqd_jihad_fatwa_seat_madhhab = 3 }
				desc = z_aqd_jfatwa.1.desc.maliki
			}
			triggered_desc = {
				trigger = { exists = scope:target_location  exists = scope:enemy  var:aqd_jihad_fatwa_seat_madhhab = 4 }
				desc = z_aqd_jfatwa.1.desc.shafii
			}
			triggered_desc = {
				trigger = { exists = scope:target_location  exists = scope:enemy  var:aqd_jihad_fatwa_seat_madhhab = 5 }
				desc = z_aqd_jfatwa.1.desc.zahiri
			}
			triggered_desc = {
				trigger = { exists = scope:target_location  exists = scope:enemy  var:aqd_jihad_fatwa_seat_madhhab = 6 }
				desc = z_aqd_jfatwa.1.desc.ismaili
			}
			triggered_desc = {
				trigger = { exists = scope:target_location  exists = scope:enemy  var:aqd_jihad_fatwa_seat_madhhab = 7 }
				desc = z_aqd_jfatwa.1.desc.jafari
			}
			triggered_desc = {
				trigger = { exists = scope:target_location  exists = scope:enemy  var:aqd_jihad_fatwa_seat_madhhab = 8 }
				desc = z_aqd_jfatwa.1.desc.zaidi
			}
			triggered_desc = {
				trigger = { exists = scope:target_location  exists = scope:enemy  var:aqd_jihad_fatwa_seat_madhhab = 9 }
				desc = z_aqd_jfatwa.1.desc.ibadi
			}
			triggered_desc = {
				desc = z_aqd_jfatwa.1.desc.fallback
			}
		}
	}
	outcome = neutral

	trigger = {
		religion.group = religion_group:muslim
		OR = {
			aqd_is_jihad_participant = yes
			aqd_is_defensive_jihad_participant = yes
		}
		any_owned_location = { has_location_modifier = z_aqd_fatwa_pending_pick }
	}

	illustration_tags = {
		10 = angry
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate  background = estate_type:clergy_estate }

		random_owned_location = {
			limit = { has_location_modifier = z_aqd_fatwa_pending_pick }
			save_scope_as = target_location
			province_definition = { save_scope_as = target_province }
		}
		if = {
			limit = { exists = scope:target_location }
			scope:target_location = { remove_location_modifier = z_aqd_fatwa_pending_pick }
		}

		if = {
			limit = { is_member_of_international_organization_of_type = { type = jihad } }
			# random_ (not any_) is the effect-context iterator that
			# supports save_scope_as - any_international_organizations_
			# member_of is trigger-only (confirmed usage: events/culture/
			# haudenosaunee.txt, inside an OR trigger, no save_scope_as).
			# Filter conditions belong inside limit, not as bare sibling
			# fields alongside save_scope_as (confirmed pattern: events/
			# situations/italian_wars.txt's own random_international_
			# organizations_member_of calls). A country can only sensibly
			# be a member of one jihad at a time, so random_ (grab
			# whichever matches) is sufficient, same reasoning as that
			# file's own use of random_ over every_.
			random_international_organizations_member_of = {
				limit = { international_organization_type = international_organization_type:jihad }
				save_scope_as = jihad_io
			}
		}
		if = {
			limit = { exists = scope:jihad_io }
			scope:jihad_io = {
				international_organization_target = { save_scope_as = enemy }
			}
		}
		if = {
			limit = { NOT = { exists = scope:enemy } }
			random_current_war = {
				limit = {
					is_an_attacker = root
					casus_belli ?= casus_belli:cb_jihad
				}
				save_scope_as = target_war
			}
			if = {
				limit = { exists = scope:target_war }
				scope:target_war = { defender_leader = { save_scope_as = enemy } }
			}
		}
		# Defensive fallback: only reached if neither the jihad-IO nor the
		# cb_jihad-attacker branch above resolved scope:enemy - i.e. this
		# firing came from aqd_is_defensive_jihad_participant instead
		# (scripted_triggers/z_aqd_jihad_fatwa_triggers.txt). Re-finds the
		# SAME war/attacker condition that trigger checks, then saves the
		# ATTACKER (not defender_leader - root is the one being attacked
		# here) as scope:enemy for the desc/tooltip to name.
		if = {
			limit = { NOT = { exists = scope:enemy } }
			random_current_war = {
				limit = {
					is_a_defender = root
					is_religious_war = yes
					attacker_leader = {
						NOT = { religion.group = root.religion.group }
					}
				}
				save_scope_as = target_war
			}
			if = {
				limit = { exists = scope:target_war }
				scope:target_war = { attacker_leader = { save_scope_as = enemy } }
			}
		}
	}

	option = {
		name = z_aqd_jfatwa.1.ok
		if = {
			limit = { exists = scope:target_location }
			# add_manpower is a country-scoped effect (confirmed idiom:
			# events/colonization/conquest_of_paradise.txt's own
			# `add_manpower = { value = root.monthly_manpower multiply = N }`,
			# always called at the event's implicit root scope) - it must NOT
			# be nested inside scope:target_location = { ... } the way
			# add_location_modifier below correctly is.
			#
			# Scaled off root.monthly_manpower, NOT scope:target_location.
			# population (the previous version) - population's actual in-
			# engine scale was never verified against anything else in this
			# mod (every other calibration comment in this file cites a
			# confirmed source; this line never did), and other confirmed
			# population thresholds elsewhere in the base game (common/
			# script_values/building_caps.txt's population * 0.05 feeding a
			# small building-level cap; population > 2.0 / total_population >
			# 10000 trigger thresholds) point to population being a heavily
			# abstracted unit with no verified relationship to a country's
			# own manpower economy - exactly the kind of unverified-scale
			# assumption that already produced two other confirmed bugs in
			# this mod. multiply = 1 ("one month's worth of mobilization")
			# is deliberately more modest than the base game's own -2/-3
			# usage for a one-off national event - this fatwa can now fire
			# more than once per war (a different eligible seat each time,
			# scripted_effects/z_aqd_jihad_fatwa_effects.txt's reservoir-
			# sampling picker), so the per-firing grant needs to stay small
			# enough that several of them stacking up over one long jihad
			# doesn't itself become the excessive amount this was meant to
			# fix. Tune the multiplier directly here if actual play still
			# feels off in either direction. The option's tooltip is
			# guaranteed to preview this accurately since it's the same
			# confirmed idiom the base game itself relies on.
			add_manpower = { value = root.monthly_manpower  multiply = 1 }
			scope:target_location = {
				aqd_apply_fatwa_of_jihad_for_seat = yes
			}
		}
	}
}
