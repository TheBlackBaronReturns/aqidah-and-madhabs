# =====================================================================
# AQIDAH SCHOOLS OF CREED - Ulema Institutions: foreign jurists arrive.
#
# No longer a blind monthly neighbor scan. Triggered DIRECTLY by events/
# religion/z_aqd_decline_events.txt's "let them go" option, via
# scripted_effects/z_aqd_decline_effects.txt's aqd_decline_migration_pick
# - a seat somewhere else has just declined (A Seat in Decline) and its
# jurists have relocated to THIS realm specifically, weighted-picked as
# a bordering Muslim neighbor (land border or across a single sea zone,
# preferring one whose own state madhhab already matches). var:aqd_
# jurists_madhhab (set on this country by the picker) is the sole
# trigger condition and the sole source of which madhhab is arriving -
# mirrors z_aqd_exile.1's own has_variable trigger pattern exactly.
#
# The target city for the non-capital option is still picked ONCE, in
# immediate (aqd_jurists_derive_seat_target -> scope:aqd_jurists_seat_loc
# / _seat_prov), not re-rolled inside the chosen option - so every
# option acts on the SAME city (scripted_effects/z_aqd_jurist_effects.
# txt). NOT naming the target city dynamically in the desc:
# [scope:X.GetName] does not resolve for a custom save_scope_as name in
# loc text - confirmed via debug.log ("Failed to find type
# 'scope:halqa_loc'..."), see z_aqd_halqa events/loc for the full note.
#
# The receiver chooses how far to go:
#   settle_capital - raise the Seat directly in the realm's own capital.
#                     Shown only when the capital doesn't already hold a
#                     seat of the arriving madhhab (aqd_capital_lacks_
#                     seat_of_compat_madhhab, scripted_triggers/z_aqd_
#                     jurist_triggers.txt) - listed FIRST when available.
#   grant_seat     - raise a standing Seat of the visitors' madhhab in
#                     the randomly-derived city instead (skips the
#                     grassroots madrasas-then-establishment chain
#                     entirely). Always available.
#   refuse         - turn them away.
#
# All three apply the aqidah-judged clergy/legitimacy reaction
# (scripted_effects/z_aqd_jurist_effects.txt's aqd_apply_jurist_reaction)
# and then aqd_jurists_cleanup, so the has_variable = aqd_jurists_madhhab
# trigger above doesn't stay permanently satisfied once this resolves -
# mirrors z_aqd_exile.1's own aqd_exile_cleanup role.
# =====================================================================

namespace = z_aqd_jurists

z_aqd_jurists.1 = {
	type = country_event
	title = z_aqd_jurists.1.title
	desc = z_aqd_jurists.1.desc
	outcome = neutral

	trigger = {
		has_variable = aqd_jurists_madhhab
	}

	illustration_tags = {
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate  background = estate_type:clergy_estate }
		set_variable = { name = aqd_compat_madhhab  value = var:aqd_jurists_madhhab }
		aqd_classify_madhhab_compatibility = { COUNTRY = root }
		# Pick the (non-capital) seat's target city now, once, so every
		# option that uses it acts on the SAME city (scripted_effects/
		# z_aqd_jurist_effects.txt).
		aqd_jurists_derive_seat_target = yes
	}

	# Raise the Seat directly in the realm's own capital - listed first,
	# shown only when the capital doesn't already hold a seat of the
	# arriving madhhab.
	option = {
		name = z_aqd_jurists.1.settle_capital
		trigger = { aqd_capital_lacks_seat_of_compat_madhhab = yes }
		aqd_jurists_accept_capital = { MADHHAB = var:aqd_jurists_madhhab }
		aqd_apply_jurist_reaction = { OUTCOME = accept  COUNTRY = root }
		aqd_jurists_cleanup = yes
		ai_chance = {
			base = 60
			modifier = { factor = 3    var:aqd_jurist_compat = 1 }
			modifier = { factor = 0.2  var:aqd_jurist_compat = -1 }
		}
	}

	# Raise a standing Seat of the visitors' madhhab in the randomly-
	# derived city. No trigger: aqd_jurists_derive_seat_target's own
	# eligibility already guaranteed a host exists (the picker in
	# scripted_effects/z_aqd_decline_effects.txt only ever fires this
	# event once a valid target realm was found), so this option is
	# always available. aqd_jurists_accept still guards internally
	# (console-forced with no real source: silently does nothing).
	option = {
		name = z_aqd_jurists.1.grant_seat
		aqd_jurists_accept_dispatch = { ENDOW = no }
		aqd_apply_jurist_reaction = { OUTCOME = accept  COUNTRY = root }
		aqd_jurists_cleanup = yes
		ai_chance = {
			base = 40
			modifier = { factor = 3    var:aqd_jurist_compat = 1 }
			modifier = { factor = 0.2  var:aqd_jurist_compat = -1 }
		}
	}

	option = {
		name = z_aqd_jurists.1.refuse
		aqd_apply_jurist_reaction = { OUTCOME = refuse  COUNTRY = root }
		aqd_jurists_cleanup = yes
		ai_chance = {
			base = 30
			modifier = { factor = 5  var:aqd_jurist_compat = -1 }
			modifier = { factor = 0  var:aqd_jurist_compat = 1 }
		}
	}
}
