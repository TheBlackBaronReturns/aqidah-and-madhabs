# =====================================================================
# AQIDAH SCHOOLS OF CREED - Ulema Institutions: foreign jurists arrive.
#
# Fired rarely by on_action/00_aqd_game_start.txt's
# aqd_jurist_migration_pulse (hooked to monthly_country_pulse). Sources
# a neighboring Muslim country holding a madhhab that is neither the
# receiver's own current madhhab nor already Plurality-or-above in the
# receiver, AND that the receiver could raise a Seat of right now
# (scripted_triggers/z_aqd_jurist_triggers.txt's
# aqd_is_eligible_jurist_source). The event grants a Seat directly, so if
# no city can host one it does not fire at all.
#
# The target city is picked ONCE, in immediate
# (aqd_jurists_derive_seat_target -> scope:aqd_jurists_seat_loc /
# _seat_prov), not re-rolled inside the chosen option - so the desc can
# name it before the player decides, and console-forcing the event with
# no real source degrades gracefully instead of erroring (var:aqd_compat_madhhab
# is always set, to 0 as a safe sentinel if no source was found, so every
# var:aqd_compat_madhhab = N comparison downstream is well-defined).
#
# The receiver chooses how far to go:
#   grant_seat - raise a standing Seat of the visitors' madhhab in the
#                named city at once (skips the grassroots
#                madrasas-then-establishment chain entirely).
#   endow      - Seat + one endowed college of the madhhab beside it:
#                gold spent (aqd_endow_gold_estimate), and goodwill
#                (z_aqd_endowed_our_school) with every realm of that
#                school in diplomatic range. Shown only when the seat's
#                province also has a free madrasa slot
#                (aqd_source_madhhab_endow_hostable).
#   refuse     - turn them away.
#
# All three apply the aqidah-judged clergy/legitimacy reaction
# (scripted_effects/z_aqd_jurist_effects.txt's aqd_apply_jurist_reaction);
# accept = welcoming, refuse = moving against.
# =====================================================================

namespace = z_aqd_jurists

z_aqd_jurists.1 = {
	type = country_event
	title = z_aqd_jurists.1.title
	# NOT naming the target city dynamically: [scope:X.GetName] does not
	# resolve for a custom save_scope_as name in loc text - confirmed via
	# debug.log ("Failed to find type 'scope:halqa_loc'..."), see
	# z_aqd_halqa events/loc for the full note. Same limitation would hit
	# scope:aqd_jurists_seat_loc here.
	desc = z_aqd_jurists.1.desc
	outcome = neutral

	trigger = {
		religion.group = religion_group:muslim
		has_policy = sharia_law_policy
		any_neighbor_country = {
			religion.group = religion_group:muslim
			aqd_is_eligible_jurist_source = yes
		}
	}

	illustration_tags = {
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate  background = estate_type:clergy_estate }
		random_neighbor_country = {
			limit = {
				religion.group = religion_group:muslim
				aqd_is_eligible_jurist_source = yes
			}
			save_scope_as = z_aqd_jurist_source
		}
		# Canonical 1-9 madhhab index on root - drives the seat-target pick
		# and the option ai_chance weights. Always ends up SET (0 = "no
		# real source", the console-force safety case) so var:aqd_compat_madhhab
		# is never compared while unset.
		if = {
			limit = { exists = scope:z_aqd_jurist_source }
			aqd_set_compat_madhhab_from_country = { SOURCE = scope:z_aqd_jurist_source  COUNTRY = root }
		}
		else = {
			root = { set_variable = { name = aqd_compat_madhhab  value = 0 } }
		}
		aqd_classify_madhhab_compatibility = { COUNTRY = root }
		# Pick the seat's target city now, once, so the desc can name it
		# and every option acts on the SAME city (scripted_effects/
		# z_aqd_jurist_effects.txt).
		aqd_jurists_derive_seat_target = yes
		# Projected endowment spend for z_aqd_jurists.1.endow's tooltip
		# (script_values/z_aqd_madhhab_counts.txt).
		set_variable = { name = aqd_endow_gold_est  value = aqd_endow_gold_estimate }
	}

	# Raise a standing Seat of the visitors' madhhab in the named city.
	# No trigger: aqd_is_eligible_jurist_source already guaranteed a host
	# exists, so this option is always available (the always-present
	# non-refuse choice). aqd_jurists_accept still guards internally
	# (console-forced with no real source: silently does nothing).
	option = {
		name = z_aqd_jurists.1.grant_seat
		aqd_jurists_accept_dispatch = { ENDOW = no }
		aqd_apply_jurist_reaction = { OUTCOME = accept  COUNTRY = root }
		ai_chance = {
			base = 55
			modifier = { factor = 3    var:aqd_jurist_compat = 1 }
			modifier = { factor = 0.2  var:aqd_jurist_compat = -1 }
		}
	}

	# Seat + an endowed college beside it: gold spent, goodwill with
	# every realm of that school. Hidden when the seat's province has no
	# free madrasa slot, so the spend always buys a real college.
	option = {
		name = z_aqd_jurists.1.endow
		trigger = { aqd_source_madhhab_endow_hostable = yes }
		custom_tooltip = aqd_endow_gold_tt
		aqd_jurists_accept_dispatch = { ENDOW = yes }
		aqd_apply_jurist_reaction = { OUTCOME = accept  COUNTRY = root }
		ai_chance = {
			base = 25
			modifier = { factor = 3    var:aqd_jurist_compat = 1 }
			modifier = { factor = 0    var:aqd_jurist_compat = -1 }
			modifier = { factor = 0.1  gold < aqd_endow_gold_estimate }
		}
	}

	option = {
		name = z_aqd_jurists.1.refuse
		aqd_apply_jurist_reaction = { OUTCOME = refuse  COUNTRY = root }
		ai_chance = {
			base = 30
			modifier = { factor = 5  var:aqd_jurist_compat = -1 }
			modifier = { factor = 0  var:aqd_jurist_compat = 1 }
		}
	}
}
