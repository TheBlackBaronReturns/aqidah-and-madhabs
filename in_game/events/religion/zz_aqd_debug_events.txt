# Walks aqd_is_eligible_jurist_source's jafari branch condition-by-condition
# against c:AHI and reports, IN THE EVENT TEXT, the first one that fails -
# settles "is Ahi even eligible, and if not, why" without touching a log.
z_aqd_debug.4 = {
	type = country_event
	title = z_aqd_debug.4.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { NOT = { exists = c:AHI } }
				desc = z_aqd_debug.4.desc.no_ahi
			}
			triggered_desc = {
				trigger = { c:AHI = { aqd_is_eligible_jurist_source = yes } }
				desc = z_aqd_debug.4.desc.eligible
			}
			triggered_desc = {
				trigger = { NOT = { any_neighbor_country = { this = c:AHI } } }
				desc = z_aqd_debug.4.desc.not_neighbor
			}
			triggered_desc = {
				trigger = { NOT = { c:AHI = { religion.group = religion_group:muslim } } }
				desc = z_aqd_debug.4.desc.not_muslim
			}
			triggered_desc = {
				trigger = { NOT = { c:AHI = { has_policy = jafari_policy } } }
				desc = z_aqd_debug.4.desc.no_jafari
			}
			triggered_desc = {
				trigger = { has_policy = jafari_policy }
				desc = z_aqd_debug.4.desc.we_have_jafari
			}
			triggered_desc = {
				trigger = { jafari_institution_tier >= 3 }
				desc = z_aqd_debug.4.desc.tier_high
			}
			triggered_desc = {
				desc = z_aqd_debug.4.desc.unknown
			}
		}
	}
	outcome = neutral

	option = {
		name = z_aqd_debug.4.ok
	}
}
