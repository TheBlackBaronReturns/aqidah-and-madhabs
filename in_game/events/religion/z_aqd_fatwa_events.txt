# =====================================================================
# AQIDAH SCHOOLS OF CREED - The Jurists' Fatwa Against an Unjust War.
#
# Fired rarely by on_action/00_aqd_game_start.txt's aqd_fatwa_pulse /
# aqd_fatwa_escalation_pulse (monthly_country_pulse) on a Muslim realm
# that is right now the attacker-leader of a weakly-justified,
# clearly-winning war against a fellow-sect Muslim state whose madhhab
# the realm's own aqidah does not condemn (scripted_triggers/
# z_aqd_fatwa_triggers.txt's aqd_fatwa_core_conditions).
#
# z_aqd_fatwa.1 (the demand): one of the realm's own jurist-seat
# provinces issues a fatwa. Refusing stamps "Local <Madhhab> Jurists
# Displeased" (main_menu/common/static_modifiers/z_aqd_fatwa_displeased.
# txt) on THAT seat's own location for a flat 12 years, keyed to its own
# madhhab only - other madhhabs' seats elsewhere in the same province are
# unaffected. The ban ALWAYS also cancels that same madhhab's currently-
# active ḥalqa across the seat's WHOLE province (scripted_effects/
# z_aqd_fatwa_effects.txt's aqd_fatwa_apply_displeased_for_seat), matching
# the ḥalqa's own province-wide footprint - a repudiated circle can never
# keep running elsewhere in the province just because only the seat's own
# location was swept. Accepting ends the war outright via white_peace.
#
# z_aqd_fatwa.2 (the escalation): if the war drags on further (more war
# exhaustion, and enough time has passed since stage 1 was refused -
# scripted_effects/z_aqd_fatwa_effects.txt / the variable scheme below),
# a rare pulse may fire a second event: every OTHER eligible jurist-seat
# location in the same AREA (not just the same province) as the original
# seat joins in demanding peace. Refusing re-extends the original seat's
# ban to the same flat 12 years alongside a fresh 12-year ban - and its
# own province-wide ḥalqa cancellation - on every other seat that joined
# in.
#
# Saved scopes do not survive trigger_event, so both immediates
# re-derive their own war/enemy/target from scratch - mirroring
# z_aqd_halqa.1 / z_aqd_establishment.1 / z_aqd_exile.1.
#
# Anti-spam: aqd_fatwa_cooldown (5y, set on either stage-1 option and
# on stage-2 refuse) blocks stage-1 re-fire. aqd_fatwa_stage1_fired (6y,
# set on stage-1 refuse, consumed by either stage-2 option) proves
# stage 1 happened for this episode. aqd_fatwa_stage2_too_soon (1.5y,
# set on stage-1 refuse) is a self-expiring "too soon" flag - stage 2
# becomes possible only once it expires, carving out a 1.5-6 year
# window after stage 1's refusal. All three use the confirmed
# `set_variable = { years = N }` self-expiry idiom (base game's
# events/government/parliaments.txt's asked_for_parliament_seat_already).
# =====================================================================

namespace = z_aqd_fatwa

z_aqd_fatwa.1 = {
	type = country_event
	title = z_aqd_fatwa.1.title
	desc = {
		first_valid = {
			# Both scope:target_location and scope:enemy are only ever set
			# inside their own `exists =` guards in immediate below - under
			# a real fire the event's own trigger already guaranteed both
			# will resolve, but a debug force-fire (zz_aqd_debug_effects.txt's
			# aqd_debug_force_fatwa_1, which bypasses this event's trigger on
			# purpose) can reach here with no real war/enemy found at all.
			# Referencing an unresolved custom scope name in loc text is a
			# silent-failure class this mod has hit before (see this file's
			# own z_aqd_halqa.1 precedent) - so both dynamic variants are
			# gated on both scopes actually existing, with a scope-free
			# fallback below for when they do not.
			triggered_desc = {
				trigger = { exists = scope:target_location  exists = scope:enemy  var:aqd_jurist_compat = 1 }
				desc = z_aqd_fatwa.1.desc.favored
			}
			triggered_desc = {
				trigger = { exists = scope:target_location  exists = scope:enemy }
				desc = z_aqd_fatwa.1.desc.neutral
			}
			triggered_desc = {
				desc = z_aqd_fatwa.1.desc.fallback
			}
		}
	}
	outcome = neutral

	trigger = {
		aqd_fatwa_core_conditions = yes
		war_exhaustion >= 5
		NOT = { has_variable = aqd_fatwa_cooldown }
		any_owned_location = { has_location_modifier = z_aqd_fatwa_pending_pick }
	}

	illustration_tags = {
		10 = angry
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate  background = estate_type:clergy_estate }

		set_variable = { name = aqd_compat_madhhab  value = 0 }

		random_current_war = {
			limit = {
				attacker_leader = root
				OR = {
					is_no_cb_war = yes
					NOT = { has_casus_belli = yes }
					casus_belli ?= casus_belli:cb_fabricated_conquer_province
					casus_belli ?= casus_belli:cb_trade_conflict
					casus_belli ?= casus_belli:cb_disrupting_trade
					casus_belli ?= casus_belli:cb_humiliate_rival
					casus_belli ?= casus_belli:cb_humiliate
					casus_belli ?= casus_belli:cb_conquer_enemy
					casus_belli ?= casus_belli:attack_threat
					casus_belli ?= casus_belli:cb_imperialism
				}
				defender_leader = {
					religion = root.religion
					aqd_madhhab_acceptable_to_root = yes
					war_score_versus = { target = root  value <= -0.25 }
				}
			}
			save_scope_as = target_war
		}

		if = {
			limit = { exists = scope:target_war }
			scope:target_war = {
				defender_leader = { save_scope_as = enemy }
			}
		}
		if = {
			limit = { exists = scope:enemy }
			aqd_set_compat_madhhab_from_country = { SOURCE = scope:enemy  COUNTRY = root }
		}
		aqd_classify_madhhab_compatibility = { COUNTRY = root }

		# Re-find the seat that scripted_effects/z_aqd_fatwa_effects.txt's
		# aqd_fatwa_attempt_stage1 already picked and rolled a YES on
		# (from the pulse, before this event ever fired - the silent-miss
		# design) via the shared pending-pick marker, and consume it
		# immediately. Saved scopes do not survive trigger_event, so this
		# cannot simply reuse a scope aqd_fatwa_attempt_stage1 set.
		random_owned_location = {
			limit = { has_location_modifier = z_aqd_fatwa_pending_pick }
			save_scope_as = target_location
			province_definition = { save_scope_as = target_province }
		}
		if = {
			limit = { exists = scope:target_location }
			scope:target_location = { remove_location_modifier = z_aqd_fatwa_pending_pick }
		}
	}

	option = {
		name = z_aqd_fatwa.1.accept
		if = {
			limit = { exists = scope:target_war }
			white_peace = scope:target_war
		}
		set_variable = { name = aqd_fatwa_cooldown  years = 5 }
		remove_variable = aqd_seat_madhhab
		remove_variable = aqd_seat_compat
		remove_variable = aqd_approval_scratch
		remove_variable = aqd_madhhab_approval_result
		remove_variable = aqd_fatwa_seat_madhhab
	}
	option = {
		name = z_aqd_fatwa.1.refuse
		if = {
			limit = { exists = scope:target_location }
			# Called bare at this option's own default scope (root) - NOT
			# nested inside scope:target_location = {...} - so var:aqd_fatwa_
			# seat_madhhab resolves against root, where it actually lives.
			# aqd_fatwa_apply_displeased_for_seat re-enters the location
			# itself via its own $LOCATION$ parameter for the modifier calls.
			aqd_fatwa_apply_displeased_for_seat = { MADHHAB = var:aqd_fatwa_seat_madhhab  LOCATION = scope:target_location }
		}
		set_variable = { name = aqd_fatwa_cooldown  years = 5 }
		set_variable = { name = aqd_fatwa_stage1_fired  years = 6 }
		set_variable = { name = aqd_fatwa_stage2_too_soon  years = 1.5 }
		remove_variable = aqd_seat_madhhab
		remove_variable = aqd_seat_compat
		remove_variable = aqd_approval_scratch
		remove_variable = aqd_madhhab_approval_result
		remove_variable = aqd_fatwa_seat_madhhab
	}
}

z_aqd_fatwa.2 = {
	type = country_event
	title = z_aqd_fatwa.2.title
	desc = {
		first_valid = {
			# Same guard as z_aqd_fatwa.1's desc - see its comment. Here
			# scope:target_province just proves a real anchor seat was
			# found (stage 2's own loop runs over scope:target_area, not
			# this province - see the refuse option below - but the desc
			# text no longer names the province, so target_province is only
			# an existence check here, not a display value).
			triggered_desc = {
				trigger = { exists = scope:target_province  exists = scope:enemy  var:aqd_jurist_compat = 1 }
				desc = z_aqd_fatwa.2.desc.favored
			}

				desc = z_aqd_fatwa.2.desc.neutral
			}
			triggered_desc = {
				desc = z_aqd_fatwa.2.desc.fallback
			}
		}
	}
	outcome = neutral

	trigger = {
		aqd_fatwa_core_conditions = yes
		war_exhaustion >= 8
		has_variable = aqd_fatwa_stage1_fired
		NOT = { has_variable = aqd_fatwa_stage2_too_soon }
		any_owned_location = { aqd_location_has_any_fatwa_ban = yes }
	}

	illustration_tags = {
		10 = angry
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate  background = estate_type:clergy_estate }

		set_variable = { name = aqd_compat_madhhab  value = 0 }

		random_current_war = {
			limit = {
				attacker_leader = root
				OR = {
					is_no_cb_war = yes
					NOT = { has_casus_belli = yes }
					casus_belli ?= casus_belli:cb_fabricated_conquer_province
					casus_belli ?= casus_belli:cb_trade_conflict
					casus_belli ?= casus_belli:cb_disrupting_trade
					casus_belli ?= casus_belli:cb_humiliate_rival
					casus_belli ?= casus_belli:cb_humiliate
					casus_belli ?= casus_belli:cb_conquer_enemy
					casus_belli ?= casus_belli:attack_threat
					casus_belli ?= casus_belli:cb_imperialism
				}
				defender_leader = {
					religion = root.religion
					aqd_madhhab_acceptable_to_root = yes
					war_score_versus = { target = root  value <= -0.25 }
				}
			}
			save_scope_as = target_war
		}

		if = {
			limit = { exists = scope:target_war }
			scope:target_war = {
				defender_leader = { save_scope_as = enemy }
			}
		}
		if = {
			limit = { exists = scope:enemy }
			aqd_set_compat_madhhab_from_country = { SOURCE = scope:enemy  COUNTRY = root }
		}
		aqd_classify_madhhab_compatibility = { COUNTRY = root }

		# Re-find the SAME seat via the stage-1 anchor (the ban itself), not
		# a fresh pick - scope:target_area then covers every seat-holding
		# location in the same AREA (not just the same province) that joins
		# in below; scope:target_province is kept only for the event's own
		# flavor desc (naming the originating province), the escalation loop
		# itself no longer uses it.
		random_owned_location = {
			limit = { aqd_location_has_any_fatwa_ban = yes }
			save_scope_as = target_location
			province_definition = { save_scope_as = target_province }
			area = { save_scope_as = target_area }
		}
	}

	option = {
		name = z_aqd_fatwa.2.accept
		if = {
			limit = { exists = scope:target_war }
			white_peace = scope:target_war
		}
		remove_variable = aqd_fatwa_stage1_fired
		remove_variable = aqd_seat_madhhab
		remove_variable = aqd_seat_compat
		remove_variable = aqd_approval_scratch
		remove_variable = aqd_madhhab_approval_result
	}
	option = {
		name = z_aqd_fatwa.2.refuse
		if = {
			limit = { exists = scope:target_area }
			scope:target_area = {
				# Iterates per (location, madhhab) PAIR, not just per
				# location - a location can legitimately hold more than one
				# eligible seat at once (building_types/z_aqd_jurist_seats.
				# txt's location_potential only ever excludes a duplicate of
				# the SAME madhhab), and each one is its own distinct voice
				# that should get its own independent say, matching this
				# event's own narration ("Other jurists' seats... are adding
				# their voice"). The single aqd_seat_madhhab_not_heterodox_
				# to_root umbrella check used to collapse a multi-seat
				# location down to whichever building a fixed priority order
				# checked first (scripted_effects/z_aqd_fatwa_effects.txt's
				# aqd_compute_seat_approval, before it accepted a pre-set
				# madhhab) - the 9 aqd_fatwa_seat_eligible_<X> checks below
				# (scripted_triggers/z_aqd_fatwa_triggers.txt - each layers
				# the doctrinal aqd_seat_madhhab_not_heterodox_to_root_<X>
				# check with an exclusion for a location already mid-pick
				# under an unrelated in-flight fatwa) replace it with nine
				# independent per-madhhab checks. Each
				# eligible seat rolls its own INVERTED approval independently
				# (aqd_fatwa_evaluate_seat_demand, given exactly which
				# madhhab to evaluate this time) - only seats that roll a YES
				# join the original in demanding peace; the rest stay quiet,
				# no separate popup needed since this all happens inside an
				# option the player already chose. Scratch vars (aqd_seat_
				# madhhab etc.) are clobbered and reconsumed every iteration -
				# safe, since each is read immediately within the same
				# iteration and the event's desc has already rendered before
				# this option runs; do not "fix" this into per-seat unique
				# scratch names.
				#
				# every_location_in_area (not _in_province_definition): the
				# escalation now pulls in every eligible seat across the
				# whole AREA the anchor seat sits in, not just its own
				# province - a madhhab's juristic network reads as a
				# regional body, not one bounded strictly by a single
				# province's borders.
				#
				# The actual ban/ḥalqa-clear per branch is applied INLINE
				# here (bare add_location_modifier + a province_definition
				# sweep for the ḥalqa clear, at this loop's own implicit
				# location scope) rather than via scripted_effects/
				# z_aqd_fatwa_effects.txt's aqd_fatwa_apply_displeased_for_
				# seat - that shared function exists for z_aqd_fatwa.1's
				# refuse option, which only learns its madhhab at RUNTIME (a
				# variable, root-scoped, needing the location re-entered
				# separately for the actual modifier calls). Here the
				# madhhab is already known statically at each of the 9
				# branches below, so no runtime comparison or cross-scope
				# parameter juggling is needed - the same reasoning the
				# reservoir-sampling pickers use throughout this mod for
				# their own literal per-position values. The ḥalqa clear is
				# swept over EACH eligible seat's own province_definition
				# (not the anchor's), same reasoning as aqd_fatwa_apply_
				# displeased_for_seat's own header: a ḥalqa's class modifier
				# is stamped uniformly across its whole originating province,
				# so cancelling it only at the one seat location that just
				# joined the escalation would leave every other location of
				# THAT seat's own province still carrying the repudiated
				# circle.
				every_location_in_area = {
					limit = { owner ?= root }
					if = {
						limit = { aqd_fatwa_seat_eligible_hanafi = yes }
						aqd_fatwa_evaluate_seat_demand = { MADHHAB = 1 }
						if = {
							limit = { root = { var:aqd_madhhab_approval_result = 1 } }
							add_location_modifier = { modifier = z_aqd_fatwa_displeased_hanafi  years = 12  mode = add_and_extend }
							province_definition = {
								every_location_in_province_definition = {
									limit = { owner ?= root }
									remove_location_modifier = z_aqd_halqa_state_hanafi
									remove_location_modifier = z_aqd_halqa_orthodox_hanafi
									remove_location_modifier = z_aqd_halqa_heterodox_hanafi
								}
							}
						}
					}
					if = {
						limit = { aqd_fatwa_seat_eligible_hanbali = yes }
						aqd_fatwa_evaluate_seat_demand = { MADHHAB = 2 }
						if = {
							limit = { root = { var:aqd_madhhab_approval_result = 1 } }
							add_location_modifier = { modifier = z_aqd_fatwa_displeased_hanbali  years = 12  mode = add_and_extend }
							province_definition = {
								every_location_in_province_definition = {
									limit = { owner ?= root }
									remove_location_modifier = z_aqd_halqa_state_hanbali
									remove_location_modifier = z_aqd_halqa_orthodox_hanbali
									remove_location_modifier = z_aqd_halqa_heterodox_hanbali
								}
							}
						}
					}
					if = {
						limit = { aqd_fatwa_seat_eligible_maliki = yes }
						aqd_fatwa_evaluate_seat_demand = { MADHHAB = 3 }
						if = {
							limit = { root = { var:aqd_madhhab_approval_result = 1 } }
							add_location_modifier = { modifier = z_aqd_fatwa_displeased_maliki  years = 12  mode = add_and_extend }
							province_definition = {
								every_location_in_province_definition = {
									limit = { owner ?= root }
									remove_location_modifier = z_aqd_halqa_state_maliki
									remove_location_modifier = z_aqd_halqa_orthodox_maliki
									remove_location_modifier = z_aqd_halqa_heterodox_maliki
								}
							}
						}
					}
					if = {
						limit = { aqd_fatwa_seat_eligible_shafii = yes }
						aqd_fatwa_evaluate_seat_demand = { MADHHAB = 4 }
						if = {
							limit = { root = { var:aqd_madhhab_approval_result = 1 } }
							add_location_modifier = { modifier = z_aqd_fatwa_displeased_shafii  years = 12  mode = add_and_extend }
							province_definition = {
								every_location_in_province_definition = {
									limit = { owner ?= root }
									remove_location_modifier = z_aqd_halqa_state_shafii
									remove_location_modifier = z_aqd_halqa_orthodox_shafii
									remove_location_modifier = z_aqd_halqa_heterodox_shafii
								}
							}
						}
					}
					if = {
						limit = { aqd_fatwa_seat_eligible_zahiri = yes }
						aqd_fatwa_evaluate_seat_demand = { MADHHAB = 5 }
						if = {
							limit = { root = { var:aqd_madhhab_approval_result = 1 } }
							add_location_modifier = { modifier = z_aqd_fatwa_displeased_zahiri  years = 12  mode = add_and_extend }
							province_definition = {
								every_location_in_province_definition = {
									limit = { owner ?= root }
									remove_location_modifier = z_aqd_halqa_state_zahiri
									remove_location_modifier = z_aqd_halqa_orthodox_zahiri
									remove_location_modifier = z_aqd_halqa_heterodox_zahiri
								}
							}
						}
					}
					if = {
						limit = { aqd_fatwa_seat_eligible_ismaili = yes }
						aqd_fatwa_evaluate_seat_demand = { MADHHAB = 6 }
						if = {
							limit = { root = { var:aqd_madhhab_approval_result = 1 } }
							add_location_modifier = { modifier = z_aqd_fatwa_displeased_ismaili  years = 12  mode = add_and_extend }
							province_definition = {
								every_location_in_province_definition = {
									limit = { owner ?= root }
									remove_location_modifier = z_aqd_halqa_state_ismaili
									remove_location_modifier = z_aqd_halqa_orthodox_ismaili
									remove_location_modifier = z_aqd_halqa_heterodox_ismaili
								}
							}
						}
					}
					if = {
						limit = { aqd_fatwa_seat_eligible_jafari = yes }
						aqd_fatwa_evaluate_seat_demand = { MADHHAB = 7 }
						if = {
							limit = { root = { var:aqd_madhhab_approval_result = 1 } }
							add_location_modifier = { modifier = z_aqd_fatwa_displeased_jafari  years = 12  mode = add_and_extend }
							province_definition = {

									remove_location_modifier = z_aqd_halqa_orthodox_jafari
									remove_location_modifier = z_aqd_halqa_heterodox_jafari
								}
							}
						}
					}
					if = {
						limit = { aqd_fatwa_seat_eligible_zaidi = yes }
						aqd_fatwa_evaluate_seat_demand = { MADHHAB = 8 }
						if = {
							limit = { root = { var:aqd_madhhab_approval_result = 1 } }
							add_location_modifier = { modifier = z_aqd_fatwa_displeased_zaidi  years = 12  mode = add_and_extend }
							province_definition = {
								every_location_in_province_definition = {
									limit = { owner ?= root }
									remove_location_modifier = z_aqd_halqa_state_zaidi
									remove_location_modifier = z_aqd_halqa_orthodox_zaidi
