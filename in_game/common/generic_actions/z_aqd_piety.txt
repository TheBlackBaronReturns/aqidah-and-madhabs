# REPLACE change_main_school: it now only changes the aqidah
# religious_school (the select_trigger auto-filters to the whitelist,
# which is creed-only after System B). The madhhab add_policy branches
# are removed - madhhab is now changed via aqd_change_madhhab.
REPLACE:change_main_school = {
	type = religious
	sound = UI_action_religion_generic

	potential = {
		scope:actor = {
			religion = { has_religious_schools = yes }
		}
	}

	allow = {
		scope:actor = {
			any_invited_religious_figure = { count > 0 }
		}
	}

	ai_tick = monthly
	ai_tick_frequency = 1
	automation_tick = monthly
	automation_tick_frequency = 1
	player_automated_category = religiousdoctrines

	price = price:change_main_school

	select_trigger = {
		looking_for_a = religion
		source = actor
		target_flag = recipient
		name = "choose_religion"
		column = { data = name }
	}

	select_trigger = {
		looking_for_a = religious_school
		source = recipient
		target_flag = target
		name = "choose_religious_school"
		none_available_msg_key = "no_valid_religious_school"
		column = { data = name }
		visible = {
			scope:actor = {
				any_invited_religious_figure ?= {
					religious_school ?= root
				}
			}
		}
	}

	effect = {
		if = {
			limit = { exists = scope:target }
			scope:actor = {
				set_religious_school = scope:target
			}
		}
	}

	ai_will_do = {
		add = "scope:actor.currency_utility(stability|10)"
	}
}

# New: change your realm's madhhab (a sharia_law policy). The single
# friction-bearing madhhab-change path now that it is no longer coupled
# to the school of creed.
aqd_change_madhhab = {
	show_message = no
	type = religious
	sound = UI_action_religion_generic

	ai_tick = monthly
	ai_tick_frequency = 12
	automation_tick = never

	potential = {
		scope:actor = {
			religion.group = religion_group:muslim
			has_policy = sharia_law_policy
		}
	}
	allow = { }
	cooldown = { type = aqd_change_madhhab  months = 60 }
	price = price:aqd_change_madhhab

	select_trigger = {
		looking_for_a = policy
		target_flag = target
		name = "aqd_madhhab_select"
		none_available_msg_key = "aqd_madhhab_none"
		column = { data = name }
		visible = {
			OR = {
				this = policy:hanafi_policy
				this = policy:hanbali_policy
				this = policy:maliki_policy
				this = policy:shafii_policy
				this = policy:zahiri_policy
				this = policy:ismaili_policy
				this = policy:jafari_policy
				this = policy:zaidi_policy
				this = policy:ibadi_policy
			}
		}
	}

	effect = {
		scope:actor = {
			add_policy = scope:target
		}
	}
	ai_will_do = { add = 0 }
}

# =====================================================================
# Ulema Institutions - crack down on a rival madhhab present in the
# realm (see building_types/z_aqd_jurist_seats.txt, z_aqd_madrasas.txt,
# script_values/z_aqd_madhhab_counts.txt). Cannot target the realm's own
# current madhhab. Knocks the target down exactly one institutional
# tier (scripted_effects/z_aqd_jurist_effects.txt's aqd_crackdown_effect,
# madrasas destroyed before the seat) rather than wiping it out - the
# 60-month cooldown is what prevents repeated use from erasing a school
# in one sitting.
#
# The action itself has no price. aqd_crackdown_effect applies the cost:
#  - confiscation gold GAINED, scaled to the number of buildings razed
#  - a stability hit scaled to how widespread the school was
#  - "Legal School Persecuted" (z_aqd_jurist_evicted_<madhhab>, the
#    per-madhhab key) on every location nationwide that holds the
#    school's buildings - significant unrest - and, in the same pass,
#    cancellation of that madhhab's own currently-active ḥalqa across
#    each such location's whole province
#  - a national modifier: base game's ulema_displeased for a fellow
#    school, or z_aqd_heretics_suppressed if the school is a heresy of
#    the state religion (SECT param differs from root.religion)
#  - the existing tier-scaled clergy estate satisfaction penalty
# plus the shared aqidah-madhhab relations reaction (OUTCOME = refuse),
# exactly as declining a seat establishment does.
# =====================================================================
aqd_crackdown_madhhab = {
	show_message = yes
	type = religious
	sound = UI_action_religion_generic

	ai_tick = monthly
	ai_tick_frequency = 12
	automation_tick = never

	potential = {
		scope:actor = {
			religion.group = religion_group:muslim
			has_policy = sharia_law_policy
		}
	}
	allow = { }
	cooldown = { type = aqd_crackdown_madhhab  months = 60 }

	select_trigger = {
		looking_for_a = policy
		target_flag = target
		name = "aqd_crackdown_select"
		none_available_msg_key = "aqd_crackdown_none"
		column = { data = name }
		visible = {
			# 'this' = the candidate policy being filtered - must stay unshadowed,
			# so scope:actor is reached explicitly per-clause rather than wrapping
			# the whole OR in `scope:actor = {}` (that rebinds 'this' to a country
			# and breaks every `this = policy:X` identity check - confirmed by
			# "Left side and right side during comparison were of different
			# types (left was 'country', right was 'policy')" in error.log).
			OR = {
				AND = { this = policy:hanafi_policy    NOT = { scope:actor = { has_policy = hanafi_policy } }    scope:actor = { hanafi_institution_tier >= 1 } }
				AND = { this = policy:hanbali_policy   NOT = { scope:actor = { has_policy = hanbali_policy } }   scope:actor = { hanbali_institution_tier >= 1 } }
				AND = { this = policy:maliki_policy    NOT = { scope:actor = { has_policy = maliki_policy } }    scope:actor = { maliki_institution_tier >= 1 } }
				AND = { this = policy:shafii_policy    NOT = { scope:actor = { has_policy = shafii_policy } }    scope:actor = { shafii_institution_tier >= 1 } }
				AND = { this = policy:zahiri_policy    NOT = { scope:actor = { has_policy = zahiri_policy } }    scope:actor = { zahiri_institution_tier >= 1 } }
				AND = { this = policy:ismaili_policy   NOT = { scope:actor = { has_policy = ismaili_policy } }   scope:actor = { ismaili_institution_tier >= 1 } }
				AND = { this = policy:jafari_policy    NOT = { scope:actor = { has_policy = jafari_policy } }    scope:actor = { jafari_institution_tier >= 1 } }
				AND = { this = policy:zaidi_policy     NOT = { scope:actor = { has_policy = zaidi_policy } }     scope:actor = { zaidi_institution_tier >= 1 } }
				AND = { this = policy:ibadi_policy     NOT = { scope:actor = { has_policy = ibadi_policy } }     scope:actor = { ibadi_institution_tier >= 1 } }
			}
		}
	}

	effect = {
		# scope:actor does NOT reliably retain script variables across
		# nested scripted_effect calls in a generic_action (confirmed in
		# error.log: aqd_compat_madhhab / aqd_jurist_compat / aqd_crackdown_*
		# all read back unset when set on scope:actor and read in a later
		# macro). save_scope_as a named country scope once and thread THAT
		# through every call - named scopes persist reliably.
		scope:actor = { save_scope_as = aqd_crackdown_actor }

		# Projected confiscation gold, shown in the action tooltip
		# (script_values/z_aqd_madhhab_counts.txt - wipeout ceiling, so
		# the text says "up to").
		scope:aqd_crackdown_actor = {
			set_variable = { name = aqd_crackdown_gold_est  value = aqd_crackdown_gold_estimate_target }
		}
		custom_tooltip = aqd_crackdown_gold_tt

		# Same aqidah-madhhab relations reaction as declining a seat
		# establishment: cracking down on a school your creed favours
		# costs clergy standing + legitimacy; cracking down on one it
		# opposes is rewarded. (scripted_effects/z_aqd_jurist_effects.txt)
		aqd_set_compat_madhhab_from_target = { COUNTRY = scope:aqd_crackdown_actor }
		scope:aqd_crackdown_actor = {
			if = { limit = { scope:target = policy:hanafi_policy }
				aqd_crackdown_effect = { MADHHAB = hanafi  SECT = sunni  SCORE = hanafi_institution_score  TIER = hanafi_institution_tier }
			}
			else_if = { limit = { scope:target = policy:hanbali_policy }
				aqd_crackdown_effect = { MADHHAB = hanbali  SECT = sunni  SCORE = hanbali_institution_score  TIER = hanbali_institution_tier }
			}
			else_if = { limit = { scope:target = policy:maliki_policy }
				aqd_crackdown_effect = { MADHHAB = maliki  SECT = sunni  SCORE = maliki_institution_score  TIER = maliki_institution_tier }
			}
			else_if = { limit = { scope:target = policy:shafii_policy }
				aqd_crackdown_effect = { MADHHAB = shafii  SECT = sunni  SCORE = shafii_institution_score  TIER = shafii_institution_tier }
			}
			else_if = { limit = { scope:target = policy:zahiri_policy }
				aqd_crackdown_effect = { MADHHAB = zahiri  SECT = sunni  SCORE = zahiri_institution_score  TIER = zahiri_institution_tier }
			}
			else_if = { limit = { scope:target = policy:ismaili_policy }
				aqd_crackdown_effect = { MADHHAB = ismaili  SECT = shia  SCORE = ismaili_institution_score  TIER = ismaili_institution_tier }
			}
			else_if = { limit = { scope:target = policy:jafari_policy }
				aqd_crackdown_effect = { MADHHAB = jafari  SECT = shia  SCORE = jafari_institution_score  TIER = jafari_institution_tier }
			}
			else_if = { limit = { scope:target = policy:zaidi_policy }
				aqd_crackdown_effect = { MADHHAB = zaidi  SECT = shia  SCORE = zaidi_institution_score  TIER = zaidi_institution_tier }
			}
			else_if = { limit = { scope:target = policy:ibadi_policy }
				aqd_crackdown_effect = { MADHHAB = ibadi  SECT = ibadi  SCORE = ibadi_institution_score  TIER = ibadi_institution_tier }
			}
		}
		aqd_apply_jurist_reaction = { OUTCOME = refuse  COUNTRY = scope:aqd_crackdown_actor }
	}
	ai_will_do = { add = 0 }
}
