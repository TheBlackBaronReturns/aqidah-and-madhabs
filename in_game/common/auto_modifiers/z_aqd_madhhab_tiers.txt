# =====================================================================
# AQIDAH SCHOOLS OF CREED - Ulema Institutions: tiered national
# modifiers per madhhab (Presence / Noteworthy / Plurality /
# Predominance), driven by script_values/z_aqd_madhhab_counts.txt's
# per-madhhab institution score (seats=2, madrasas=1).
#
# Predominance also fires automatically if the madhhab is the nation's
# own current sharia_law policy, regardless of score - holding the
# official law IS predominance. The three lower tiers explicitly
# exclude that case (NOT has_policy) so they never overlap with it.
# A madhhab with zero seats/madrasas that is also not the active law
# matches none of the four blocks below, i.e. shows no modifier at
# all - this is intentional, not a missing "tier 0" block.
#
# Thresholds (1/3/5/8) MUST be kept in sync with:
#  - scripted_triggers/z_aqd_jurist_triggers.txt's
#    aqd_is_eligible_jurist_source (Plurality lower bound = 5);
#  - scripted_effects/z_aqd_jurist_effects.txt's aqd_crackdown_effect
#    (all four thresholds);
#  - on_action/00_aqd_game_start.txt's aqd_seat_establishment_pulse and
#    scripted_triggers/z_aqd_jurist_triggers.txt's
#    aqd_province_has_madhhab_density (Noteworthy lower bound = 3, and
#    the "3 madrasas in one province" literal in the density trigger).
# =====================================================================

aqd_hanafi_predominance = {
	potential_trigger = { OR = { hanafi_institution_score >= 8  has_policy = hanafi_policy } }
	monthly_towards_mysticism = societal_value_minor_monthly_move
	global_clergy_estate_power = 0.15
	global_clergy_max_literacy = 10
}
aqd_hanafi_plurality = {
	potential_trigger = { hanafi_institution_score >= 5  hanafi_institution_score < 8  NOT = { has_policy = hanafi_policy } }
	monthly_towards_mysticism = societal_value_minor_monthly_move
	global_clergy_estate_power = 0.10
	global_clergy_max_literacy = 5
}
aqd_hanafi_noteworthy = {
	potential_trigger = { hanafi_institution_score >= 3  hanafi_institution_score < 5  NOT = { has_policy = hanafi_policy } }
	monthly_towards_mysticism = societal_value_minor_monthly_move
	global_clergy_estate_power = 0.05
}
aqd_hanafi_presence = {
	potential_trigger = { hanafi_institution_score >= 1  hanafi_institution_score < 3  NOT = { has_policy = hanafi_policy } }
	monthly_towards_mysticism = societal_value_minor_monthly_move
}
