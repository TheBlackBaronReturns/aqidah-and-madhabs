# aqd_fatwa_evaluate_seat_demand = { MADHHAB }: called at a candidate
# seat's own LOCATION scope, given the canonical 1-9 index of exactly
# which seat there to evaluate - $MADHHAB$ is always either a literal
# (stage 2's per-madhhab loop below) or a variable reference like
# var:aqd_fatwa_seat_madhhab (stage 1, via aqd_fatwa_attempt_stage1) -
# both are confirmed-safe as a substituted parameter value (mirrors
# aqd_classify_seat_compatibility's own SEAT_MADHHAB = var:aqd_seat_
# madhhab call convention, scripted_effects/z_aqd_madhhab_approval_
# effects.txt). Pre-seeds root's var:aqd_seat_madhhab from it before
# calling aqd_compute_seat_approval, whose own guard (scripted_effects/
# z_aqd_madhhab_approval_effects.txt) then skips its has_building auto-
# detection - required because a location can hold more than one seat,
# so guessing "whichever building matches first" could evaluate (and,
# via aqd_fatwa_apply_displeased_for_seat below, punish) an entirely
# different co-located madhhab than the one actually being asked.
# Computes that seat's centralized approval, then INVERTS it before
# rolling - this fatwa asks "does this seat turn AGAINST the ruler", the
# opposite question from the jihad fatwa's direct read of the same
# engine. Leaves var:aqd_madhhab_approval_result readable at root (1 =
# this seat steps up to demand peace, 0 = it stays quiet).
# ---------------------------------------------------------------------
aqd_fatwa_evaluate_seat_demand = {
	root = { set_variable = { name = aqd_seat_madhhab  value = $MADHHAB$ } }
	aqd_compute_seat_approval = yes
	root = {
		aqd_invert_approval_scratch = yes
		aqd_roll_approval_percentage = yes
	}
}

# 9-branch dispatch, same shape and same reasoning as scripted_effects/
# z_aqd_jihad_fatwa_effects.txt's aqd_jihad_fatwa_stamp_declined_for_seat:
# stamps z_aqd_fatwa_declined_<X> (main_menu/common/static_modifiers/
# z_aqd_fatwa_declined.txt) on the SPECIFIC seat that just declined stage
# 1, reading root's var:aqd_fatwa_seat_madhhab rather than re-deriving
# from has_building for the same multi-seat-location reason. Called from
# aqd_fatwa_attempt_stage1's own decline branch, at scope:target_location,
# BEFORE that variable is cleaned up. years = 50 is a safety-net duration
# only - the marker is always cleared explicitly once the unjust-war
# episode ends (on_action/00_aqd_game_start.txt's aqd_fatwa_episode_
# cleanup_pulse). This is the "one shot per war" fix: previously a
# decline cost this seat nothing, so the very next month's pulse could -
# and eventually, given a long enough war, would - pick it again, making
# eventual success for SOME seat in the realm a near-certainty regardless
# of how low any individual seat's own approval score was.
aqd_fatwa_stamp_declined_for_seat = {
	if      = { limit = { root = { var:aqd_fatwa_seat_madhhab = 1 } }  add_location_modifier = { modifier = z_aqd_fatwa_declined_hanafi   years = 50  mode = add_and_extend } }
	else_if = { limit = { root = { var:aqd_fatwa_seat_madhhab = 2 } }  add_location_modifier = { modifier = z_aqd_fatwa_declined_hanbali  years = 50  mode = add_and_extend } }
	else_if = { limit = { root = { var:aqd_fatwa_seat_madhhab = 3 } }  add_location_modifier = { modifier = z_aqd_fatwa_declined_maliki   years = 50  mode = add_and_extend } }
	else_if = { limit = { root = { var:aqd_fatwa_seat_madhhab = 4 } }  add_location_modifier = { modifier = z_aqd_fatwa_declined_shafii   years = 50  mode = add_and_extend } }
	else_if = { limit = { root = { var:aqd_fatwa_seat_madhhab = 5 } }  add_location_modifier = { modifier = z_aqd_fatwa_declined_zahiri   years = 50  mode = add_and_extend } }
	else_if = { limit = { root = { var:aqd_fatwa_seat_madhhab = 6 } }  add_location_modifier = { modifier = z_aqd_fatwa_declined_ismaili  years = 50  mode = add_and_extend } }
	else_if = { limit = { root = { var:aqd_fatwa_seat_madhhab = 7 } }  add_location_modifier = { modifier = z_aqd_fatwa_declined_jafari   years = 50  mode = add_and_extend } }
	else_if = { limit = { root = { var:aqd_fatwa_seat_madhhab = 8 } }  add_location_modifier = { modifier = z_aqd_fatwa_declined_zaidi    years = 50  mode = add_and_extend } }
	else_if = { limit = { root = { var:aqd_fatwa_seat_madhhab = 9 } }  add_location_modifier = { modifier = z_aqd_fatwa_declined_ibadi    years = 50  mode = add_and_extend } }

# aqd_fatwa_episode_start_check: country-scoped, called at the top of the
# pulse's effect (on_action/00_aqd_game_start.txt's aqd_fatwa_pulse) -
# i.e. every month root qualifies as any_current_war = { attacker_leader
# = root }, the episode-boundary proxy (deliberately broader than the
# full aqd_fatwa_core_conditions used inside aqd_fatwa_attempt_stage1
# itself, so the episode doesn't flicker closed/reopen just because CB/
# war-score details drift month to month - the war continuing is what
# defines "still the same episode"). Exactly mirrors scripted_effects/
# z_aqd_jihad_fatwa_effects.txt's aqd_jihad_episode_start_check - see its
# own comment for the full flag-toggle reasoning; var:aqd_fatwa_episode_
# active is this system's half of the same start/end pair with aqd_fatwa_
# episode_cleanup_pulse (same on_action file).
aqd_fatwa_episode_start_check = {
	if = {
		limit = { NOT = { has_variable = aqd_fatwa_episode_active } }
		set_variable = { name = aqd_fatwa_episode_active  value = 1 }
		every_owned_location = {
			remove_location_modifier = z_aqd_fatwa_declined_hanafi
			remove_location_modifier = z_aqd_fatwa_declined_hanbali
			remove_location_modifier = z_aqd_fatwa_declined_maliki
			remove_location_modifier = z_aqd_fatwa_declined_shafii
			remove_location_modifier = z_aqd_fatwa_declined_zahiri
			remove_location_modifier = z_aqd_fatwa_declined_ismaili
			remove_location_modifier = z_aqd_fatwa_declined_jafari
			remove_location_modifier = z_aqd_fatwa_declined_zaidi
			remove_location_modifier = z_aqd_fatwa_declined_ibadi
		}
	}
}

# ---------------------------------------------------------------------
# aqd_fatwa_attempt_stage1: country-scoped, called from the pulse's
# effect (on_action/00_aqd_game_start.txt's aqd_fatwa_pulse), NOT from an
# event - the silent-miss design mirrors the jihad fatwa's own aqd_jihad_
# fatwa_attempt exactly. The full aqd_fatwa_core_conditions is checked
# here (not just the pulse's own cheap prefilter) since there is no
# event trigger left to lean on before deciding whether to pop anything;
# z_aqd_fatwa.1's own trigger still re-checks it a second time at fire
# time as a defensive re-validation, same as before this change.
#
# Feeds aqd_fatwa_pick_seat's resolved var:aqd_fatwa_seat_madhhab
# straight into aqd_fatwa_evaluate_seat_demand's $MADHHAB$ parameter -
# this variable, unlike a saved scope, survives the trigger_event below,
# so z_aqd_fatwa.1's own refuse option can apply the ban to the correct
# specific seat instead of re-guessing from has_building.
# ---------------------------------------------------------------------
aqd_fatwa_attempt_stage1 = {
	if = {
		limit = { aqd_fatwa_core_conditions = yes }
		aqd_fatwa_pick_seat = yes
		if = {
			limit = { exists = scope:target_location }
			scope:target_location = { aqd_fatwa_evaluate_seat_demand = { MADHHAB = var:aqd_fatwa_seat_madhhab } }
			if = {
				limit = { var:aqd_madhhab_approval_result = 1 }
				scope:target_location = { aqd_stamp_pending_pick = yes }
				trigger_event_non_silently = { id = z_aqd_fatwa.1  days = 1 }
			}
			else = {
				scope:target_location = { aqd_fatwa_stamp_declined_for_seat = yes }
				remove_variable = aqd_fatwa_seat_madhhab
			}
		}
	}
	remove_variable = aqd_seat_madhhab
	remove_variable = aqd_seat_compat
	remove_variable = aqd_approval_scratch
	remove_variable = aqd_madhhab_approval_result
}

# ---------------------------------------------------------------------
# aqd_fatwa_apply_displeased_for_seat = { YEARS  MADHHAB  LOCATION }:
# given the canonical 1-9 index of exactly which seat to punish and the
# location to punish it at, applies that seat's ban. NO LONGER branches
# on has_building - a location can hold more than one seat, so re-
# deriving "which one" from building presence here could stamp the ban
# on an entirely different co-located madhhab than the one that actually
# refused (mirrors scripted_effects/z_aqd_jihad_fatwa_effects.txt's
# aqd_apply_fatwa_of_jihad_for_seat, fixed the same way for the same
# reason).
#
# MUST be called from a scope where a bare $MADHHAB$ resolves correctly
# - always root, for both call sites (events/religion/z_aqd_fatwa_events.
# txt's z_aqd_fatwa.1 refuse option passes its own var:aqd_fatwa_seat_
# madhhab directly since that option already runs at root by default;
# z_aqd_fatwa.2's refuse option is inside a location-scoped every_
# location_in_province_definition loop, so IT wraps its own call in
# root = {...} and passes a literal 1-9 instead). NEVER call this from
# inside $LOCATION$ = {...} itself - unlike a literal, $MADHHAB$ = var:
# aqd_fatwa_seat_madhhab needs the real root object to find the variable
# on, and this function does not root-wrap the comparison itself (it
# can't - the same $MADHHAB$ token also has to work for a bare literal,
# and wrapping a literal in root = { 3 = 1 } is meaningless). $LOCATION$
# is a scope reference (e.g. scope:target_location, or a saved loop
# scope) re-entered only for the actual modifier calls below - confirmed-
# safe "pass a scope, re-enter via $PARAM$ = {}" idiom, mirrors aqd_
# classify_seat_compatibility's own $COUNTRY$ = {...} convention
# (scripted_effects/z_aqd_madhhab_approval_effects.txt).
#
# Per madhhab:
#   - applies/extends THAT location's own madhhab-specific fatwa ban to
#     $YEARS$ (main_menu/common/static_modifiers/
#     z_aqd_fatwa_displeased.txt);
#   - clears ONLY that same madhhab's own currently-active ḥalqa (its 3
#     class-variant keys), leaving any other madhhab's circle elsewhere
#     in the province untouched - the fatwa is madhhab-specific, so its
#     consequences must be too.
# Deliberately NOT a 28-key blanket sweep (contrast
# zz_aqd_debug_effects.txt's aqd_debug_clear_halqa, which is fine for a
# debug reset but would be wrong here: wiping every madhhab's circle
# would undo the whole point of tracking these bans per-madhhab).
# ---------------------------------------------------------------------
aqd_fatwa_apply_displeased_for_seat = {
	if = {
		limit = { $MADHHAB$ = 1 }
		$LOCATION$ = {
