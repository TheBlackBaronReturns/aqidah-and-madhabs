# Bare random_owned_location picker for the seat issuing the fatwa -
# structurally mirrors events/religion/z_aqd_halqa_events.txt's
# z_aqd_halqa.1 picker, but simpler: no "no active halqa anywhere in
# the province" / density gating here, since this feature's own
# occupancy is tracked entirely by its 9 concrete
# z_aqd_fatwa_displeased_<madhhab> keys (scripted_triggers/
# z_aqd_fatwa_triggers.txt's aqd_location_has_any_fatwa_ban), not by
# aqd_location_has_active_halqa - the two systems are deliberately kept
# independent (see that trigger file's own header note on why).
aqd_fatwa_pick_seat = {
	random_owned_location = {
		# aqd_seat_madhhab_not_heterodox_to_root (scripted_triggers/
		# z_aqd_fatwa_triggers.txt) REPLACES the old bare 9-way has_building
		# OR that used to live here - every one of its branches already
		# requires has_building = seat_of_<X>_jurists, so it subsumes "does
		# this location hold any seat at all" while additionally excluding
		# heterodox seats from ever being picked to demand peace.
		limit = { aqd_seat_madhhab_not_heterodox_to_root = yes }
		save_scope_as = target_location
		province_definition = { save_scope_as = target_province }
	}
}

# ---------------------------------------------------------------------
# aqd_fatwa_evaluate_seat_demand: called at a candidate seat's own scope.
# Computes that seat's centralized approval (scripted_effects/
# z_aqd_madhhab_approval_effects.txt's aqd_compute_seat_approval), then
# INVERTS it before rolling - this fatwa asks "does this seat turn
# AGAINST the ruler", the opposite question from the jihad fatwa's direct
# read of the same engine. Leaves var:aqd_madhhab_approval_result
# readable at root (1 = this seat steps up to demand peace, 0 = it stays
# quiet).
# ---------------------------------------------------------------------
aqd_fatwa_evaluate_seat_demand = {
	aqd_compute_seat_approval = yes
	root = {
		aqd_invert_approval_scratch = yes
		aqd_roll_approval_percentage = yes
	}
}

# ---------------------------------------------------------------------
# aqd_fatwa_attempt_stage1: country-scoped, called from the pulse's
# effect (on_action/00_aqd_game_start.txt's aqd_fatwa_pulse), NOT from an
# event - the silent-miss design mirrors the jihad fatwa's own aqd_jihad_
# fatwa_attempt exactly. The full aqd_fatwa_core_conditions is checked
# here (not just the pulse's own cheap prefilter) since there is no
# event trigger left to lean on before deciding whether to pop anything;
# z_aqd_fatwa.1's own trigger still re-checks it a second time at fire
# time as a defensive re-validation, same as before this change.
# ---------------------------------------------------------------------
aqd_fatwa_attempt_stage1 = {
	if = {
		limit = { aqd_fatwa_core_conditions = yes }
		aqd_fatwa_pick_seat = yes
		if = {
			limit = { exists = scope:target_location }
			scope:target_location = { aqd_fatwa_evaluate_seat_demand = yes }
			if = {
				limit = { var:aqd_madhhab_approval_result = 1 }
				scope:target_location = { aqd_stamp_pending_pick = yes }
				trigger_event_non_silently = { id = z_aqd_fatwa.1  days = 1 }
			}
		}
	}
	remove_variable = aqd_seat_madhhab
	remove_variable = aqd_seat_compat
	remove_variable = aqd_approval_scratch
	remove_variable = aqd_madhhab_approval_result
}

# ---------------------------------------------------------------------
# aqd_fatwa_apply_displeased_for_seat = { YEARS }: called at a seat-
# holding location's own scope (e.g. `scope:target_location = { ... }`,
# or bare inside an every_location_in_province_definition loop body).
# Branches on which seat building is actually present (mirrors this
# mod's existing 9-concrete-branch style, e.g.
# aqd_is_eligible_jurist_source in scripted_triggers/
# z_aqd_jurist_triggers.txt) and:
#   - applies/extends THAT location's own madhhab-specific fatwa ban to
#     $YEARS$ (main_menu/common/static_modifiers/
#     z_aqd_fatwa_displeased.txt);
#   - clears ONLY that same madhhab's own currently-active ḥalqa (its 3
#     class-variant keys), leaving any other madhhab's circle elsewhere
#     in the province untouched - the fatwa is madhhab-specific, so its
#     consequences must be too.
# Deliberately NOT a 28-key blanket sweep (contrast
# zz_aqd_debug_effects.txt's aqd_debug_clear_halqa, which is fine for a
# debug reset but would be wrong here: wiping every madhhab's circle
# would undo the whole point of tracking these bans per-madhhab).
# ---------------------------------------------------------------------
aqd_fatwa_apply_displeased_for_seat = {
	if = {
		limit = { has_building = building_type:seat_of_hanafi_jurists }
		add_location_modifier = { modifier = z_aqd_fatwa_displeased_hanafi  years = $YEARS$  mode = add_and_extend }
