# aqd_jihad_fatwa_pick_seat: picks ONE random eligible owned location
# (uniform over locations, via the umbrella aqd_jihad_seat_eligible OR,
# scripted_triggers/z_aqd_jihad_fatwa_triggers.txt), then - since that
# location can legitimately hold more than one eligible seat at once
# (building_types/z_aqd_jurist_seats.txt's location_potential for each
# seat_of_<X>_jurists only ever excludes a duplicate of the SAME
# madhhab, never a different one, and the establishment event fires
# independently per madhhab with no check for another madhhab's seat
# already standing there) - picks uniformly among however many of its
# seats are ACTUALLY eligible right now via classic reservoir sampling
# (Algorithm R): walk the 9 madhhabs in a fixed order, and each time an
# eligible one is found, replace the current pick with probability
# 1/(running count so far). EU5's random_list only accepts literal
# integer weights (script_values/z_aqd_madhhab_approval.txt's own header
# notes the identical constraint), so the promotion odds are hardcoded
# per possible running-count value (1 through the position's own index -
# the most that count could ever be there, since at most that many
# madhhabs could have preceded it) rather than computed - this still
# yields an exactly uniform pick over N eligible seats for any N from 1
# to 9, regardless of which specific madhhabs turned out to be present.
#
# Sets var:aqd_jihad_fatwa_seat_madhhab (1-9) to the winner.
# aqd_jihad_fatwa_attempt below reads it straight into aqd_compute_seat_
# approval (scripted_effects/z_aqd_madhhab_approval_effects.txt, which
# skips its own has_building auto-detection whenever this is already
# set) instead of letting that effect re-guess from building presence -
# re-guessing would just reintroduce the same "first match in a fixed
# priority order always wins, even when a different co-located seat is
# the one that actually got picked" bug this whole file exists to fix.
# ---------------------------------------------------------------------

# 9-branch dispatch, mirrors scripted_effects/z_aqd_fatwa_effects.txt's
# aqd_fatwa_apply_displeased_for_seat in shape. Called from the event's
# own option (not immediate) so the manpower/modifier grant renders in
# the option's tooltip preview. Reads root's var:aqd_jihad_fatwa_seat_
# madhhab (set by the picker above, still alive - aqd_jihad_fatwa_attempt
# only cleans it up on a declined roll, and the event's own option below
# removes it once consumed) rather than re-deriving from has_building -
# a location can hold more than one seat, so re-deriving here would risk
# stamping the WRONG madhhab's modifier if a different one happens to be
# checked first.
#
# Each branch now stamps TWO modifiers, not one:
#   - z_aqd_fatwa_of_jihad_<X> (main_menu/common/static_modifiers/z_aqd_
#     fatwa_of_jihad.txt) - the real manpower/levy mobilization bonus.
#     years = 50 here is a safety-net duration only (see that file's own
#     header) - its REAL post-success cross-war throttle role is
#     unaffected by this change, since nothing clears it early any more
#     than it always could have naturally expired before.
#   - z_aqd_jihad_blessed_<X> (main_menu/common/static_modifiers/z_aqd_
#     jihad_blessed.txt) - the war-scoped "don't ask this seat again"
#     gate, checked by scripted_triggers/z_aqd_jihad_fatwa_triggers.txt's
#     aqd_jihad_seat_eligible_<X> and cleared at episode end (on_action/
#     00_aqd_game_start.txt's aqd_jihad_episode_cleanup_pulse) - this is
#     what actually stops "A Fatwa for the Jihad" from popping up again
#     for the same seat every time the mobilization bonus happens to
#     lapse before the war itself does.
aqd_apply_fatwa_of_jihad_for_seat = {
	if = {
		limit = { root = { var:aqd_jihad_fatwa_seat_madhhab = 1 } }
		add_location_modifier = { modifier = z_aqd_fatwa_of_jihad_hanafi  years = 50  mode = add_and_extend }
		add_location_modifier = { modifier = z_aqd_jihad_blessed_hanafi   years = 50  mode = add_and_extend }
	}
	else_if = {
		limit = { root = { var:aqd_jihad_fatwa_seat_madhhab = 2 } }
		add_location_modifier = { modifier = z_aqd_fatwa_of_jihad_hanbali  years = 50  mode = add_and_extend }

	}
	else_if = {
		limit = { root = { var:aqd_jihad_fatwa_seat_madhhab = 5 } }

# 9-branch dispatch, same shape and same reasoning as aqd_apply_fatwa_of_
# jihad_for_seat above, for the OPPOSITE outcome: stamps z_aqd_jihad_
# declined_<X> (main_menu/common/static_modifiers/z_aqd_jihad_declined.
# txt) on the SPECIFIC seat that just declined, reading root's var:aqd_
# jihad_fatwa_seat_madhhab rather than re-deriving from has_building for
# the same multi-seat-location reason. Called from aqd_jihad_fatwa_
# attempt's own decline branch, at scope:target_location, BEFORE that
# variable is cleaned up. years = 50 is a safety-net duration only - the
# marker is always cleared explicitly once the jihad episode ends
# (on_action/00_aqd_game_start.txt's aqd_jihad_episode_cleanup_pulse).
# This is the "one shot per war" fix: previously a decline cost this
# seat nothing, so the very next month's pulse could - and eventually,
# given a long enough war, would - pick it again, making eventual
# success for SOME seat in the realm a near-certainty regardless of how
# low any individual seat's own approval score was.

	if      = { limit = { root = { var:aqd_jihad_fatwa_seat_madhhab = 1 } }  add_location_modifier = { modifier = z_aqd_jihad_declined_hanafi   years = 50  mode = add_and_extend } }

# aqd_jihad_episode_start_check: country-scoped, called at the top of
# the pulse's effect (on_action/00_aqd_game_start.txt's aqd_jihad_fatwa_
# pulse) - i.e. every month root qualifies as EITHER aqd_is_jihad_
# participant OR aqd_is_defensive_jihad_participant (scripted_triggers/
# z_aqd_jihad_fatwa_triggers.txt) - offensive and defensive jihads share
# one episode-boundary flag, so switching from one to the other (or being
# in both at once) never itself resets or re-triggers this sweep.
# var:aqd_jihad_episode_active is the flag-toggle half of a start/end
# pair with aqd_jihad_episode_cleanup_pulse (same on_action file): the
# FIRST month of a fresh jihad (flag not set yet) sweeps every owned
# location clear of all nine z_aqd_jihad_declined_<X> AND all nine
# z_aqd_jihad_blessed_<X> markers before setting the flag, so a seat that
# declined OR already blessed a PREVIOUS, already-concluded jihad starts
# this new one with a clean slate. Deliberately does NOT touch z_aqd_
# fatwa_of_jihad_<X> itself - that one's real cross-war throttle role is
# untouched by episode boundaries on purpose. On every subsequent month
# of the SAME ongoing jihad the flag is already set, so this is a single
# cheap has_variable check and nothing else runs - the sweep only ever
# fires once per episode. The actual "already asked this seat, one way or
# the other" cost is meant to persist ONLY for the remainder of one war;

	if = {
		limit = { NOT = { has_variable = aqd_jihad_episode_active } }

		every_owned_location = {
			remove_location_modifier = z_aqd_jihad_declined_hanafi

			remove_location_modifier = z_aqd_jihad_declined_zaidi

# ---------------------------------------------------------------------
# Country-scoped. Called from the pulse's effect (on_action/
# 00_aqd_game_start.txt's aqd_jihad_fatwa_pulse), NOT from an event -
# this is the silent-miss design: the roll happens here, before any
# popup exists, so a declining seat produces no event at all this month.
# Runs identically regardless of WHICH of the two participation paths
# (scripted_triggers/z_aqd_jihad_fatwa_triggers.txt's aqd_is_jihad_
# participant or aqd_is_defensive_jihad_participant) is what let the
# pulse fire this month - this effect never needs to know or care which
# one it was, since the only path-dependent behaviour (heterodox seats
# allowed only when defending) already lives entirely inside aqd_jihad_
# seat_eligible_<X>, upstream of everything here.
# On success, stamps the shared pending-pick marker (scripted_effects/
# z_aqd_madhhab_approval_effects.txt's aqd_stamp_pending_pick) and fires
# the narration event non-silently; the event's own immediate re-finds
# the seat via that marker (saved scopes do not survive trigger_event -
# var:aqd_jihad_fatwa_seat_madhhab, a plain country variable rather than
# a saved scope, DOES survive it, which is exactly why the event reads
# that instead of re-deriving the madhhab from the re-found location's
# buildings). On failure - no eligible seat at all, or the roll declines
# - this does nothing, a true silent miss.
#
# NO country-wide cooldown here any more - a per-seat one already exists
# and is enough: a successful pick stamps a 4-year z_aqd_fatwa_of_jihad_
# <X> modifier on that SPECIFIC (location, madhhab) pair (scripted_
# triggers/z_aqd_jihad_fatwa_triggers.txt's aqd_jihad_seat_eligible_<X>
# checks it directly), so THAT seat can't fire again for 4 years while
# every OTHER eligible seat - a different madhhab at the same location
# included - stays independently pickable next month. A blanket root-
# scoped cooldown here would have defeated that: it used to block the
# entire pulse for 5 years after any single success, regardless of which
# seat or madhhab, which is why a whole jihad war (usually well under 5
# years) only ever produced one fatwa event no matter how many other
# eligible seats existed.
# ---------------------------------------------------------------------
aqd_jihad_fatwa_attempt = {
	aqd_jihad_fatwa_pick_seat = yes
	if = {
		limit = { exists = scope:target_location }
		# Pre-seed aqd_compute_seat_approval's own var:aqd_seat_madhhab
		# from the reservoir pick's result BEFORE calling it - the two
		# variables are deliberately named differently (aqd_seat_madhhab
		# is the shared scratch var scripted_effects/z_aqd_madhhab_
		# approval_effects.txt's aqd_compute_seat_approval and the white-
		# peace fatwa both read/clear every call; aqd_jihad_fatwa_seat_
		# madhhab is this file's own persistent result that survives all
		# the way to the event). Without this copy, aqd_compute_seat_
		# approval's "skip auto-detection if already set" guard would
		# never trigger (it checks aqd_seat_madhhab specifically), and it
		# would silently fall back to re-guessing the seat's madhhab from
		# has_building - exactly the ambiguous-at-a-multi-seat-location
		# behavior this whole file exists to avoid.
		set_variable = { name = aqd_seat_madhhab  value = var:aqd_jihad_fatwa_seat_madhhab }
		scope:target_location = { aqd_compute_seat_approval = yes }
		aqd_roll_approval_percentage = yes
		if = {
			limit = { var:aqd_madhhab_approval_result = 1 }
			scope:target_location = { aqd_stamp_pending_pick = yes }
			trigger_event_non_silently = { id = z_aqd_jfatwa.1  days = 1 }
		}
		else = {
			scope:target_location = { aqd_jihad_fatwa_stamp_declined_for_seat = yes }
			remove_variable = aqd_jihad_fatwa_seat_madhhab
		}
	}
	remove_variable = aqd_seat_madhhab
	remove_variable = aqd_seat_compat
	remove_variable = aqd_approval_scratch
	remove_variable = aqd_madhhab_approval_result
}
