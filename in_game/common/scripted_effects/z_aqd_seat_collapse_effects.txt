# =====================================================================
# AQIDAH SCHOOLS OF CREED - Seat Collapse: effects.
#
# Replaces building_types/z_aqd_jurist_seats.txt's own (now-removed)
# remove_if - the mod, not the engine, now owns "what happens when a
# seat's sect population collapses below 5%". Reads the identical
# condition (scripted_triggers/z_aqd_seat_collapse_triggers.txt's
# aqd_seat_collapsing_<X>) every month via on_action/00_aqd_game_start.
# txt's aqd_seat_collapse_pulse, with NO chance gate on whether
# resolution happens at all - unlike every other pulse in this mod, this
# must fire the very month the condition is met, exactly as remove_if
# used to. The only roll here is whether an eligible neighbour - any
# madhhab whose own sect clears 25% of the province, see aqd_seat_
# collapse_find_vacuum_candidate below - fills the vacuum instead of
# losing the institution outright.
# =====================================================================

# ---------------------------------------------------------------------

# reservoir-samples which of its (possibly several) collapsing madhhabs
# is the one being resolved - identical Algorithm-R idiom to
# aqd_jihad_fatwa_pick_seat (scripted_effects/z_aqd_jihad_fatwa_effects.
# txt). Only ONE per tick, deliberately - see aqd_seat_collapse_pulse's
# own header (on_action/00_aqd_game_start.txt) for why resolving more
# than one collapse in the same tick is not attempted.
#
# Also captures the seat's own area (area = { save_scope_as = ... }, the
# confirmed location -> area scope transition idiom already used by
# events/religion/z_aqd_fatwa_events.txt's z_aqd_fatwa.2 and scripted_

# find_vacuum_candidate below has it ready without a second lookup.
#
# Sets scope:collapse_location, scope:collapse_area, var:aqd_collapse_
# madhhab (1-9).

# aqd_seat_collapse_find_vacuum_candidate: given the collapsing seat's
# own madhhab (var:aqd_collapse_madhhab, scope:collapse_location, scope:
# collapse_area from aqd_seat_collapse_pick above), looks for a madhhab
# to "inherit" the vacancy - REVISED from the original same-SECT-only
# design (see the mod's own commit history) to the final agreed rule:
# ANY of the other 8 madhhabs is eligible, sect-relationship to the
# collapsing one irrelevant, provided (a) it isn't the exact same
# madhhab that just collapsed and (b) its own sect's general population
# exceeds 25% of the collapsing seat's own province ("religion_
# percentage" confirmed valid at province scope: base game's own common/
# attribute_columns/20_province.txt evaluates it directly on a province
# row). No more sect-group outer dispatch, no more heterodoxy check -
# scripted_triggers/z_aqd_vacancy_fill_triggers.txt's aqd_vacancy_
# candidate_base_eligible_<X> (NOT aqd_debate_seat_eligible_<X> any
# more - that one still carries a heterodoxy check this feature no
# longer wants) plus the population line is the whole rule, applied
# uniformly to all 8.
#
# The pick itself is now WEIGHTED by the same madhhab-approval engine
# the jihad/unjust-war fatwas use (aqd_compute_seat_approval / aqd_roll_
# approval_percentage, scripted_effects/z_aqd_madhhab_approval_effects.
# txt) rather than a flat uniform draw - full reasoning, including why a
# clean "pick proportional to score" draw isn't possible given random_
# list's literal-weights-only constraint, and why this uses "every
# eligible candidate gets its own independent roll, draw uniformly among
# the survivors" instead of "first success wins," lives in scripted_
# effects/z_aqd_crackdown_vacancy_effects.txt's aqd_crackdown_pick_
# vacancy_candidate - the identical technique, just keyed to this
# feature's own var:aqd_collapse_madhhab/scope:collapse_location/scope:
# collapse_area instead of that one's crackdown-specific names.
#
# Sets var:aqd_vacuum_fill_madhhab (1-9) only if at least one candidate
# survives its own roll. A second, purely cosmetic step then picks ONE
# location actually holding that winning madhhab's own eligible seat in
# the area (scope:vacuum_candidate_location) - guaranteed to succeed,
# since the winning madhhab's own existence check already confirmed at
# least one such location exists - used only so the eventual z_aqd_
# seat_collapse.2 event can still name a specific city, not a mechanical
# input to who actually wins.
# ---------------------------------------------------------------------
aqd_seat_collapse_find_vacuum_candidate = {
	set_variable = { name = aqd_vacuum_survivor_count  value = 0 }

	if = {
		limit = {
			NOT = { var:aqd_collapse_madhhab = 1 }
			scope:collapse_area = { any_location_in_area = { owner ?= root  aqd_vacancy_candidate_base_eligible_hanafi = yes } }
			scope:collapse_location = { province_definition = { "religion_percentage(religion:sunni)" > 0.25 } }
		}

		aqd_roll_approval_percentage = yes

			limit = { var:aqd_madhhab_approval_result = 1 }

		}

# ---------------------------------------------------------------------
# aqd_seat_collapse_attempt: country-scoped, called from the pulse's
# effect (on_action/00_aqd_game_start.txt's aqd_seat_collapse_pulse) -
# NOT from an event. The pick, the candidate search, and the roll all
# happen here, before any event exists - same silent-pre-roll idiom as
# every other fatwa system in this mod, so the event that eventually
# fires already has its outcome fully decided.
#
# No flat chance here any more - "does the institution survive at all"
# is now entirely a question of whether aqd_seat_collapse_find_vacuum_
# candidate finds a survivor (each eligible candidate weighted through
# the real madhhab-approval engine, see that effect's own header). A
# maximally-opposed candidate still keeps a real, non-zero shot (the
# approval engine's own 5-95 clamp guarantees at least a 10% roll), so
# "no eligible candidate exists at all" is the only path that's ever a
# guaranteed loss.
# ---------------------------------------------------------------------
aqd_seat_collapse_attempt = {
	aqd_seat_collapse_pick = yes
	if = {
		limit = {
			exists = scope:collapse_location
			exists = var:aqd_collapse_madhhab
		}
		aqd_seat_collapse_find_vacuum_candidate = yes
		if = {
			limit = {
				exists = scope:vacuum_candidate_location
				exists = var:aqd_vacuum_fill_madhhab
			}
			# No outer chance roll here any more - aqd_seat_collapse_find_
			# vacuum_candidate's own weighted survival mechanism (each
			# eligible candidate independently rolled through the real
			# madhhab-approval engine) already IS the "small chance, more
			# likely for a favoured madhhab" gate. Layering a further flat
			# roll on top here would double-dilute it - a strongly favoured
			# candidate could survive its own ~90% roll and still lose to an
			# unrelated coin flip, which defeats the point of weighting by
			# approval at all. Finding a survivor now means the fill happens.
			scope:collapse_location = { add_location_modifier = { modifier = z_aqd_seat_collapse_pending  months = 1 } }
			scope:vacuum_candidate_location = { add_location_modifier = { modifier = z_aqd_vacuum_fill_pending  months = 1 } }
			trigger_event_non_silently = { id = z_aqd_seat_collapse.2  days = 1 }
		}
		else = {
			scope:collapse_location = { add_location_modifier = { modifier = z_aqd_seat_collapse_pending  months = 1 } }
			trigger_event_non_silently = { id = z_aqd_seat_collapse.1  days = 1 }
		}
	}
}
