# =====================================================================
# AQIDAH SCHOOLS OF CREED - Ulema Institutions: shared effects.
#
# Parameters are text substitution (see base game's own scripted_effects/
# readme.txt) - $OUTCOME$ (accept/refuse) is used to build a dynamic
# effect name (aqd_jurist_reaction_$OUTCOME$_favored/_opposed), never
# compared inside a trigger directly, since text substitution cannot
# satisfy a trigger comparison the way a real value can.
#
# Canonical madhhab index (var:aqd_compat_madhhab, var:aqd_establish_madhhab):
#   1 hanafi  2 hanbali  3 maliki  4 shafii  5 zahiri
#   6 ismaili 7 jafari   8 zaidi   9 ibadi
# =====================================================================

# ---------------------------------------------------------------------
# Foreign-jurist madrasa seeding. Called from events/religion/
# z_aqd_jurist_events.txt's accept option with SOURCE = the chosen
# neighbour country scope. Branches on the SOURCE's held madhhab
# policy, picks one qualifying province of root, and plants 1-3
# madrasas of that madhhab INSIDE THAT ONE PROVINCE so the density can
# accumulate toward the establishment threshold (3 madrasas + a
# mosque) without conquest.
#
# change_building_level_in_location (not construct_building): the
# madrasa's location_potential OR-branch is false for a foreign
# madhhab with no owned seat in the area, so construct_building would
# silently place nothing. The raw level-delta bypasses
# location_potential / allow / country_potential.
# ---------------------------------------------------------------------

# ---------------------------------------------------------------------
# Set var:aqd_compat_madhhab (canonical index 1..9) on $COUNTRY$.
#
#  ..._from_country { SOURCE COUNTRY } - from a source country's held
#     sharia_law madhhab policy (foreign-jurist event).
#  ..._from_target  { COUNTRY }        - from scope:target, the policy
#     picked in a generic_action's select_trigger (crackdown action).
#
# The establishment event sets the var directly from
# var:aqd_establish_madhhab. All callers then run
# aqd_apply_jurist_reaction { OUTCOME COUNTRY } so the aqidah-judged
# clergy/legitimacy reaction is identical across every path.
# ---------------------------------------------------------------------
aqd_set_compat_madhhab_from_country = {
	$COUNTRY$ = {
		if      = { limit = { $SOURCE$ = { has_policy = hanafi_policy } }  set_variable = { name = aqd_compat_madhhab  value = 1 } }
		else_if = { limit = { $SOURCE$ = { has_policy = hanbali_policy } } set_variable = { name = aqd_compat_madhhab  value = 2 } }
		else_if = { limit = { $SOURCE$ = { has_policy = maliki_policy } }  set_variable = { name = aqd_compat_madhhab  value = 3 } }
		else_if = { limit = { $SOURCE$ = { has_policy = shafii_policy } }  set_variable = { name = aqd_compat_madhhab  value = 4 } }
		else_if = { limit = { $SOURCE$ = { has_policy = zahiri_policy } }  set_variable = { name = aqd_compat_madhhab  value = 5 } }
		else_if = { limit = { $SOURCE$ = { has_policy = ismaili_policy } } set_variable = { name = aqd_compat_madhhab  value = 6 } }
		else_if = { limit = { $SOURCE$ = { has_policy = jafari_policy } }  set_variable = { name = aqd_compat_madhhab  value = 7 } }
		else_if = { limit = { $SOURCE$ = { has_policy = zaidi_policy } }   set_variable = { name = aqd_compat_madhhab  value = 8 } }
		else_if = { limit = { $SOURCE$ = { has_policy = ibadi_policy } }   set_variable = { name = aqd_compat_madhhab  value = 9 } }
	}
}

aqd_set_compat_madhhab_from_target = {
	$COUNTRY$ = {
		if      = { limit = { scope:target = policy:hanafi_policy }  set_variable = { name = aqd_compat_madhhab  value = 1 } }

# ---------------------------------------------------------------------
# Aqidah reaction classification. Sets var:aqd_jurist_compat on
# $COUNTRY$ to 1 (favored), -1 (opposed), or leaves it 0 (neutral - no
# branch below matches, which is also din_i_ilahi's permanent state: a
# syncretic creed with no fiqh loyalty, deliberately given no branch at
# all). Reads var:aqd_compat_madhhab (canonical index of the madhhab in
# question) - callers set it first (aqd_set_compat_madhhab_from_country
# / _from_target, or a direct set_variable from var:aqd_establish_madhhab).

aqd_classify_madhhab_compatibility = {

		set_variable = { name = aqd_jurist_compat  value = 0 }

				set_variable = { name = aqd_jurist_compat  value = 1 }
			}
			else_if = {
				limit = { OR = { var:aqd_compat_madhhab = 5  var:aqd_compat_madhhab = 6  var:aqd_compat_madhhab = 7  var:aqd_compat_madhhab = 8  var:aqd_compat_madhhab = 9 } }

				limit = { OR = { var:aqd_compat_madhhab = 1  var:aqd_compat_madhhab = 2  var:aqd_compat_madhhab = 3  var:aqd_compat_madhhab = 4  var:aqd_compat_madhhab = 5 } }
				set_variable = { name = aqd_jurist_compat  value = -1 }
			}
		}
		else_if = {
			limit = { religious_school = religious_school:alevism_school }
			if = {
				limit = { var:aqd_compat_madhhab = 7 }
				set_variable = { name = aqd_jurist_compat  value = 1 }
			}
			else_if = {
				limit = { OR = { var:aqd_compat_madhhab = 1  var:aqd_compat_madhhab = 2  var:aqd_compat_madhhab = 3  var:aqd_compat_madhhab = 4  var:aqd_compat_madhhab = 5 } }
				set_variable = { name = aqd_jurist_compat  value = -1 }
			}
		}
		else_if = {
			limit = { religious_school = religious_school:alawi_school }
			if = {
				limit = { var:aqd_compat_madhhab = 7 }
				set_variable = { name = aqd_jurist_compat  value = 1 }

				set_variable = { name = aqd_jurist_compat  value = -1 }

		}
		else_if = {

			if = {

				set_variable = { name = aqd_jurist_compat  value = 1 }

# ---------------------------------------------------------------------
# Apply the accept/refuse consequence on $COUNTRY$. $OUTCOME$ = accept
# or refuse (text-substituted into the effect name below, per base
# game's own scripted_effects/readme.txt "argumented effect" idiom).
# Reads var:aqd_compat_madhhab (set by the caller) via
# aqd_classify_madhhab_compatibility.
#
#   accept + favored : you welcomed a school your creed favours   -> clergy + legitimacy bonus
#   accept + opposed : you welcomed a school your creed opposes    -> clergy + legitimacy penalty
#   refuse + favored : you moved against a school your creed favours -> clergy + legitimacy penalty
#   refuse + opposed : you moved against a school your creed opposes  -> clergy + legitimacy bonus
#
# Both the foreign-jurist event (accept/refuse), the establishment
# event (allow = accept / crack down = refuse) and the crackdown
# generic action (refuse) run this, so the aqidah-judged reaction is
# identical on every hostile-or-welcoming act toward a madhhab.
# ---------------------------------------------------------------------
aqd_apply_jurist_reaction = {
	aqd_classify_madhhab_compatibility = { COUNTRY = $COUNTRY$ }
	if = {
		limit = { $COUNTRY$ = { var:aqd_jurist_compat = 1 } }
		aqd_jurist_reaction_$OUTCOME$_favored = { COUNTRY = $COUNTRY$ }
	}
	else_if = {
		limit = { $COUNTRY$ = { var:aqd_jurist_compat = -1 } }
		aqd_jurist_reaction_$OUTCOME$_opposed = { COUNTRY = $COUNTRY$ }
	}
}

aqd_jurist_reaction_accept_favored = {
	$COUNTRY$ = {
		add_estate_satisfaction = { type = estate_type:clergy_estate  value = estate_satisfaction_weak_bonus }
		add_legitimacy = legitimacy_weak_bonus
	}
}
aqd_jurist_reaction_accept_opposed = {
	$COUNTRY$ = {
		add_estate_satisfaction = { type = estate_type:clergy_estate  value = estate_satisfaction_weak_penalty }
		add_legitimacy = legitimacy_weak_penalty
	}
}

}

	}

# ---------------------------------------------------------------------
# Crackdown: destroy just enough of $MADHHAB$'s buildings (madrasas
# first, then the seat) to drop $SCORE$ below the current tier's lower
# bound - thresholds here (8/5/3/1) MUST match auto_modifiers/
# z_aqd_madhhab_tiers.txt, scripted_triggers/z_aqd_jurist_triggers.txt,
# on_action/00_aqd_game_start.txt's establishment pre-filter, and
# scripted_effects/z_aqd_establishment_effects.txt exactly. Called from
# generic_actions/z_aqd_piety.txt's aqd_crackdown_madhhab and from the
# establishment event's crackdown option (via
# z_aqd_establishment_effects.txt's aqd_establishment_crackdown) at the
# acting country's scope, with:
#   MADHHAB - bare identifier, e.g. "hanafi"
#   SECT    - bare religion id: sunni / shia / ibadi (used to decide
#             whether this is a heresy of the realm's own religion)
#   SCORE   - that madhhab's institution score
#
# Beyond the tier knockdown the effect also: gains confiscation gold per
# razed building; applies a footprint-scaled stability hit; stamps the
# nationwide "Legal School Persecuted" location modifier
# (z_aqd_jurist_evicted) on every location holding the school's
# buildings; adds a national modifier - z_aqd_heretics_suppressed if
# SECT differs from the realm's religion, else base game's
# ulema_displeased; and drops the opinion of every Muslim realm within
# diplomatic range whose own law is this madhhab
# (common/biases/z_aqd_opinion.txt).
#
# NOTE: destroy_building's "building(X|Y)" target-link string does NOT
# tolerate a $param$ substituted inside the quotes - confirmed broken
# in testing (the engine mangles it: 'buildingbuilding_type:...$|root)').
# So the actual destroy_building calls are split into 18 fully-literal
# per-madhhab effects below, dispatched via dynamic EFFECT NAME
# construction instead (aqd_crackdown_destroy_$MADHHAB$_madrasa/_seat) -
# that pattern IS confirmed safe, since it's the same one already used
# successfully by aqd_apply_jurist_reaction's
# aqd_jurist_reaction_$OUTCOME$_favored/_opposed dispatch above.
# ---------------------------------------------------------------------
aqd_crackdown_effect = {
	# Current scope is the acting country at both call sites
	# (z_aqd_piety.txt wraps this in scope:actor = { }, the establishment
	# event runs it at root). Save it so the location sweep below can
	# write back to a country variable without relying on root, which is
	# unreliable inside a generic_action.
	save_scope_as = aqd_crackdown_country

	if      = { limit = { $SCORE$ >= 8 } set_variable = { name = aqd_crackdown_ceiling  value = 8 } }
	else_if = { limit = { $SCORE$ >= 5 } set_variable = { name = aqd_crackdown_ceiling  value = 5 } }
	else_if = { limit = { $SCORE$ >= 3 } set_variable = { name = aqd_crackdown_ceiling  value = 3 } }
	else    = { set_variable = { name = aqd_crackdown_ceiling  value = 1 } }

	if      = { limit = { var:aqd_crackdown_ceiling = 8 } set_variable = { name = aqd_crackdown_tier_before  value = 4 } }
	else_if = { limit = { var:aqd_crackdown_ceiling = 5 } set_variable = { name = aqd_crackdown_tier_before  value = 3 } }
	else_if = { limit = { var:aqd_crackdown_ceiling = 3 } set_variable = { name = aqd_crackdown_tier_before  value = 2 } }
	else    = { set_variable = { name = aqd_crackdown_tier_before  value = 1 } }

	# Nationwide backlash: BEFORE any building is razed, stamp "Legal
	# School Persecuted" (z_aqd_jurist_evicted) on every owned location
	# that holds a madrasa or seat of this madhhab - locations about to
	# be razed still hold their building here, so they are covered too -
	# and count them into var:aqd_crackdown_footprint for the stability
	# hit below. One linear pass over owned locations, run once per
	# crackdown (60-month cooldown / 4-yearly-gated event): no tick cost.
	set_variable = { name = aqd_crackdown_footprint  value = 0 }
	# Jurist Seats (not madrasas) razed by the while loop below - drives
	# the exiled-jurists follow-up chance at the end of this effect.
	set_variable = { name = aqd_crackdown_seats_razed  value = 0 }
	every_owned_location = {
		limit = {
			OR = {
				has_building = building_type:madrasa_of_$MADHHAB$_fiqh
				has_building = building_type:seat_of_$MADHHAB$_jurists
			}
		}
		add_location_modifier = { modifier = z_aqd_jurist_evicted  years = 10  mode = add_and_extend }
		scope:aqd_crackdown_country = { change_variable = { name = aqd_crackdown_footprint  add = 1 } }
	}

	# Diplomatic backlash: every Muslim realm within diplomatic range
	# whose own law is this madhhab takes the persecution of their school
	# personally - their opinion of us drops (common/biases/z_aqd_opinion.txt).
	every_country_in_diplomatic_range = {
		limit = {
			NOT = { this = scope:aqd_crackdown_country }
			religion.group = religion_group:muslim
			has_policy = $MADHHAB$_policy
		}
		add_opinion = { target = scope:aqd_crackdown_country  modifier = z_aqd_crackdown_on_our_school }
	}

	while = {
		limit = { $SCORE$ >= var:aqd_crackdown_ceiling }
		if = {
			limit = { any_owned_location = { has_building = building_type:madrasa_of_$MADHHAB$_fiqh } }
			aqd_crackdown_destroy_$MADHHAB$_madrasa = yes
			# Confiscation of the college's endowed property - gold gained.
			add_gold = { value = scope:aqd_crackdown_country.monthly_income_trade_and_tax  multiply = 0.25 }
		}
		else_if = {
			limit = { any_owned_location = { has_building = building_type:seat_of_$MADHHAB$_jurists } }
			aqd_crackdown_destroy_$MADHHAB$_seat = yes
			# A seat's waqf endowment is worth far more than a madrasa's.
			add_gold = { value = scope:aqd_crackdown_country.monthly_income_trade_and_tax  multiply = 1 }
			scope:aqd_crackdown_country = { change_variable = { name = aqd_crackdown_seats_razed  add = 1 } }
		}
		else = {
			set_variable = { name = aqd_crackdown_ceiling  value = 0 }
		}
	}

	# Stability hit, scaled to how widespread the school was - so the
	# cost depends on which madhhab was targeted.
	if      = { limit = { var:aqd_crackdown_footprint >= 6 } add_stability = stability_mild_penalty }
	else_if = { limit = { var:aqd_crackdown_footprint >= 1 } add_stability = stability_weak_penalty }

	# National reaction. If the school is a heresy of the state religion
	# (its sect differs from the realm's religion) the ulema treat the
	# suppression as righteous (z_aqd_heretics_suppressed); moving against
	# a fellow school of law instead aggrieves them (base game's
	# ulema_displeased).
	if = {
		limit = { NOT = { religion = religion:$SECT$ } }
		add_country_modifier = { modifier = z_aqd_heretics_suppressed  years = 10  mode = add_and_extend }
	}
	else = {
		add_country_modifier = { modifier = ulema_displeased  years = 10  mode = add_and_extend }
	}

	if      = { limit = { var:aqd_crackdown_tier_before = 1 } add_estate_satisfaction = { type = estate_type:clergy_estate  value = estate_satisfaction_weak_penalty } }
	else_if = { limit = { var:aqd_crackdown_tier_before = 2 } add_estate_satisfaction = { type = estate_type:clergy_estate  value = estate_satisfaction_mild_penalty } }
	else_if = { limit = { var:aqd_crackdown_tier_before >= 3 } add_estate_satisfaction = { type = estate_type:clergy_estate  value = estate_satisfaction_severe_penalty } }

	# Exiled-jurists follow-up: a chance the broken school's jurists flee
	# to ONE bordering realm (land border or across a single sea zone),
	# which is then offered z_aqd_exile.1 - grant or deny a Seat of that
	# school in one of its own cities. Only if the school stood at
	# Noteworthy or higher before this crackdown (tier_before >= 2).
	#
	# This hook writes NOTHING on the acting country beyond the
	# uniquely-named scratch vars set above (aqd_crackdown_seats_razed,
	# alongside the effect's existing aqd_crackdown_* scratch vars) and
	# defers every consequence to a 3-day event on the chosen neighbour,
	# so the trailing actor-scope aqd_apply_jurist_reaction at both call
	# sites (generic_actions/z_aqd_piety.txt, events/religion/
	# z_aqd_establishment_events.txt) is untouched.
	#
	# $MADHHAB$ is a bare id and cannot be compared inside a trigger, so
	# dispatch through a dynamic effect name - the same pattern used by
	# aqd_crackdown_destroy_$MADHHAB$_madrasa/_seat below.
	if = {
		limit = { var:aqd_crackdown_tier_before >= 2 }
		aqd_crackdown_exile_offer_$MADHHAB$ = yes
	}
}

# ---------------------------------------------------------------------
# EXILED-JURISTS FOLLOW-UP to a crackdown. Entry point is the hook at
# the end of aqd_crackdown_effect: aqd_crackdown_exile_offer_$MADHHAB$.
#
# These nine wrappers bind the canonical 1-9 index (see the file header)
# that aqd_crackdown_effect itself never carries - it only has $MADHHAB$
# and $SECT$. All run at the acting country's scope, where
# scope:aqd_crackdown_country (saved at the top of aqd_crackdown_effect)
# is live - use it, never root, which is unreliable inside a
# generic_action.
# ---------------------------------------------------------------------
aqd_crackdown_exile_offer_hanafi  = { aqd_crackdown_exile_offer = { MADHHAB = hanafi   SECT = sunni  IDX = 1 } }
aqd_crackdown_exile_offer_hanbali = { aqd_crackdown_exile_offer = { MADHHAB = hanbali  SECT = sunni  IDX = 2 } }
aqd_crackdown_exile_offer_maliki  = { aqd_crackdown_exile_offer = { MADHHAB = maliki   SECT = sunni  IDX = 3 } }
aqd_crackdown_exile_offer_shafii  = { aqd_crackdown_exile_offer = { MADHHAB = shafii   SECT = sunni  IDX = 4 } }
aqd_crackdown_exile_offer_zahiri  = { aqd_crackdown_exile_offer = { MADHHAB = zahiri   SECT = sunni  IDX = 5 } }
aqd_crackdown_exile_offer_ismaili = { aqd_crackdown_exile_offer = { MADHHAB = ismaili  SECT = shia   IDX = 6 } }
aqd_crackdown_exile_offer_jafari  = { aqd_crackdown_exile_offer = { MADHHAB = jafari   SECT = shia   IDX = 7 } }
aqd_crackdown_exile_offer_zaidi   = { aqd_crackdown_exile_offer = { MADHHAB = zaidi    SECT = shia   IDX = 8 } }
aqd_crackdown_exile_offer_ibadi   = { aqd_crackdown_exile_offer = { MADHHAB = ibadi    SECT = ibadi  IDX = 9 } }

# Fire chance = 30% + 10% per Jurist Seat razed by this crackdown,
# capped at 75%. var:aqd_crackdown_seats_razed lives on the acting
# country (= current scope here). A one-tier knockdown almost always
# razes 0 or 1 seats, so in practice this is 30% / 40% - the event is
# meant to stay rare. Discrete branches, not a computed `chance =`
# (there is no base-game precedent for a dynamic chance in a random
# effect block). The Noteworthy+ gate is already applied by the caller.
aqd_crackdown_exile_offer = {
	if = {
		limit = { var:aqd_crackdown_seats_razed >= 3 }
		random = { chance = 60  aqd_crackdown_exile_pick = { MADHHAB = $MADHHAB$  SECT = $SECT$  IDX = $IDX$ } }
	}
	else_if = {
		limit = { var:aqd_crackdown_seats_razed = 2 }
		random = { chance = 50  aqd_crackdown_exile_pick = { MADHHAB = $MADHHAB$  SECT = $SECT$  IDX = $IDX$ } }
	}
	else_if = {
		limit = { var:aqd_crackdown_seats_razed = 1 }
		random = { chance = 40  aqd_crackdown_exile_pick = { MADHHAB = $MADHHAB$  SECT = $SECT$  IDX = $IDX$ } }
	}
	else = {
		random = { chance = 30  aqd_crackdown_exile_pick = { MADHHAB = $MADHHAB$  SECT = $SECT$  IDX = $IDX$ } }
	}
}

# Two-pass weighted neighbour selection, run at the acting country's
# scope. Pass 1 (70%) prefers a bordering realm whose OWN official
# madhhab already matches the exiled school; pass 2 falls back to any
# eligible bordering realm. "Bordering" = a land border OR across a
# single sea zone (is_neighbor_of_country_or_across_one_seazone, an
# engine trigger). Exactly one realm is ever picked, so "only one
# neighbour asked at a time" holds by construction.
#
# Limit clauses run cheap -> expensive: random_country sweeps every
# country, so the adjacency test comes before aqd_can_host_exile_seat's
# owned-location scan. scope:aqd_exile_target is transient and is NOT
# consumed by z_aqd_exile.1 (the event re-derives its own seat
# location); the only cross-event handoff is the numeric
# var:aqd_exile_madhhab set on the target.
aqd_crackdown_exile_pick = {
	random = {
		chance = 70
		random_country = {
			limit = {
				NOT = { this = scope:aqd_crackdown_country }
				religion = religion:$SECT$
				has_policy = sharia_law_policy
				has_policy = $MADHHAB$_policy
				is_neighbor_of_country_or_across_one_seazone = scope:aqd_crackdown_country
				aqd_can_host_exile_seat = { MADHHAB = $MADHHAB$  SECT = $SECT$ }
			}
			save_scope_as = aqd_exile_target
		}
	}
	if = {
		limit = { NOT = { exists = scope:aqd_exile_target } }
		random_country = {
			limit = {
				NOT = { this = scope:aqd_crackdown_country }
				religion = religion:$SECT$
				has_policy = sharia_law_policy
				is_neighbor_of_country_or_across_one_seazone = scope:aqd_crackdown_country
				aqd_can_host_exile_seat = { MADHHAB = $MADHHAB$  SECT = $SECT$ }
			}
			save_scope_as = aqd_exile_target
		}
	}
	if = {
		limit = { exists = scope:aqd_exile_target }
		scope:aqd_exile_target = {
			set_variable = { name = aqd_exile_madhhab  value = $IDX$ }
			trigger_event_non_silently = { id = z_aqd_exile.1  days = 3 }
		}
	}
}
