# =====================================================================
# AQIDAH SCHOOLS OF CREED - Ulema Institutions: shared effects.
#
# Parameters are text substitution (see base game's own scripted_effects/
# readme.txt) - $OUTCOME$ (accept/refuse) is used to build a dynamic
# effect name (aqd_jurist_reaction_$OUTCOME$_favored/_opposed), never
# compared inside a trigger directly, since text substitution cannot
# satisfy a trigger comparison the way a real value can.
#
# Canonical madhhab index (var:aqd_compat_madhhab, var:aqd_establish_madhhab):
#   1 hanafi  2 hanbali  3 maliki  4 shafii  5 zahiri
#   6 ismaili 7 jafari   8 zaidi   9 ibadi
# =====================================================================

# ---------------------------------------------------------------------
# Cache the realm's total jurist strength (the denominator for the
# share-based tiers) in var:aqd_realm_jurist_total, so the 9-way sum in
# script_values/z_aqd_madhhab_counts.txt is not recomputed inside every
# auto_modifier block. Refreshed monthly + at game start (on_action/
# 00_aqd_game_start.txt) and immediately after any jurist building is
# placed or razed (crackdown, establishment grant, exile grant,
# foreign-jurist seat/college grant, foreign-jurist madrasa seeding).
# Run at the country's own scope.
# ---------------------------------------------------------------------
aqd_recount_jurist_total = {
	set_variable = { name = aqd_realm_jurist_total  value = aqd_realm_jurist_total_recount }
}
aqd_recount_all_jurist_totals = {
	every_country = {
		limit = {
			religion.group = religion_group:muslim
			has_policy = sharia_law_policy
		}
		aqd_recount_jurist_total = yes
	}
}

# when that event only seeded madrasas; the event now grants a Seat
# directly (aqd_jurists_accept below) and does not fire at all unless a
# city can host one (scripted_triggers/z_aqd_jurist_triggers.txt's
# aqd_is_eligible_jurist_source folds in aqd_can_host_jurist_seat), so
# there is no longer a "no host" path for this to serve. Kept because it
# is a clean, self-contained "seed N madrasas of a foreign madhhab in
# one province, bypassing location_potential" primitive that a future
# event may want.
#
# SOURCE = a neighbour country scope. Branches on the SOURCE's held
# madhhab policy, picks one qualifying province of root, and plants 1-3
# madrasas of that madhhab INSIDE THAT ONE PROVINCE so the density can
# accumulate toward the establishment threshold (3 madrasas + a mosque)
# without conquest.
#
# change_building_level_in_location (not construct_building): the
# madrasa's location_potential OR-branch is false for a foreign
# madhhab with no owned seat in the area, so construct_building would
# silently place nothing. The raw level-delta bypasses
# location_potential / allow / country_potential.

# ---------------------------------------------------------------------
# "Foreign Jurists at the Gate" (events/religion/z_aqd_jurist_events.txt).
# The visitors' madhhab is given a standing Seat in one of the realm's
# own cities immediately - the grassroots madrasas-then-establishment
# chain is bypassed. The event never fires unless a city can host that
# Seat (aqd_is_eligible_jurist_source), so there is no "no host" fallback
# path.
#
# TARGET PICK (aqd_jurists_derive_seat_target, called ONCE from the
# event's immediate): branches on var:aqd_compat_madhhab (canonical 1-9,
# set on root in the immediate; 0 is the "no real source" sentinel and
# matches no branch below, so this - and everything downstream - is a
# clean no-op rather than an unset-variable error when the event is
# console-forced with no eligible neighbour) into
# aqd_jurists_find_seat_loc { MADHHAB SECT }, which picks the city once
# and saves scope:aqd_jurists_seat_loc / _seat_prov - so the event desc
# can name the city BEFORE the player chooses, and every option acts on
# that same, already-resolved city (no per-option re-roll).
#
# ACCEPT (aqd_jurists_accept_dispatch / aqd_jurists_accept, called from
# the option effects): re-derives nothing, just guards
# scope:aqd_jurists_seat_loc still exists and hands off to a dynamic
# effect NAME (aqd_jurists_finish_$ENDOW$) - ENDOW (yes/no) is never
# compared in a trigger, per this file's header note on text
# substitution. Both finish_ variants place the Seat (so $MADHHAB$ is a
# real reference in each - no unused-parameter branch).
#
# aqd_jurists_finish_yes additionally: plants one madrasa of the madhhab
# in the seat's province BEFORE the Seat (so raising the Seat cannot take
# the province's last free slot); spends aqd_endow_gold_estimate
# (script_values/z_aqd_madhhab_counts.txt) in gold; and grants
# z_aqd_endowed_our_school (common/biases/z_aqd_opinion.txt) with every
# Muslim realm of that madhhab in diplomatic range. The "endow" option's
# own trigger (aqd_source_madhhab_endow_hostable) guarantees a free
# madrasa slot exists, so no player is ever charged for a college that
# cannot be built.
# ---------------------------------------------------------------------
aqd_jurists_derive_seat_target = {
	if      = { limit = { var:aqd_compat_madhhab = 1 } aqd_jurists_find_seat_loc = { MADHHAB = hanafi   SECT = sunni } }
	else_if = { limit = { var:aqd_compat_madhhab = 2 } aqd_jurists_find_seat_loc = { MADHHAB = hanbali  SECT = sunni } }
	else_if = { limit = { var:aqd_compat_madhhab = 3 } aqd_jurists_find_seat_loc = { MADHHAB = maliki   SECT = sunni } }
	else_if = { limit = { var:aqd_compat_madhhab = 4 } aqd_jurists_find_seat_loc = { MADHHAB = shafii   SECT = sunni } }
	else_if = { limit = { var:aqd_compat_madhhab = 5 } aqd_jurists_find_seat_loc = { MADHHAB = zahiri   SECT = sunni } }
	else_if = { limit = { var:aqd_compat_madhhab = 6 } aqd_jurists_find_seat_loc = { MADHHAB = ismaili  SECT = shia } }
	else_if = { limit = { var:aqd_compat_madhhab = 7 } aqd_jurists_find_seat_loc = { MADHHAB = jafari   SECT = shia } }
	else_if = { limit = { var:aqd_compat_madhhab = 8 } aqd_jurists_find_seat_loc = { MADHHAB = zaidi    SECT = shia } }
	else_if = { limit = { var:aqd_compat_madhhab = 9 } aqd_jurists_find_seat_loc = { MADHHAB = ibadi    SECT = ibadi } }
}

# Per-LOCATION check, inlined - a country-scope any_owned_location
# trigger used as a random_owned_location limit is a scope mismatch.
# Effect context, so $SECT$ / $MADHHAB$ substitute fine here.
aqd_jurists_find_seat_loc = {
	random_owned_location = {
		limit = {
			# TESTING: sect population gate disabled (was >= 0.25) - kept as
			# a >= 0 check so $SECT$ stays referenced in this body; a bare
			# `always = yes` here drops the only $SECT$ reference and makes
			# every SECT-passing call site fail to compile silently. This
			# was exactly why z_aqd_jurists.1 granted no seat: this effect
			# never ran, so scope:aqd_jurists_seat_loc was never set and
			# aqd_jurists_accept's exists-guard always failed - the event
			# still fired and applied aqd_apply_jurist_reaction (the
			# opinion/legitimacy "ulema modifiers"), but placed nothing.
			religion_percentage = { religion = religion:$SECT$  value >= 0 }
			always = yes # TESTING: rural_settlement allowed
			NOT = { has_building = building_type:seat_of_$MADHHAB$_jurists }
			NOT = { has_location_modifier = z_aqd_establishment_pending }
		}
		save_scope_as = aqd_jurists_seat_loc
		province_definition = { save_scope_as = aqd_jurists_seat_prov }
	}
}

	# option's own eligibility already proved a host exists), but guard
	# anyway so a vanished city - or the var:aqd_compat_madhhab = 0
	# no-real-source case - cannot half-apply the outcome.
	if = {
		limit = { exists = scope:aqd_jurists_seat_loc }
		aqd_jurists_finish_$ENDOW$ = { MADHHAB = $MADHHAB$ }
		# The new Seat (and madrasa) changed the realm's jurist footprint -
		# refresh the cached tier denominator so tiers update at once.
		root = { aqd_recount_jurist_total = yes }
	}
}

# Endowment college first (placed in a location OTHER than the seat city

# ---------------------------------------------------------------------
# Apply the accept/refuse consequence on $COUNTRY$. $OUTCOME$ = accept
# or refuse (text-substituted into the effect name below, per base
# game's own scripted_effects/readme.txt "argumented effect" idiom).
# Reads var:aqd_compat_madhhab (set by the caller) via
# aqd_classify_madhhab_compatibility.
#
#   accept + favored : you welcomed a school your creed favours   -> clergy + legitimacy bonus
#   accept + opposed : you welcomed a school your creed opposes    -> clergy + legitimacy penalty
#   refuse + favored : you moved against a school your creed favours -> clergy + legitimacy penalty
#   refuse + opposed : you moved against a school your creed opposes  -> clergy + legitimacy bonus
#
# Both the foreign-jurist event (accept/refuse), the establishment
# event (allow = accept / crack down = refuse) and the crackdown
# generic action (refuse) run this, so the aqidah-judged reaction is
# identical on every hostile-or-welcoming act toward a madhhab.
# ---------------------------------------------------------------------
aqd_apply_jurist_reaction = {
	aqd_classify_madhhab_compatibility = { COUNTRY = $COUNTRY$ }
	if = {
		limit = { $COUNTRY$ = { var:aqd_jurist_compat = 1 } }
		aqd_jurist_reaction_$OUTCOME$_favored = { COUNTRY = $COUNTRY$ }
	}
	else_if = {
		limit = { $COUNTRY$ = { var:aqd_jurist_compat = -1 } }
		aqd_jurist_reaction_$OUTCOME$_opposed = { COUNTRY = $COUNTRY$ }
	}
}

		add_estate_satisfaction = { type = estate_type:clergy_estate  value = estate_satisfaction_weak_bonus }

}

# ---------------------------------------------------------------------
# Crackdown on $MADHHAB$. Two regimes, split on the school's footprint
# (owned locations holding one of its buildings), which this effect
# counts up front anyway:
#
#   footprint < 3  -> TOTAL WIPEOUT: raze every madrasa and seat of the
#                     madhhab. Two colleges or fewer can be stamped out.
#   footprint >= 3 -> KNOCK DOWN ONE TIER: raze (madrasas first) until
#                     $MADHHAB$_institution_tier drops below the tier
#                     the school held when the crackdown began, OR until
#                     half the school's pre-crackdown strength
#                     ($SCORE$ / 2) has been razed - whichever first.
#                     The half-strength cap guarantees a real blow every
#                     time; a school that swamps the realm's jurist
#                     total is halved, not demoted, and finished by the
#                     next crackdown or the wipeout branch once it is
#                     down to <= 2 buildings.
#
# Called from generic_actions/z_aqd_piety.txt's aqd_crackdown_madhhab
# and from the establishment event's crackdown option (via
# z_aqd_establishment_effects.txt's aqd_establishment_crackdown) at the
# acting country's scope, with:
#   MADHHAB - bare identifier, e.g. "hanafi"
#   SECT    - bare religion id: sunni / shia / ibadi (whether this is a
#             heresy of the realm's own religion)
#   SCORE   - that madhhab's raw institution score (script_value name),
#             used only for the half-strength cap
#   TIER    - that madhhab's 0-4 institution tier (script_value name),
#             the value the tier-drop loop chases down by one
#
# Beyond the razing the effect also: gains confiscation gold per razed
# MADRASA only (seats are razed too but pay no gold - a seat's waqf is
# assumed tied up in its own institution, not liquid); applies a
# footprint-scaled stability hit; stamps the
# nationwide "Legal School Persecuted" location modifier
# (z_aqd_jurist_evicted) on every location holding the school's
# buildings; adds a national modifier - z_aqd_heretics_suppressed if
# SECT differs from the realm's religion, else base game's
# ulema_displeased; drops the opinion of every Muslim realm within
# diplomatic range whose own law is this madhhab
# (common/biases/z_aqd_opinion.txt); refreshes the cached realm jurist
# total; and, if the school was Noteworthy+ before, may trigger the
# exiled-jurists follow-up (z_aqd_exile.1) on a bordering realm.
#
# NOTE: destroy_building's "building(X|Y)" target-link string does NOT
# tolerate a $param$ substituted inside the quotes (the engine mangles
# it). So the destroy_building calls are split into 18 fully-literal
# per-madhhab effects below, dispatched via dynamic EFFECT NAME
# construction (aqd_crackdown_destroy_$MADHHAB$_madrasa/_seat) - that
# pattern IS safe.
# ---------------------------------------------------------------------
aqd_crackdown_effect = {
	# Current scope is the acting country at both call sites
	# (z_aqd_piety.txt wraps this in scope:actor = { }, the establishment
	# event runs it at root). Save it so the sweeps below can write back
	# to a country variable without relying on root, which is unreliable
	# inside a generic_action.
	save_scope_as = aqd_crackdown_country

	# The tier the school holds right now (0 none / 1 Presence /
	# 2 Noteworthy / 3 Plurality; 4 Predominance is impossible for a
	# crackdown target). Drives the tier-drop loop and the clergy hit.
	set_variable = { name = aqd_crackdown_tier_before  value = $TIER$ }

	# Half the pre-crackdown strength, snapshotted so it does not track
	# the shrinking live $SCORE$. Cap for the tier-drop branch.
	set_variable    = { name = aqd_crackdown_strength_cap  value = $SCORE$ }
	change_variable = { name = aqd_crackdown_strength_cap  multiply = 0.5 }

	# Nationwide backlash: BEFORE any building is razed, stamp "Legal
	# School Persecuted" (z_aqd_jurist_evicted) on every owned location
	# that holds a madrasa or seat of this madhhab, and count them into
	# var:aqd_crackdown_footprint - which also decides the wipeout vs
	# tier-drop branch below. One linear pass, run once per crackdown
	# (60-month cooldown / 4-yearly-gated event): no tick cost.
	set_variable = { name = aqd_crackdown_footprint  value = 0 }
	# Jurist Seats (not madrasas) razed below - drives the exiled-jurists
	# follow-up chance at the end of this effect.
	set_variable = { name = aqd_crackdown_seats_razed  value = 0 }
	# Total strength razed below (seat 4 / madrasa 1) - the tier-drop
	# branch stops at var:aqd_crackdown_strength_cap.
	set_variable = { name = aqd_crackdown_strength_razed  value = 0 }
	# Hard iteration backstop for the razing while-loops below. With
	# destroy_building_forcefully the natural has_building limits terminate
	# on their own; this only stops a stuck loop from flooding the log.
	# 50 is far above any realistic single-madhhab footprint. Reset before
	# each loop; incremented inside via scope:aqd_crackdown_country like
	# every other counter here (root is unreliable in a generic_action).
	set_variable = { name = aqd_crackdown_iter  value = 0 }
	every_owned_location = {
		limit = {
			OR = {
				has_building = building_type:madrasa_of_$MADHHAB$_fiqh
				has_building = building_type:seat_of_$MADHHAB$_jurists
			}
		}
		add_location_modifier = { modifier = z_aqd_jurist_evicted  years = 10  mode = add_and_extend }
		scope:aqd_crackdown_country = { change_variable = { name = aqd_crackdown_footprint  add = 1 } }
	}

	# Diplomatic backlash: every Muslim realm within diplomatic range
	# whose own law is this madhhab takes the persecution of their school
	# personally - their opinion of us drops (common/biases/z_aqd_opinion.txt).
	every_country_in_diplomatic_range = {
		limit = {
			NOT = { this = scope:aqd_crackdown_country }
			religion.group = religion_group:muslim
			has_policy = $MADHHAB$_policy
		}
		add_opinion = { target = scope:aqd_crackdown_country  modifier = z_aqd_crackdown_on_our_school }
	}

	if = {
		# SMALL SCHOOL (< 3 institution locations) -> stamp it out.
		limit = { var:aqd_crackdown_footprint < 3 }
		scope:aqd_crackdown_country = { set_variable = { name = aqd_crackdown_iter  value = 0 } }
		while = {
			limit = {
				var:aqd_crackdown_iter < 50
				any_owned_location = { has_building = building_type:madrasa_of_$MADHHAB$_fiqh }
			}
			scope:aqd_crackdown_country = { change_variable = { name = aqd_crackdown_iter  add = 1 } }
			aqd_crackdown_destroy_$MADHHAB$_madrasa = yes
			add_gold = { value = scope:aqd_crackdown_country.monthly_income_trade_and_tax  multiply = 0.25 }
			scope:aqd_crackdown_country = { change_variable = { name = aqd_crackdown_strength_razed  add = 1 } }
		}
		scope:aqd_crackdown_country = { set_variable = { name = aqd_crackdown_iter  value = 0 } }
		while = {
			limit = {
				var:aqd_crackdown_iter < 50
				any_owned_location = { has_building = building_type:seat_of_$MADHHAB$_jurists }
			}
			scope:aqd_crackdown_country = { change_variable = { name = aqd_crackdown_iter  add = 1 } }
			aqd_crackdown_destroy_$MADHHAB$_seat = yes
			# No add_gold here - only madrasas pay out (see the doc comment
			# above aqd_crackdown_gold_estimate_target in script_values/
			# z_aqd_madhhab_counts.txt). A seat still gets razed and still
			# counts toward strength_razed for tier-drop/wipeout bookkeeping,
			# it just isn't a source of confiscation gold.
			scope:aqd_crackdown_country = {
				change_variable = { name = aqd_crackdown_seats_razed     add = 1 }
				change_variable = { name = aqd_crackdown_strength_razed  add = 4 }
			}
		}
	}
	else = {
		# ESTABLISHED SCHOOL (3+ locations) -> knock the school down one
		# tier, but raze at least half its pre-crackdown strength so a
		# crackdown is never a no-op. $TIER$ is re-read each pass; the
		# recount at the end of the loop body refreshes the cached
		# denominator so $TIER$ reflects the TRUE post-raze share, not a
		# stale one - otherwise the loop can stop a building early and
		# the school pops back up a tier on the next evaluation.
		scope:aqd_crackdown_country = { set_variable = { name = aqd_crackdown_iter  value = 0 } }
		while = {
			limit = {
				var:aqd_crackdown_iter < 50
				$TIER$ >= var:aqd_crackdown_tier_before
				var:aqd_crackdown_strength_razed < var:aqd_crackdown_strength_cap
			}
			scope:aqd_crackdown_country = { change_variable = { name = aqd_crackdown_iter  add = 1 } }
			if = {
				limit = { any_owned_location = { has_building = building_type:madrasa_of_$MADHHAB$_fiqh } }
				aqd_crackdown_destroy_$MADHHAB$_madrasa = yes
				# Confiscation of the college's endowed property - gold gained.
				add_gold = { value = scope:aqd_crackdown_country.monthly_income_trade_and_tax  multiply = 0.25 }
				scope:aqd_crackdown_country = { change_variable = { name = aqd_crackdown_strength_razed  add = 1 } }
			}
			else_if = {
				limit = { any_owned_location = { has_building = building_type:seat_of_$MADHHAB$_jurists } }
				aqd_crackdown_destroy_$MADHHAB$_seat = yes
				# No add_gold here - only madrasas pay out (see the doc
				# comment above aqd_crackdown_gold_estimate_target in
				# script_values/z_aqd_madhhab_counts.txt).
				scope:aqd_crackdown_country = {
					change_variable = { name = aqd_crackdown_seats_razed     add = 1 }
					change_variable = { name = aqd_crackdown_strength_razed  add = 4 }
				}
			}
			else = {
				# Nothing of this madhhab left - force the loop to end
				# (the limit reads strength_razed, so push it past the cap).
				set_variable = { name = aqd_crackdown_strength_razed  value = 99 }
			}
			scope:aqd_crackdown_country = { aqd_recount_jurist_total = yes }
		}
	}

	# The realm's jurist footprint just changed - refresh the cached
	# denominator so tiers (auto_modifiers/z_aqd_madhhab_tiers.txt) are
	# right at once rather than next month. tier_before was captured
	# above, so the tail below is unaffected.
	scope:aqd_crackdown_country = { aqd_recount_jurist_total = yes }

	# Stability hit, scaled to how widespread the school was - so the
	# cost depends on which madhhab was targeted.
	if      = { limit = { var:aqd_crackdown_footprint >= 6 } add_stability = stability_mild_penalty }
	else_if = { limit = { var:aqd_crackdown_footprint >= 1 } add_stability = stability_weak_penalty }

	# National reaction. If the school is a heresy of the state religion
	# (its sect differs from the realm's religion) the ulema treat the
	# suppression as righteous (z_aqd_heretics_suppressed); moving against
	# a fellow school of law instead aggrieves them (base game's
	# ulema_displeased).
	if = {
		limit = { NOT = { religion = religion:$SECT$ } }
		add_country_modifier = { modifier = z_aqd_heretics_suppressed  years = 10  mode = add_and_extend }
	}
	else = {
		add_country_modifier = { modifier = ulema_displeased  years = 10  mode = add_and_extend }
	}

	if      = { limit = { var:aqd_crackdown_tier_before = 1 } add_estate_satisfaction = { type = estate_type:clergy_estate  value = estate_satisfaction_weak_penalty } }
	else_if = { limit = { var:aqd_crackdown_tier_before = 2 } add_estate_satisfaction = { type = estate_type:clergy_estate  value = estate_satisfaction_mild_penalty } }
	else_if = { limit = { var:aqd_crackdown_tier_before >= 3 } add_estate_satisfaction = { type = estate_type:clergy_estate  value = estate_satisfaction_severe_penalty } }

	# Exiled-jurists follow-up: a chance the broken school's jurists flee
	# to ONE bordering realm (land border or across a single sea zone),
	# which is then offered z_aqd_exile.1 - grant or deny a Seat of that
	# school in one of its own cities. Only if the school stood at
	# Noteworthy or higher before this crackdown (tier_before >= 2).
	#
	# This hook writes NOTHING on the acting country beyond the
	# uniquely-named scratch vars set above (aqd_crackdown_seats_razed,
	# alongside the effect's existing aqd_crackdown_* scratch vars) and
	# defers every consequence to a 3-day event on the chosen neighbour,
	# so the trailing actor-scope aqd_apply_jurist_reaction at both call
	# sites (generic_actions/z_aqd_piety.txt, events/religion/
	# z_aqd_establishment_events.txt) is untouched.
	#
	# $MADHHAB$ is a bare id and cannot be compared inside a trigger, so
	# dispatch through a dynamic effect name - the same pattern used by
	# aqd_crackdown_destroy_$MADHHAB$_madrasa/_seat below.
	if = {
		limit = { var:aqd_crackdown_tier_before >= 2 }
		aqd_crackdown_exile_offer_$MADHHAB$ = yes
	}
}

# EXILED-JURISTS FOLLOW-UP to a crackdown. Entry point is the hook at
# the end of aqd_crackdown_effect: aqd_crackdown_exile_offer_$MADHHAB$.
#
# These nine wrappers bind the canonical 1-9 index (see the file header)
# that aqd_crackdown_effect itself never carries - it only has $MADHHAB$
# and $SECT$. All run at the acting country's scope, where
# scope:aqd_crackdown_country (saved at the top of aqd_crackdown_effect)
# is live - use it, never root, which is unreliable inside a
# generic_action.
# ---------------------------------------------------------------------
aqd_crackdown_exile_offer_hanafi  = { aqd_crackdown_exile_offer = { MADHHAB = hanafi   SECT = sunni  IDX = 1 } }
aqd_crackdown_exile_offer_hanbali = { aqd_crackdown_exile_offer = { MADHHAB = hanbali  SECT = sunni  IDX = 2 } }
aqd_crackdown_exile_offer_maliki  = { aqd_crackdown_exile_offer = { MADHHAB = maliki   SECT = sunni  IDX = 3 } }
aqd_crackdown_exile_offer_shafii  = { aqd_crackdown_exile_offer = { MADHHAB = shafii   SECT = sunni  IDX = 4 } }
aqd_crackdown_exile_offer_zahiri  = { aqd_crackdown_exile_offer = { MADHHAB = zahiri   SECT = sunni  IDX = 5 } }
aqd_crackdown_exile_offer_ismaili = { aqd_crackdown_exile_offer = { MADHHAB = ismaili  SECT = shia   IDX = 6 } }
aqd_crackdown_exile_offer_jafari  = { aqd_crackdown_exile_offer = { MADHHAB = jafari   SECT = shia   IDX = 7 } }
aqd_crackdown_exile_offer_zaidi   = { aqd_crackdown_exile_offer = { MADHHAB = zaidi    SECT = shia   IDX = 8 } }
aqd_crackdown_exile_offer_ibadi   = { aqd_crackdown_exile_offer = { MADHHAB = ibadi    SECT = ibadi  IDX = 9 } }

# Fire chance = 30% + 10% per Jurist Seat razed by this crackdown,
# capped at 75%. var:aqd_crackdown_seats_razed lives on the acting
# country (= current scope here). A one-tier knockdown almost always
# razes 0 or 1 seats, so in practice this is 30% / 40% - the event is
# meant to stay rare. Discrete branches, not a computed `chance =`
# (there is no base-game precedent for a dynamic chance in a random
# effect block). The Noteworthy+ gate is already applied by the caller.
aqd_crackdown_exile_offer = {
	if = {
		limit = { var:aqd_crackdown_seats_razed >= 3 }
		random = { chance = 60  aqd_crackdown_exile_pick = { MADHHAB = $MADHHAB$  SECT = $SECT$  IDX = $IDX$ } }
	}
	else_if = {
		limit = { var:aqd_crackdown_seats_razed = 2 }
		random = { chance = 50  aqd_crackdown_exile_pick = { MADHHAB = $MADHHAB$  SECT = $SECT$  IDX = $IDX$ } }
	}
	else_if = {
		limit = { var:aqd_crackdown_seats_razed = 1 }
		random = { chance = 40  aqd_crackdown_exile_pick = { MADHHAB = $MADHHAB$  SECT = $SECT$  IDX = $IDX$ } }
	}
	else = {
		random = { chance = 30  aqd_crackdown_exile_pick = { MADHHAB = $MADHHAB$  SECT = $SECT$  IDX = $IDX$ } }
	}
}

# Two-pass weighted neighbour selection, run at the acting country's
# scope. Pass 1 (70%) prefers a bordering realm whose OWN official
# madhhab already matches the exiled school; pass 2 falls back to any
# eligible bordering realm. "Bordering" = a land border OR across a
# single sea zone (is_neighbor_of_country_or_across_one_seazone, an
# engine trigger). Exactly one realm is ever picked, so "only one
# neighbour asked at a time" holds by construction.
#
# Limit clauses run cheap -> expensive: random_country sweeps every
# country, so the adjacency test comes before aqd_can_host_exile_seat's
# owned-location scan. scope:aqd_exile_target is transient and is NOT
# consumed by z_aqd_exile.1 (the event re-derives its own seat
# location); the only cross-event handoff is the numeric
# var:aqd_exile_madhhab set on the target.
aqd_crackdown_exile_pick = {
	random = {
		chance = 70
		random_country = {
			limit = {
				NOT = { this = scope:aqd_crackdown_country }
				religion = religion:$SECT$
				has_policy = sharia_law_policy
				has_policy = $MADHHAB$_policy
				is_neighbor_of_country_or_across_one_seazone = scope:aqd_crackdown_country
				aqd_can_host_exile_seat = { MADHHAB = $MADHHAB$  SECT = $SECT$ }
			}
			save_scope_as = aqd_exile_target
		}
	}
	if = {
		limit = { NOT = { exists = scope:aqd_exile_target } }
		random_country = {
			limit = {
				NOT = { this = scope:aqd_crackdown_country }
				religion = religion:$SECT$
				has_policy = sharia_law_policy
				is_neighbor_of_country_or_across_one_seazone = scope:aqd_crackdown_country
				aqd_can_host_exile_seat = { MADHHAB = $MADHHAB$  SECT = $SECT$ }
			}
			save_scope_as = aqd_exile_target
		}
	}
	if = {
		limit = { exists = scope:aqd_exile_target }
		scope:aqd_exile_target = {
			set_variable = { name = aqd_exile_madhhab  value = $IDX$ }
			trigger_event_non_silently = { id = z_aqd_exile.1  days = 3 }
		}
	}
}

# =====================================================================
# AQIDAH SCHOOLS OF CREED - Ulema Institutions: Ḥalqa madhhab-pick
# leader weighting.
#
# Called from z_aqd_halqa_events.txt's immediate, at country scope,
# right before the random_list, once scope:halqa_prov exists. Decides
# whether exactly one ready madhhab in scope:halqa_prov owns STRICTLY
# more madrasas of its own fiqh than every other ready madhhab there -
# if so, var:aqd_halqa_leader_madhhab is set to its canonical 1-9 index
# so the random_list can give it a higher (not guaranteed) weight. Left
# at the sentinel 0 - NEVER left unset, so every `var:aqd_halqa_leader_
# madhhab = N` comparison in the random_list is always safe (an unset-
# variable comparison is a hard error, not a silent false - the exact
# bug class that broke establishment/exile/foreign-jurist placement
# earlier this project) - whenever two or more ready madhhabs tie for
# the top count (including an all-zero tie, the common early-game case
# now that a ḥalqa no longer requires any madrasas at all) or only one
# madhhab is ready (moot: its two random_list entries sum to the same
# weight either way). All scratch variables here (aqd_halqa_count_1..9,
# aqd_halqa_max_count, aqd_halqa_max_ties, aqd_halqa_leader_madhhab) are
# cleaned up by z_aqd_halqa.1's option, alongside var:aqd_halqa_madhhab.
# =====================================================================
aqd_halqa_compute_leader = {
	# Per-candidate counts - only for madhhabs actually ready here, so a
	# typical single-seat province computes exactly one of these instead
	# of all nine.
	if = { limit = { scope:halqa_prov = { aqd_halqa_prov_ready_hanafi = yes } }   set_variable = { name = aqd_halqa_count_1  value = aqd_prov_madrasa_count_hanafi } }
	if = { limit = { scope:halqa_prov = { aqd_halqa_prov_ready_hanbali = yes } }  set_variable = { name = aqd_halqa_count_2  value = aqd_prov_madrasa_count_hanbali } }
	if = { limit = { scope:halqa_prov = { aqd_halqa_prov_ready_maliki = yes } }   set_variable = { name = aqd_halqa_count_3  value = aqd_prov_madrasa_count_maliki } }
	if = { limit = { scope:halqa_prov = { aqd_halqa_prov_ready_shafii = yes } }   set_variable = { name = aqd_halqa_count_4  value = aqd_prov_madrasa_count_shafii } }
	if = { limit = { scope:halqa_prov = { aqd_halqa_prov_ready_zahiri = yes } }   set_variable = { name = aqd_halqa_count_5  value = aqd_prov_madrasa_count_zahiri } }
	if = { limit = { scope:halqa_prov = { aqd_halqa_prov_ready_ismaili = yes } }  set_variable = { name = aqd_halqa_count_6  value = aqd_prov_madrasa_count_ismaili } }
	if = { limit = { scope:halqa_prov = { aqd_halqa_prov_ready_jafari = yes } }   set_variable = { name = aqd_halqa_count_7  value = aqd_prov_madrasa_count_jafari } }
	if = { limit = { scope:halqa_prov = { aqd_halqa_prov_ready_zaidi = yes } }    set_variable = { name = aqd_halqa_count_8  value = aqd_prov_madrasa_count_zaidi } }
	if = { limit = { scope:halqa_prov = { aqd_halqa_prov_ready_ibadi = yes } }    set_variable = { name = aqd_halqa_count_9  value = aqd_prov_madrasa_count_ibadi } }

	# Max scan. Starts below any real count (including 0) so the first
	# ready candidate always wins its own comparison.
	set_variable = { name = aqd_halqa_max_count  value = -1 }
	if = { limit = { exists = var:aqd_halqa_count_1  var:aqd_halqa_count_1 > var:aqd_halqa_max_count }  set_variable = { name = aqd_halqa_max_count  value = var:aqd_halqa_count_1 } }
	if = { limit = { exists = var:aqd_halqa_count_2  var:aqd_halqa_count_2 > var:aqd_halqa_max_count }  set_variable = { name = aqd_halqa_max_count  value = var:aqd_halqa_count_2 } }
	if = { limit = { exists = var:aqd_halqa_count_3  var:aqd_halqa_count_3 > var:aqd_halqa_max_count }  set_variable = { name = aqd_halqa_max_count  value = var:aqd_halqa_count_3 } }
	if = { limit = { exists = var:aqd_halqa_count_4  var:aqd_halqa_count_4 > var:aqd_halqa_max_count }  set_variable = { name = aqd_halqa_max_count  value = var:aqd_halqa_count_4 } }
	if = { limit = { exists = var:aqd_halqa_count_5  var:aqd_halqa_count_5 > var:aqd_halqa_max_count }  set_variable = { name = aqd_halqa_max_count  value = var:aqd_halqa_count_5 } }
	if = { limit = { exists = var:aqd_halqa_count_6  var:aqd_halqa_count_6 > var:aqd_halqa_max_count }  set_variable = { name = aqd_halqa_max_count  value = var:aqd_halqa_count_6 } }
	if = { limit = { exists = var:aqd_halqa_count_7  var:aqd_halqa_count_7 > var:aqd_halqa_max_count }  set_variable = { name = aqd_halqa_max_count  value = var:aqd_halqa_count_7 } }
	if = { limit = { exists = var:aqd_halqa_count_8  var:aqd_halqa_count_8 > var:aqd_halqa_max_count }  set_variable = { name = aqd_halqa_max_count  value = var:aqd_halqa_count_8 } }
	if = { limit = { exists = var:aqd_halqa_count_9  var:aqd_halqa_count_9 > var:aqd_halqa_max_count }  set_variable = { name = aqd_halqa_max_count  value = var:aqd_halqa_count_9 } }

	# Tie count - how many ready candidates equal the max.
	set_variable = { name = aqd_halqa_max_ties  value = 0 }
	if = { limit = { exists = var:aqd_halqa_count_1  var:aqd_halqa_count_1 = var:aqd_halqa_max_count }  change_variable = { name = aqd_halqa_max_ties  add = 1 } }
	if = { limit = { exists = var:aqd_halqa_count_2  var:aqd_halqa_count_2 = var:aqd_halqa_max_count }  change_variable = { name = aqd_halqa_max_ties  add = 1 } }
	if = { limit = { exists = var:aqd_halqa_count_3  var:aqd_halqa_count_3 = var:aqd_halqa_max_count }  change_variable = { name = aqd_halqa_max_ties  add = 1 } }
	if = { limit = { exists = var:aqd_halqa_count_4  var:aqd_halqa_count_4 = var:aqd_halqa_max_count }  change_variable = { name = aqd_halqa_max_ties  add = 1 } }
	if = { limit = { exists = var:aqd_halqa_count_5  var:aqd_halqa_count_5 = var:aqd_halqa_max_count }  change_variable = { name = aqd_halqa_max_ties  add = 1 } }

# =====================================================================
# AQIDAH SCHOOLS OF CREED - Ulema Institutions: Ḥalqa apply + dispatch.
#
# Entry point is aqd_halqa_dispatch, called from events/religion/
# z_aqd_halqa_events.txt's z_aqd_halqa.1 immediate after it has picked
# scope:halqa_prov (a province_definition) and set var:aqd_halqa_madhhab
# (canonical 1-9) + var:aqd_jurist_compat (via
# aqd_classify_madhhab_compatibility). All run at the event country's
# scope; root is the event country throughout.
#
# One of the four class effects stamps a z_aqd_halqa_* modifier
# (main_menu/common/static_modifiers/z_aqd_halqa.txt) on every owned
# location of the province for 4 years - the modifier also being the
# one-ḥalqa-per-province marker - plus the class's one-shot effects.
# Class rules, in dispatch order:
#   tolerated  - religious_school din_i_ilahi_school (ALL seats, first).
#   state      - the seat's madhhab is the realm's own <X>_policy.
#   heterodox  - var:aqd_jurist_compat = -1 (creed opposes the madhhab).
#   orthodox   - otherwise (compat 1 or 0; also no-state-madhhab realms).
# =====================================================================

aqd_halqa_dispatch = {
	# The pick always sets var:aqd_halqa_madhhab (the random_list entry
	# triggers mirror the picker limit); guard anyway so a stray unset
	# cannot stamp a class-less ḥalqa.
	if = {
		limit = { exists = var:aqd_halqa_madhhab }

		if = {
			limit = { religious_school = religious_school:din_i_ilahi_school }
			aqd_halqa_apply_tolerated = yes
		}
		else_if = {
			limit = {
				OR = {
					AND = { var:aqd_halqa_madhhab = 1  has_policy = hanafi_policy }
					AND = { var:aqd_halqa_madhhab = 2  has_policy = hanbali_policy }
					AND = { var:aqd_halqa_madhhab = 3  has_policy = maliki_policy }
					AND = { var:aqd_halqa_madhhab = 4  has_policy = shafii_policy }
					AND = { var:aqd_halqa_madhhab = 5  has_policy = zahiri_policy }
					AND = { var:aqd_halqa_madhhab = 6  has_policy = ismaili_policy }
					AND = { var:aqd_halqa_madhhab = 7  has_policy = jafari_policy }
					AND = { var:aqd_halqa_madhhab = 8  has_policy = zaidi_policy }
					AND = { var:aqd_halqa_madhhab = 9  has_policy = ibadi_policy }
				}
			}
			aqd_halqa_apply_state = yes
		}
		else_if = {
			limit = { var:aqd_jurist_compat = -1 }
			if = {
				limit = { OR = { var:aqd_halqa_madhhab = 1  var:aqd_halqa_madhhab = 2  var:aqd_halqa_madhhab = 3  var:aqd_halqa_madhhab = 4  var:aqd_halqa_madhhab = 5 } }
				aqd_halqa_apply_heterodox = { SECT = sunni }
			}
			else_if = {
				limit = { OR = { var:aqd_halqa_madhhab = 6  var:aqd_halqa_madhhab = 7  var:aqd_halqa_madhhab = 8 } }
				aqd_halqa_apply_heterodox = { SECT = shia }
			}
			else = {
				aqd_halqa_apply_heterodox = { SECT = ibadi }
			}
		}
		else = {
			aqd_halqa_apply_orthodox = yes
		}
	}
}

# STATE - consolidation. No one-shot change_control (removed - control

# TODO in static_modifiers/z_aqd_halqa.txt). Cross-sect Muslims
# (heretics, NOT non-Muslim heathens - the religion_group limit is what
# excludes them) drift to the state religion, in the same 20%/10%
# (seat/rest of province) shape as aqd_halqa_apply_heterodox's own
# conversion - just aimed at root.religion instead of a SECT param, and
# with no "already that religion" guard needed (root can never already
# be cross-sect of itself).
#
# The modifier key itself is one of nine (z_aqd_halqa_state_hanafi ...
# _ibadi, static_modifiers/z_aqd_halqa.txt) so the display NAME reads
# "<Madhhab> Madhhab Ḥalqa" - picked by a 9-way branch on
# var:aqd_halqa_madhhab, read directly (NOT passed as a $MADHHAB$
# parameter - this effect already runs at root/country scope, where the
# bare var: form reads correctly; a new $PARAM$ pathway here would risk
# the exact "unknown arguments" silent-no-op class that broke
# establishment/exile/foreign-jurist placement earlier this project).
# All nine variants have IDENTICAL fields - this only changes the name.

	else_if = { limit = { var:aqd_halqa_madhhab = 5 }  scope:halqa_prov = { every_location_in_province_definition = { limit = { owner ?= root }  add_location_modifier = { modifier = z_aqd_halqa_state_zahiri   years = 4 } } } }
	else_if = { limit = { var:aqd_halqa_madhhab = 6 }  scope:halqa_prov = { every_location_in_province_definition = { limit = { owner ?= root }  add_location_modifier = { modifier = z_aqd_halqa_state_ismaili  years = 4 } } } }
	else_if = { limit = { var:aqd_halqa_madhhab = 7 }  scope:halqa_prov = { every_location_in_province_definition = { limit = { owner ?= root }  add_location_modifier = { modifier = z_aqd_halqa_state_jafari   years = 4 } } } }
	else_if = { limit = { var:aqd_halqa_madhhab = 8 }  scope:halqa_prov = { every_location_in_province_definition = { limit = { owner ?= root }  add_location_modifier = { modifier = z_aqd_halqa_state_zaidi    years = 4 } } } }
	else_if = { limit = { var:aqd_halqa_madhhab = 9 }  scope:halqa_prov = { every_location_in_province_definition = { limit = { owner ?= root }  add_location_modifier = { modifier = z_aqd_halqa_state_ibadi    years = 4 } } } }

					NOT = { religion = root.religion }

				split_pop = { fraction = 0.2  religion = root.religion }
			}
		}
	}

				NOT = { this = scope:halqa_loc }
			}
			every_pop = {
				limit = {
					religion.group = religion_group:muslim
					NOT = { religion = root.religion }
				}
				split_pop = { fraction = 0.1  religion = root.religion }
			}
		}
	}
