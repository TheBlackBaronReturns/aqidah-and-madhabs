# AQIDAH SCHOOLS OF CREED - Ulema Institutions: shared triggers.
#
# *** GOTCHA (cost real hours to find - read this before touching any
# $PARAM$ near a population-percentage check) ***
# "religion_percentage(religion:$SECT$)" - the QUOTED-STRING function-
# call form of this trigger - never substitutes a $PARAM$ placed inside
# the quotes. It parses the string with its own mini-parser, which does
# not go through the normal script-wide text substitution pass, so
# "religion_percentage(religion:$SECT$)" is evaluated literally with an
# EMPTY religion ("Type: none" / "field not set" in error.log) no matter
# what $SECT$ is bound to - in a trigger, an effect, ANY calling context.
# It fails perfectly silently as far as gameplay is concerned: the
# surrounding limit/trigger just reads false, so every check built on it
# (seat/madrasa placement, establishment, exile, foreign-jurist
# eligibility) looked like it was "doing nothing" with no error a casual
# read of the log would associate with the actual line.
# FIX: use the STRUCTURED block form instead when the religion must be
# parameterised - `religion = religion:$SECT$` there is a normal token,
# not inside a string, so ordinary text substitution applies and it
# works fine:
#     religion_percentage = { religion = religion:$SECT$  value >= 0.25 }
# A LITERAL religion inside the quoted form (e.g. "religion:sunni") is
# fine and unaffected - only a $PARAM$ inside the quotes is broken.
#
# *** SECOND GOTCHA, same family, also cost real hours: an UNREFERENCED
# $PARAM$ is a hard compile error, not a silent false ***
# When the "sect population gate disabled" TESTING edits above replaced
# the religion_percentage(...$SECT$...) line with two bare `always = yes`
# lines, that deleted the ONLY textual occurrence of $SECT$ in several
# scripted_trigger/scripted_effect bodies (aqd_province_has_madhhab_density,
# aqd_establish_save_target, aqd_exile_save_target,
# aqd_jurists_find_seat_loc). EU5's argument compiler scans a definition's
# body for every $PARAM$ token to learn its accepted parameters; a call
# site that then passes SECT = ... to a definition with no $SECT$ left in
# its body fails with "Compiling source for <name> failed for unknown
# arguments: SECT" in debug.log - and the WHOLE invocation silently
# no-ops (any random_owned_location/random_country inside it never runs,
# so any scope it would have saved is simply never set). This is exactly
# what broke establishment/exile/foreign-jurist seat placement for a
# whole session: the events still fired and applied their non-placement
# effects (opinion/legitimacy reactions), but placed no building, because
# the save_target/find_seat_loc effect that was supposed to do so never
# ran. FIX: never fully delete a parameter's last usage while disabling a
# gate under TESTING - replace it with a trivially-true check that still
# references the token, e.g. `religion_percentage = { religion =
# religion:$SECT$  value >= 0 }` instead of `always = yes`. That is also
# a one-line revert (0 -> 0.25) instead of restoring two deleted lines.
#
# aqd_is_eligible_jurist_source: is 'this' (a candidate neighbor
# country) a valid source of "foreign jurists" for 'root' (the
# receiving country)? A candidate qualifies for a given madhhab if it
# holds that madhhab as its own current law, root does NOT already
# hold that same madhhab as its own current law (never offer your own
# madhhab as "foreign"), that madhhab's tier in root has not already
# reached Plurality (root.<X>_institution_tier < 3 - see
# script_values/z_aqd_madhhab_counts.txt; tier 3 = Plurality), AND root
# owns a town+ location with >= 25% of the CANDIDATE's own religion, no
# seat of that madhhab there yet, no z_aqd_establishment_pending marker
# (inlined per branch below, not delegated to a shared sub-trigger -
# see the note further down on trigger-calling-trigger param binding).
# "Foreign Jurists at the Gate" grants a Seat directly, so if no city
# can host one the event must not fire at all - the whole point of
# folding that check in here. The 0.25 threshold must match
# building_types/z_aqd_madrasas.txt and aqd_province_has_madhhab_density.
#
# aqd_province_has_madhhab_density = { MADHHAB SECT }: does 'this' (a
# country, = root here) own a province_definition holding >= 3 of its
# own madrasas of MADHHAB plus >= 1 mosque (temple), with no seat of
# MADHHAB and whose capital carries no z_aqd_establishment_pending
# marker? Drives the establishment pulse in on_action/00_aqd_game_start.txt.
# This is a province-granularity STRUCTURAL gate. Under the share-based
# tier model it does NOT imply any realm-wide tier - it and the pulse
# prefilter in 00_aqd_game_start.txt are independent constraints; do
# not try to re-sync them.
# =====================================================================

# One fiqh madrasa per location, whatever the madhhab: a location that
# already holds any madrasa_of_<X>_fiqh takes no other. Used by the
# madrasa location_potential (building_types/z_aqd_madrasas.txt) and by

	OR = {
		has_building = building_type:madrasa_of_hanafi_fiqh

		has_building = building_type:madrasa_of_zaidi_fiqh
		has_building = building_type:madrasa_of_ibadi_fiqh
	}
}

aqd_province_has_madhhab_density = {
	any_owned_location = {
		province_definition = {
			# 3+ of our own madrasas of the madhhab. owner ?= root, not
			# owner = root: a province_definition can include lake / sea
			# locations with no owner, and owner = root errors on those.
			any_location_in_province_definition = {
				count >= 3
				owner ?= root
				has_building = building_type:madrasa_of_$MADHHAB$_fiqh
			}
			# a mosque
			any_location_in_province_definition = {
				count >= 1
				owner ?= root
				has_building = building_type:temple
			}
			# no seat of the madhhab yet
			NOT = { any_location_in_province_definition = { has_building = building_type:seat_of_$MADHHAB$_jurists } }
			# not recently settled here
			NOT = { any_location_in_province_definition = { has_location_modifier = z_aqd_establishment_pending } }
			# a valid seat target we own: a town (or larger) with the sect present
			any_location_in_province_definition = {
				count >= 1
				owner ?= root
				# TESTING: sect population gate disabled (was >= 0.25) - kept
				# as a >= 0 check, not `always = yes`, so $SECT$ stays a
				# referenced parameter. Dropping the only $SECT$ reference
				# from this body makes every call site that passes SECT
				# fail to compile ("unknown arguments: SECT"), which
				# silently no-ops the whole trigger - this is exactly what
				# broke establishment/exile/foreign-jurist seat placement.
				religion_percentage = { religion = religion:$SECT$  value >= 0 }
				always = yes # TESTING: rural_settlement allowed
			}
		}
	}
}

# aqd_can_host_exile_seat = { MADHHAB SECT }: could 'this' country
# shelter the jurists of MADHHAB exiled by a crackdown next door - i.e.
# raise a Seat of that school in some city it owns? Deliberately WITHOUT
# aqd_province_has_madhhab_density's 3-madrasa/mosque gate: the exiles
# bring the school with them, so a bare city with the sect present is
# enough.
#
# religion = religion:$SECT$ is the STATE religion (not the group), so
# the seat building's country_potential (building_types/
# z_aqd_jurist_seats.txt) can never later prune a seat placed this way.
#
# Scope-free by design: this trigger is reused by z_aqd_exile.1's own
# target picker, and saved scopes do not survive trigger_event. The
# "not the crackdown country itself" check therefore lives inline in the
# selection-phase random_country limits in scripted_effects/
# z_aqd_jurist_effects.txt, not here.
#
# The 0.25 threshold matches building_types/z_aqd_madrasas.txt and
# aqd_province_has_madhhab_density above.
# ---------------------------------------------------------------------
aqd_can_host_exile_seat = {
	religion = religion:$SECT$
	has_policy = sharia_law_policy
	any_owned_location = {
		always = yes # TESTING: sect population gate disabled
		always = yes # TESTING: rural_settlement allowed
		NOT = { has_building = building_type:seat_of_$MADHHAB$_jurists }
		NOT = { has_location_modifier = z_aqd_establishment_pending }
	}
}

# ---------------------------------------------------------------------
# aqd_source_madhhab_endow_hostable: gates z_aqd_jurists.1's "endow a
# college as well" option - true when a Seat host for the offered madhhab
# (var:aqd_compat_madhhab, canonical 1-9, set on root in the event's
# immediate) ALSO has a free madrasa slot in its province. Branches on
# the numeric index; the per-branch check is a concrete no-param trigger
# (trigger->trigger param substitution does not work in EU5).
# ---------------------------------------------------------------------
aqd_source_madhhab_endow_hostable = {
	OR = {
		AND = { var:aqd_compat_madhhab = 1  aqd_can_host_seat_madrasa_hanafi = yes }
		AND = { var:aqd_compat_madhhab = 2  aqd_can_host_seat_madrasa_hanbali = yes }
		AND = { var:aqd_compat_madhhab = 3  aqd_can_host_seat_madrasa_maliki = yes }
		AND = { var:aqd_compat_madhhab = 4  aqd_can_host_seat_madrasa_shafii = yes }
		AND = { var:aqd_compat_madhhab = 5  aqd_can_host_seat_madrasa_zahiri = yes }
		AND = { var:aqd_compat_madhhab = 6  aqd_can_host_seat_madrasa_ismaili = yes }

# Country-scoped. Cheap short-circuiting prefilter for the halqa pulse
# (on_action/00_aqd_game_start.txt): does this realm hold ANY jurist
# seat at all? Most Muslim realms hold none, so this keeps the province
# scan below off all but ~15-20 realms.
aqd_country_owns_any_jurist_seat = {
	any_owned_location = {
		OR = {
			has_building = building_type:seat_of_hanafi_jurists
			has_building = building_type:seat_of_hanbali_jurists
			has_building = building_type:seat_of_maliki_jurists
			has_building = building_type:seat_of_shafii_jurists
			has_building = building_type:seat_of_zahiri_jurists
			has_building = building_type:seat_of_ismaili_jurists
			has_building = building_type:seat_of_jafari_jurists
			has_building = building_type:seat_of_zaidi_jurists
			has_building = building_type:seat_of_ibadi_jurists
		}
	}
}

# province_definition-scoped. Does this province hold an owned seat of
# <X>? Previously also required >= 3 of the realm's own madrasas of <X>
# in the same province (the same clause as aqd_province_has_madhhab_density
# above) - removed by design: a ḥalqa is now initiated off a bare seat, no
# madrasa density needed. NINE concrete triggers, NOT one parameterised one: EU5 does not
# substitute $PARAM$ when a scripted_trigger is invoked from inside
# another scripted_trigger (it only works from an effect / an effect's
# limit). Used by aqd_country_has_any_halqa_ready below and by
# events/religion/z_aqd_halqa_events.txt (the picker limit and each
# random_list weight entry).
#
# Each also excludes a location currently under THAT SAME madhhab's own
# fatwa ban (main_menu/common/static_modifiers/z_aqd_fatwa_displeased.txt,
# events/religion/z_aqd_fatwa_events.txt) - one combined scan per
# trigger, not two: the ban is checked on the SAME location as the seat
# building, both more correct (the ban is applied at the seat's own
# location) and free (no second any_location_in_province_definition
# pass). This is per-madhhab by design: banning Hanafi must not stop
# Shafi'i (or any other madhhab) from rolling its own ḥalqa in the same
# province - see z_aqd_fatwa_displeased.txt's header for why this is 9
# concrete keys rather than one shared one, and why
# aqd_location_has_active_halqa below is deliberately NOT touched by
# this feature.
aqd_halqa_prov_ready_hanafi  = { any_location_in_province_definition = { has_building = building_type:seat_of_hanafi_jurists   NOT = { has_location_modifier = z_aqd_fatwa_displeased_hanafi } } }
aqd_halqa_prov_ready_hanbali = { any_location_in_province_definition = { has_building = building_type:seat_of_hanbali_jurists  NOT = { has_location_modifier = z_aqd_fatwa_displeased_hanbali } } }
aqd_halqa_prov_ready_maliki  = { any_location_in_province_definition = { has_building = building_type:seat_of_maliki_jurists   NOT = { has_location_modifier = z_aqd_fatwa_displeased_maliki } } }
aqd_halqa_prov_ready_shafii  = { any_location_in_province_definition = { has_building = building_type:seat_of_shafii_jurists   NOT = { has_location_modifier = z_aqd_fatwa_displeased_shafii } } }
aqd_halqa_prov_ready_zahiri  = { any_location_in_province_definition = { has_building = building_type:seat_of_zahiri_jurists   NOT = { has_location_modifier = z_aqd_fatwa_displeased_zahiri } } }
aqd_halqa_prov_ready_ismaili = { any_location_in_province_definition = { has_building = building_type:seat_of_ismaili_jurists  NOT = { has_location_modifier = z_aqd_fatwa_displeased_ismaili } } }
aqd_halqa_prov_ready_jafari  = { any_location_in_province_definition = { has_building = building_type:seat_of_jafari_jurists   NOT = { has_location_modifier = z_aqd_fatwa_displeased_jafari } } }
aqd_halqa_prov_ready_zaidi   = { any_location_in_province_definition = { has_building = building_type:seat_of_zaidi_jurists    NOT = { has_location_modifier = z_aqd_fatwa_displeased_zaidi } } }
aqd_halqa_prov_ready_ibadi   = { any_location_in_province_definition = { has_building = building_type:seat_of_ibadi_jurists    NOT = { has_location_modifier = z_aqd_fatwa_displeased_ibadi } } }

# Country-scoped. Does the realm own at least one province that could
# host a ḥalqa right now - a seat of some madhhab (no madrasa density
# required), and no ḥalqa already running anywhere in that province?
# ONE any_owned_location pass: the 9
# madhhab branches live inside one OR, and has_building = seat_of_X is
# first in each AND so the province sub-scan only runs for the handful
# of locations that actually hold that seat.
aqd_country_has_any_halqa_ready = {
	any_owned_location = {
		province_definition = {
			NOT = { any_location_in_province_definition = { aqd_location_has_active_halqa = yes } }
		}
		OR = {
			AND = { has_building = building_type:seat_of_hanafi_jurists   province_definition = { aqd_halqa_prov_ready_hanafi = yes } }
			AND = { has_building = building_type:seat_of_hanbali_jurists  province_definition = { aqd_halqa_prov_ready_hanbali = yes } }
			AND = { has_building = building_type:seat_of_maliki_jurists   province_definition = { aqd_halqa_prov_ready_maliki = yes } }
			AND = { has_building = building_type:seat_of_shafii_jurists   province_definition = { aqd_halqa_prov_ready_shafii = yes } }
			AND = { has_building = building_type:seat_of_zahiri_jurists   province_definition = { aqd_halqa_prov_ready_zahiri = yes } }
			AND = { has_building = building_type:seat_of_ismaili_jurists  province_definition = { aqd_halqa_prov_ready_ismaili = yes } }
			AND = { has_building = building_type:seat_of_jafari_jurists   province_definition = { aqd_halqa_prov_ready_jafari = yes } }
			AND = { has_building = building_type:seat_of_zaidi_jurists    province_definition = { aqd_halqa_prov_ready_zaidi = yes } }
			AND = { has_building = building_type:seat_of_ibadi_jurists    province_definition = { aqd_halqa_prov_ready_ibadi = yes } }
		}
	}
}
