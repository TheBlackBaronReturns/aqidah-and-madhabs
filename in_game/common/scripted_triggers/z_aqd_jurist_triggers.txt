# AQIDAH SCHOOLS OF CREED - Ulema Institutions: shared triggers.
#
# aqd_is_eligible_jurist_source: is 'this' (a candidate neighbor
# country) a valid source of "foreign jurists" for 'root' (the
# receiving country)? A candidate qualifies for a given madhhab if it
# holds that madhhab as its own current law, root does NOT already
# hold that same madhhab as its own current law (never offer your own
# madhhab as "foreign"), that madhhab's tier in root has not already
# reached Plurality (root.<X>_institution_tier < 3 - see
# script_values/z_aqd_madhhab_counts.txt; tier 3 = Plurality), AND
# root actually has a location whose population meets the population
# threshold for the CANDIDATE's own religion (sunni/shia/ibadi per
# madhhab) - so the event is never offered when accepting it could
# never actually place a seat anywhere (e.g. a wholly-Sunni root has
# no province any Shia jurists could realistically settle in). The
# 0.25 threshold here must match building_types/z_aqd_madrasas.txt and
# scripted_effects/z_aqd_jurist_effects.txt's
# aqd_build_madrasas_of_madhhab_from exactly.
#
# aqd_province_has_madhhab_density = { MADHHAB SECT }: does 'this' (a
# country, = root here) own a province_definition holding >= 3 of its
# own madrasas of MADHHAB plus >= 1 mosque (temple), with no seat of
# MADHHAB and whose capital carries no z_aqd_establishment_pending
# marker? Drives the establishment pulse in on_action/00_aqd_game_start.txt.
# This is a province-granularity STRUCTURAL gate. Under the share-based
# tier model it does NOT imply any realm-wide tier - it and the pulse
# prefilter in 00_aqd_game_start.txt are independent constraints; do
# not try to re-sync them.
# =====================================================================

# One fiqh madrasa per location, whatever the madhhab: a location that
# already holds any madrasa_of_<X>_fiqh takes no other. Used by the
# madrasa location_potential (building_types/z_aqd_madrasas.txt) and by
# the foreign-jurist placement (scripted_effects/z_aqd_jurist_effects.txt
# aqd_build_madrasas_of_madhhab_from). Location-scoped.
aqd_location_has_fiqh_madrasa = {
	OR = {
		has_building = building_type:madrasa_of_hanafi_fiqh
		has_building = building_type:madrasa_of_hanbali_fiqh
		has_building = building_type:madrasa_of_maliki_fiqh
		has_building = building_type:madrasa_of_shafii_fiqh
		has_building = building_type:madrasa_of_zahiri_fiqh
		has_building = building_type:madrasa_of_ismaili_fiqh
		has_building = building_type:madrasa_of_jafari_fiqh
		has_building = building_type:madrasa_of_zaidi_fiqh
		has_building = building_type:madrasa_of_ibadi_fiqh
	}
}

aqd_is_eligible_jurist_source = {
	OR = {
		AND = {
			this = { has_policy = hanafi_policy }
			NOT = { root = { has_policy = hanafi_policy } }
			root.hanafi_institution_tier < 3
			root = { any_owned_location = { NOT = { location_rank ?= location_rank:rural_settlement }  "religion_percentage(religion:sunni)" >= 0.25 } }
		}
		AND = {
			this = { has_policy = hanbali_policy }
			NOT = { root = { has_policy = hanbali_policy } }
			root.hanbali_institution_tier < 3
			root = { any_owned_location = { NOT = { location_rank ?= location_rank:rural_settlement }  "religion_percentage(religion:sunni)" >= 0.25 } }

aqd_province_has_madhhab_density = {
	any_owned_location = {
		province_definition = {
			# 3+ of our own madrasas of the madhhab
			any_location_in_province_definition = {
				count >= 3
				owner = root
				has_building = building_type:madrasa_of_$MADHHAB$_fiqh
			}
			# a mosque
			any_location_in_province_definition = {
				count >= 1
				owner = root
				has_building = building_type:temple
			}
			# no seat of the madhhab yet
			NOT = { any_location_in_province_definition = { has_building = building_type:seat_of_$MADHHAB$_jurists } }
			# not recently settled here
			NOT = { any_location_in_province_definition = { has_location_modifier = z_aqd_establishment_pending } }
			# a valid seat target we own: a city (or larger) with the sect present
			any_location_in_province_definition = {
				count >= 1
				owner = root
				"religion_percentage(religion:$SECT$)" >= 0.25
				OR = {
					location_rank ?= location_rank:city
					location_rank ?= location_rank:megalopolis
				}
			}
		}
	}
}

# ---------------------------------------------------------------------
# aqd_can_host_exile_seat = { MADHHAB SECT }: could 'this' country
# shelter the jurists of MADHHAB exiled by a crackdown next door - i.e.
# raise a Seat of that school in some city it owns? Deliberately WITHOUT
# aqd_province_has_madhhab_density's 3-madrasa/mosque gate: the exiles
# bring the school with them, so a bare city with the sect present is
# enough.
#
# religion = religion:$SECT$ is the STATE religion (not the group), so
# the seat building's country_potential (building_types/
# z_aqd_jurist_seats.txt) can never later prune a seat placed this way.
#
# Scope-free by design: this trigger is reused by z_aqd_exile.1's own
# target picker, and saved scopes do not survive trigger_event. The
# "not the crackdown country itself" check therefore lives inline in the
# selection-phase random_country limits in scripted_effects/
# z_aqd_jurist_effects.txt, not here.
#
# The 0.25 threshold matches building_types/z_aqd_madrasas.txt and
# aqd_province_has_madhhab_density above.
# ---------------------------------------------------------------------
aqd_can_host_exile_seat = {
	religion = religion:$SECT$
	has_policy = sharia_law_policy
	any_owned_location = {
		"religion_percentage(religion:$SECT$)" >= 0.25
		OR = {
			location_rank ?= location_rank:city
			location_rank ?= location_rank:megalopolis
		}
		NOT = { has_building = building_type:seat_of_$MADHHAB$_jurists }
		NOT = { has_location_modifier = z_aqd_establishment_pending }
	}
}
