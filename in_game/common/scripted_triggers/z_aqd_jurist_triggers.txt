# =====================================================================
# AQIDAH SCHOOLS OF CREED - Ulema Institutions: shared triggers.
#
# *** GOTCHA (cost real hours to find - read this before touching any
# $PARAM$ near a population-percentage check) ***
# "religion_percentage(religion:$SECT$)" - the QUOTED-STRING function-
# call form of this trigger - never substitutes a $PARAM$ placed inside
# the quotes. It parses the string with its own mini-parser, which does
# not go through the normal script-wide text substitution pass, so
# "religion_percentage(religion:$SECT$)" is evaluated literally with an
# EMPTY religion ("Type: none" / "field not set" in error.log) no matter
# what $SECT$ is bound to - in a trigger, an effect, ANY calling context.
# It fails perfectly silently as far as gameplay is concerned: the
# surrounding limit/trigger just reads false, so every check built on it
# (seat/madrasa placement, establishment, exile, foreign-jurist
# eligibility) looked like it was "doing nothing" with no error a casual
# read of the log would associate with the actual line.
# FIX: use the STRUCTURED block form instead when the religion must be
# parameterised - `religion = religion:$SECT$` there is a normal token,
# not inside a string, so ordinary text substitution applies and it
# works fine:
#     religion_percentage = { religion = religion:$SECT$  value >= 0.25 }
# A LITERAL religion inside the quoted form (e.g. "religion:sunni") is
# fine and unaffected - only a $PARAM$ inside the quotes is broken.
#
# *** SECOND GOTCHA, same family, also cost real hours: an UNREFERENCED
# $PARAM$ is a hard compile error, not a silent false ***
# When the "sect population gate disabled" TESTING edits above replaced
# the religion_percentage(...$SECT$...) line with two bare `always = yes`
# lines, that deleted the ONLY textual occurrence of $SECT$ in several
# scripted_trigger/scripted_effect bodies (aqd_province_has_madhhab_density,
# aqd_establish_save_target, aqd_exile_save_target,
# aqd_jurists_find_seat_loc). EU5's argument compiler scans a definition's
# body for every $PARAM$ token to learn its accepted parameters; a call
# site that then passes SECT = ... to a definition with no $SECT$ left in
# its body fails with "Compiling source for <name> failed for unknown
# arguments: SECT" in debug.log - and the WHOLE invocation silently
# no-ops (any random_owned_location/random_country inside it never runs,
# so any scope it would have saved is simply never set). This is exactly
# what broke establishment/exile/foreign-jurist seat placement for a
# whole session: the events still fired and applied their non-placement
# effects (opinion/legitimacy reactions), but placed no building, because
# the save_target/find_seat_loc effect that was supposed to do so never
# ran. FIX: never fully delete a parameter's last usage while disabling a
# gate under TESTING - replace it with a trivially-true check that still
# references the token, e.g. `religion_percentage = { religion =
# religion:$SECT$  value >= 0 }` instead of `always = yes`. That is also
# a one-line revert (0 -> 0.25) instead of restoring two deleted lines.
#
# aqd_is_eligible_jurist_source (formerly here - checked whether a
# candidate neighbor's own madhhab qualified as a "foreign jurists"
# source) has been REMOVED. "Foreign Jurists at the Gate" (z_aqd_jurists.
# 1) no longer fires from a blind monthly neighbor scan; it is now
# triggered directly by scripted_effects/z_aqd_decline_effects.txt's
# aqd_decline_migration_pick, dispatched from events/religion/z_aqd_
# decline_events.txt's "let them go" option, which reuses aqd_can_host_
# exile_seat (below) as its own host-eligibility check instead - see
# that trigger's own header and aqd_decline_migration_pick's header for
# the explicitly-accepted behavior differences from the old trigger.
#
# aqd_province_has_madhhab_density = { MADHHAB SECT }: does 'this' (a
# country, = root here) own a province_definition holding >= 3 of its
# own madrasas of MADHHAB plus >= 1 mosque (temple), with no seat of
# MADHHAB and whose capital carries no z_aqd_establishment_pending
# marker? Drives the establishment pulse in on_action/00_aqd_game_start.txt.
# This is a province-granularity STRUCTURAL gate. Under the share-based
# tier model it does NOT imply any realm-wide tier - it and the pulse
# prefilter in 00_aqd_game_start.txt are independent constraints; do
# not try to re-sync them.
# =====================================================================

# One fiqh madrasa per location, whatever the madhhab: a location that
# already holds any madrasa_of_<X>_fiqh takes no other. Used by the
# madrasa location_potential (building_types/z_aqd_madrasas.txt) and by
# the foreign-jurist placement (scripted_effects/z_aqd_jurist_effects.txt
# aqd_build_madrasas_of_madhhab_from). Location-scoped.
aqd_location_has_fiqh_madrasa = {
	OR = {
		has_building = building_type:madrasa_of_hanafi_fiqh
		has_building = building_type:madrasa_of_hanbali_fiqh
		has_building = building_type:madrasa_of_maliki_fiqh
		has_building = building_type:madrasa_of_shafii_fiqh
		has_building = building_type:madrasa_of_zahiri_fiqh
		has_building = building_type:madrasa_of_ismaili_fiqh
		has_building = building_type:madrasa_of_jafari_fiqh
		has_building = building_type:madrasa_of_zaidi_fiqh
		has_building = building_type:madrasa_of_ibadi_fiqh
	}
}

aqd_province_has_madhhab_density = {
	any_owned_location = {
		province_definition = {
			# 3+ of our own madrasas of the madhhab. owner ?= root, not
			# owner = root: a province_definition can include lake / sea
			# locations with no owner, and owner = root errors on those.
			any_location_in_province_definition = {
				count >= 3
				owner ?= root
				has_building = building_type:madrasa_of_$MADHHAB$_fiqh
			}
			# a mosque
			any_location_in_province_definition = {
				count >= 1
				owner ?= root
				has_building = building_type:temple
			}
			# no seat of the madhhab yet
			NOT = { any_location_in_province_definition = { has_building = building_type:seat_of_$MADHHAB$_jurists } }
			# not recently settled here
			NOT = { any_location_in_province_definition = { has_location_modifier = z_aqd_establishment_pending } }
			# not a province this SAME madhhab's jurists departed from within
			# the last 30 years (A Seat in Decline - main_menu/common/static_
			# modifiers/z_aqd_seat_departed.txt). Deliberately NOT added to
			# aqd_can_host_exile_seat below - a departed province can still
			# receive exile/foreign-jurist arrivals of the same madhhab, only
			# the natural three-madrasas path is blocked here.
			NOT = { any_location_in_province_definition = { has_location_modifier = z_aqd_seat_departed_$MADHHAB$ } }
			# a valid seat target we own: a town (or larger) with the sect present
			any_location_in_province_definition = {
				count >= 1
				owner ?= root
				# TESTING: sect population gate disabled (was >= 0.25) - kept
				# as a >= 0 check, not `always = yes`, so $SECT$ stays a
				# referenced parameter. Dropping the only $SECT$ reference
				# from this body makes every call site that passes SECT
				# fail to compile ("unknown arguments: SECT"), which
				# silently no-ops the whole trigger - this is exactly what
				# broke establishment/exile/foreign-jurist seat placement.
				religion_percentage = { religion = religion:$SECT$  value >= 0 }
				always = yes # TESTING: rural_settlement allowed
			}
		}
	}
}

# ---------------------------------------------------------------------
# aqd_can_host_exile_seat = { MADHHAB SECT }: could 'this' country
# shelter the jurists of MADHHAB exiled by a crackdown next door - i.e.
# raise a Seat of that school in some city it owns? Deliberately WITHOUT
# aqd_province_has_madhhab_density's 3-madrasa/mosque gate: the exiles
# bring the school with them, so a bare city with the sect present is
# enough.
#
# religion = religion:$SECT$ is the STATE religion (not the group), so
# the seat building's country_potential (building_types/
# z_aqd_jurist_seats.txt) can never later prune a seat placed this way.
#
# Scope-free by design: this trigger is reused by z_aqd_exile.1's own
# target picker, and saved scopes do not survive trigger_event. The
# "not the crackdown country itself" check therefore lives inline in the
# selection-phase random_country limits in scripted_effects/
# z_aqd_jurist_effects.txt, not here.
#
# The 0.25 threshold matches building_types/z_aqd_madrasas.txt and
# aqd_province_has_madhhab_density above.
# ---------------------------------------------------------------------
aqd_can_host_exile_seat = {
	religion = religion:$SECT$
	has_policy = sharia_law_policy
	any_owned_location = {
		always = yes # TESTING: sect population gate disabled
		always = yes # TESTING: rural_settlement allowed
		NOT = { has_building = building_type:seat_of_$MADHHAB$_jurists }
		NOT = { has_location_modifier = z_aqd_establishment_pending }
	}
}

# Country-scoped. Cheap short-circuiting prefilter for the halqa pulse
# (on_action/00_aqd_game_start.txt): does this realm hold ANY jurist
# seat at all? Most Muslim realms hold none, so this keeps the province
# scan below off all but ~15-20 realms.
aqd_country_owns_any_jurist_seat = {
	any_owned_location = {
		OR = {
			has_building = building_type:seat_of_hanafi_jurists
			has_building = building_type:seat_of_hanbali_jurists
			has_building = building_type:seat_of_maliki_jurists
			has_building = building_type:seat_of_shafii_jurists
			has_building = building_type:seat_of_zahiri_jurists
			has_building = building_type:seat_of_ismaili_jurists
			has_building = building_type:seat_of_jafari_jurists
			has_building = building_type:seat_of_zaidi_jurists
			has_building = building_type:seat_of_ibadi_jurists
		}
	}
}

# province_definition-scoped. Does this province hold an owned seat of
# <X> that could host ITS OWN ḥalqa right now? Previously also required
# >= 3 of the realm's own madrasas of <X> in the same province (the same
# clause as aqd_province_has_madhhab_density above) - removed by design:
# a ḥalqa is now initiated off a bare seat, no madrasa density needed.
# NINE concrete triggers, NOT one parameterised one: EU5 does not
# substitute $PARAM$ when a scripted_trigger is invoked from inside
# another scripted_trigger (it only works from an effect / an effect's
# limit). Used by aqd_country_has_any_halqa_ready below and by
# events/religion/z_aqd_halqa_events.txt (the picker limit and each
# random_list weight entry).
#
# Each excludes a location currently under THAT SAME madhhab's own fatwa
# ban (main_menu/common/static_modifiers/z_aqd_fatwa_displeased.txt,
# events/religion/z_aqd_fatwa_events.txt) OR currently under THAT SAME
# madhhab's own crackdown eviction (main_menu/common/static_modifiers/
# z_aqd_jurist_evicted.txt, scripted_effects/z_aqd_jurist_effects.txt's
# aqd_crackdown_effect) AND currently carrying an unexpired ḥalqa of its
# OWN madhhab, in any of its three classes
# (main_menu/common/static_modifiers/z_aqd_halqa.txt's z_aqd_halqa_
# state_<X> / _orthodox_<X> / _heterodox_<X> - tolerated is deliberately
# not checked here, see below) - one combined scan per trigger, not
# three: every exclusion is checked on the SAME location as the seat
# building, both more correct (a ḥalqa fired from this seat is what
# would need to have lapsed) and free (no extra any_location_in_
# province_definition passes). Checking at the seat's own location is
# equivalent to checking the whole province, since aqd_halqa_apply_state/
# _orthodox/_heterodox (scripted_effects/z_aqd_jurist_effects.txt) always
# stamp every owned location in the province uniformly - if this one
# location doesn't have the key, none in the province do.
#
# This is what makes ḥalqas run in PARALLEL per madhhab: previously a
# single realm-wide aqd_location_has_active_halqa 28-way OR (checking
# every madhhab AND every class at once) blocked the entire province the
# moment ANY seat's circle was active, so a co-located or co-province
# seat of a DIFFERENT madhhab could never start its own circle until the
# first one fully expired. Folding the "not already active" check in
# HERE instead, scoped to just this one madhhab's own three class keys,
# means a Hanafi circle running no longer blocks a Maliki circle from
# starting at the same time - each madhhab's own occupancy is tracked
# (implicitly, via its own modifier's presence) completely independently
# of every other madhhab's. aqd_halqa_tolerated is intentionally excluded
# from every one of these nine checks - it is madhhab-agnostic (din_i_
# ilahi has no fiqh loyalty to any one rite) and applies to the whole
# province as a single shared state, so it plays no role in any single
# madhhab's own readiness.
aqd_halqa_prov_ready_hanafi = {
	any_location_in_province_definition = {
		has_building = building_type:seat_of_hanafi_jurists
		NOT = { has_location_modifier = z_aqd_fatwa_displeased_hanafi }
		NOT = { has_location_modifier = z_aqd_jurist_evicted_hanafi }
		NOT = { has_location_modifier = z_aqd_halqa_state_hanafi }
		NOT = { has_location_modifier = z_aqd_halqa_orthodox_hanafi }
		NOT = { has_location_modifier = z_aqd_halqa_heterodox_hanafi }
	}
}

# Country-scoped. Does the realm own at least one province that could
# host a ḥalqa right now - a seat of some madhhab (no madrasa density
# required) whose own three ḥalqa classes are all currently free (folded
# into aqd_halqa_prov_ready_<X> itself now, see that trigger's own header
# for why the old realm-wide "any ḥalqa active anywhere" blanket check
# was removed from here). ONE any_owned_location pass: the 9 madhhab
# branches live inside one OR, and has_building = seat_of_X is first in
# each AND so the province sub-scan only runs for the handful of
# locations that actually hold that seat.
aqd_country_has_any_halqa_ready = {
	any_owned_location = {
		OR = {
			AND = { has_building = building_type:seat_of_hanafi_jurists   province_definition = { aqd_halqa_prov_ready_hanafi = yes } }
			AND = { has_building = building_type:seat_of_hanbali_jurists  province_definition = { aqd_halqa_prov_ready_hanbali = yes } }
			AND = { has_building = building_type:seat_of_maliki_jurists   province_definition = { aqd_halqa_prov_ready_maliki = yes } }
			AND = { has_building = building_type:seat_of_shafii_jurists   province_definition = { aqd_halqa_prov_ready_shafii = yes } }
			AND = { has_building = building_type:seat_of_zahiri_jurists   province_definition = { aqd_halqa_prov_ready_zahiri = yes } }
			AND = { has_building = building_type:seat_of_ismaili_jurists  province_definition = { aqd_halqa_prov_ready_ismaili = yes } }
			AND = { has_building = building_type:seat_of_jafari_jurists   province_definition = { aqd_halqa_prov_ready_jafari = yes } }
			AND = { has_building = building_type:seat_of_zaidi_jurists    province_definition = { aqd_halqa_prov_ready_zaidi = yes } }
			AND = { has_building = building_type:seat_of_ibadi_jurists    province_definition = { aqd_halqa_prov_ready_ibadi = yes } }
		}
	}
}

# Location-scoped: does a seat of the REALM'S OWN current sharia_law
# policy already stand here? Used by scripted_effects/z_aqd_crackdown_
# vacancy_effects.txt's aqd_crackdown_offer_seat_vacancy to skip the
# whole vacancy-offer roll when placing the state madhhab's seat at a
# just-razed location would be redundant - a location can hold more than
# one madhhab's seat at once (building_types/z_aqd_jurist_seats.txt's
# location_potential only ever excludes a duplicate of the SAME madhhab),
# so a razed rival seat's location could easily already carry the state
# madhhab's own seat separately.
aqd_state_madhhab_seat_already_here = {
	OR = {
		AND = { root = { has_policy = hanafi_policy }    has_building = building_type:seat_of_hanafi_jurists }
		AND = { root = { has_policy = hanbali_policy }   has_building = building_type:seat_of_hanbali_jurists }
		AND = { root = { has_policy = maliki_policy }    has_building = building_type:seat_of_maliki_jurists }
		AND = { root = { has_policy = shafii_policy }    has_building = building_type:seat_of_shafii_jurists }
		AND = { root = { has_policy = zahiri_policy }    has_building = building_type:seat_of_zahiri_jurists }
		AND = { root = { has_policy = ismaili_policy }   has_building = building_type:seat_of_ismaili_jurists }
		AND = { root = { has_policy = jafari_policy }    has_building = building_type:seat_of_jafari_jurists }
		AND = { root = { has_policy = zaidi_policy }     has_building = building_type:seat_of_zaidi_jurists }
		AND = { root = { has_policy = ibadi_policy }     has_building = building_type:seat_of_ibadi_jurists }
	}
}

# Country-scoped sibling of aqd_state_madhhab_seat_already_here above -
# same OR/AND idiom, but keyed to var:aqd_compat_madhhab (the incoming
# madhhab z_aqd_jurists.1's immediate resolves, events/religion/z_aqd_
# jurist_events.txt) instead of root's own current policy, and scoped
# through `capital` (the correct EU5 scope link for a country's capital
# location - confirmed via the base game's own scripted_triggers, e.g.
# `capital.continent` / `capital ?= { ... }`) instead of any_owned_
# location. Gates z_aqd_jurists.1's new "settle them in the capital"
# option: true only when the capital does not already hold a seat of the
# madhhab being offered.
aqd_capital_lacks_seat_of_compat_madhhab = {
	capital = {
		NOT = {
			OR = {
				AND = { root = { var:aqd_compat_madhhab = 1 }  has_building = building_type:seat_of_hanafi_jurists }
				AND = { root = { var:aqd_compat_madhhab = 2 }  has_building = building_type:seat_of_hanbali_jurists }
				AND = { root = { var:aqd_compat_madhhab = 3 }  has_building = building_type:seat_of_maliki_jurists }
				AND = { root = { var:aqd_compat_madhhab = 4 }  has_building = building_type:seat_of_shafii_jurists }
				AND = { root = { var:aqd_compat_madhhab = 5 }  has_building = building_type:seat_of_zahiri_jurists }
				AND = { root = { var:aqd_compat_madhhab = 6 }  has_building = building_type:seat_of_ismaili_jurists }
				AND = { root = { var:aqd_compat_madhhab = 7 }  has_building = building_type:seat_of_jafari_jurists }
				AND = { root = { var:aqd_compat_madhhab = 8 }  has_building = building_type:seat_of_zaidi_jurists }
				AND = { root = { var:aqd_compat_madhhab = 9 }  has_building = building_type:seat_of_ibadi_jurists }
			}
		}
	}
}

# Province-wide sibling of aqd_state_madhhab_seat_already_here above - does
# a seat of the realm's own current sharia_law policy already stand
# ANYWHERE in this location's own province (not just this exact location)?
# change_building_level_in_location bypasses the building's own
# location_potential (which only ever excludes a duplicate at the SAME
# location, never elsewhere in the same province), so without this check
# z_aqd_crackdown_vacancy.1's "fill it with our own madhhab" option could
# plant a second seat of a madhhab that already holds one elsewhere in the
# same province. Same 9-branch has_policy/has_building shape, just wrapped
# in province_definition/any_location_in_province_definition (confirmed
# idiom: scripted_effects/z_aqd_establishment_effects.txt's own seat-
# duplicate guard) instead of a bare location check.
aqd_state_madhhab_seat_already_in_province = {
	OR = {
		AND = { root = { has_policy = hanafi_policy }    province_definition = { any_location_in_province_definition = { owner ?= root  has_building = building_type:seat_of_hanafi_jurists } } }
		AND = { root = { has_policy = hanbali_policy }   province_definition = { any_location_in_province_definition = { owner ?= root  has_building = building_type:seat_of_hanbali_jurists } } }
		AND = { root = { has_policy = maliki_policy }    province_definition = { any_location_in_province_definition = { owner ?= root  has_building = building_type:seat_of_maliki_jurists } } }
		AND = { root = { has_policy = shafii_policy }    province_definition = { any_location_in_province_definition = { owner ?= root  has_building = building_type:seat_of_shafii_jurists } } }
		AND = { root = { has_policy = zahiri_policy }    province_definition = { any_location_in_province_definition = { owner ?= root  has_building = building_type:seat_of_zahiri_jurists } } }
		AND = { root = { has_policy = ismaili_policy }   province_definition = { any_location_in_province_definition = { owner ?= root  has_building = building_type:seat_of_ismaili_jurists } } }
		AND = { root = { has_policy = jafari_policy }    province_definition = { any_location_in_province_definition = { owner ?= root  has_building = building_type:seat_of_jafari_jurists } } }
		AND = { root = { has_policy = zaidi_policy }     province_definition = { any_location_in_province_definition = { owner ?= root  has_building = building_type:seat_of_zaidi_jurists } } }
		AND = { root = { has_policy = ibadi_policy }     province_definition = { any_location_in_province_definition = { owner ?= root  has_building = building_type:seat_of_ibadi_jurists } } }
	}
}

# Location-scoped: does THIS crackdown's own razed-madrasa footprint still
# have anything left to convert anywhere in this location's own province?
# Reads var:aqd_crackdown_razed_seat_madhhab (root, numeric 1-9 - the
# CRACKED-DOWN madhhab, set by whichever destroy-seat wrapper ran,
# scripted_effects/z_aqd_jurist_effects.txt) to pick which of the 9
# z_aqd_crackdown_madrasa_razed_<X> markers to look for (stamped on every
# madrasa of that madhhab this SAME crackdown razed, scripted_effects/
# z_aqd_crackdown_vacancy_effects.txt's aqd_crackdown_sweep_province_
# state_madrasas consumes them). Used by z_aqd_crackdown_vacancy.1's own
# "fill it with our own madhhab" option: even when aqd_state_madhhab_
