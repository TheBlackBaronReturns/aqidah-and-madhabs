# =====================================================================
# AQIDAH SCHOOLS OF CREED - Seat Collapse: triggers.
#
# NINE concrete location-scoped triggers, one per madhhab, each
# replicating EXACTLY what that madhhab's own seat_of_<X>_jurists used
# to check via its own (now-removed) remove_if field in building_types/
# z_aqd_jurist_seats.txt - the demographic floor (that madhhab's own
# SECT population under 5% at this location) or the owner ceasing to be
# Muslim. Keyed to each madhhab's own SECT: sunni for hanafi/hanbali/
# maliki/shafii/zahiri, shia for ismaili/jafari/zaidi, ibadi for ibadi
# alone - same grouping aqd_halqa_dispatch (scripted_effects/z_aqd_
# jurist_effects.txt) already uses.
#
# Consumed by on_action/00_aqd_game_start.txt's aqd_seat_collapse_pulse
# (cheap prefilter: the umbrella aqd_seat_collapsing below) and
# scripted_effects/z_aqd_seat_collapse_effects.txt's aqd_seat_collapse_
# pick (the actual per-madhhab reservoir-sampling pick, same Algorithm-R
# idiom as aqd_jihad_fatwa_pick_seat).
# =====================================================================

aqd_seat_collapsing_hanafi = {
	has_building = building_type:seat_of_hanafi_jurists
	OR = {
		owner ?= { religion.group != religion_group:muslim }
		"religion_percentage(religion:sunni)" < 0.05
	}
}
