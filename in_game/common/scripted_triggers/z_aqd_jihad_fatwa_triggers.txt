# =====================================================================
# AQIDAH SCHOOLS OF CREED - the Fatwa of Jihad: triggers.
#
# Covers TWO participation paths that share the whole rest of this
# feature (same event z_aqd_jfatwa.1, same z_aqd_fatwa_of_jihad_<X>/
# z_aqd_jihad_blessed_<X>/z_aqd_jihad_declined_<X> modifier keys, same
# picker/approval engine) - aqd_is_jihad_participant (OFFENSIVE: root is
# fighting under cb_jihad or is a jihad IO member) and aqd_is_defensive_
# jihad_participant (DEFENSIVE: a heathen power has declared a religious
# war on root). The only behavioural difference between the two paths is
# that heterodox seats are excluded from the offensive call but allowed
# to answer the defensive one - see aqd_jihad_seat_eligible_<X> below.
# =====================================================================

# Country-scoped: is root part of a jihad right now, either as a formal
# member of the jihad international_organization_type (leader or any
# joiner alike - jihad.txt's own on_joined/can_join_trigger make no
# leader/member distinction once membership exists), or as a war
# participant fighting under casus_belli:cb_jihad without formal
# membership (a subject/ally dragged into the jihad's war by alliance/
# call-to-arms). is_an_attacker (not attacker_leader) is deliberate - it
# must catch every co-attacker, not just the original declarer (confirmed
# idiom: events/disaster/decline_of_mali.txt's own any_attacker usage;
# is_member_of_international_organization_of_type = { type = jihad } is
# the confirmed bare-key form for a TYPE, not a single named IO instance -
# common/government_reforms/republic.txt's / events/debug/qa_debug.txt's
# own type = union usage). This is the OFFENSIVE path only - see aqd_is_
# defensive_jihad_participant below for the mirror-image defensive one.
aqd_is_jihad_participant = {
	OR = {
		is_member_of_international_organization_of_type = { type = jihad }
		any_current_war = {
			is_an_attacker = root
			casus_belli ?= casus_belli:cb_jihad
		}
	}
}

# Country-scoped: is root currently defending against a heathen (non-
# Muslim) power's RELIGIOUS war - the defensive counterpart to aqd_is_
# jihad_participant above. is_a_defender = root (not defender_leader)
# deliberately catches every co-defender - a subject/ally dragged in by
# alliance/call-to-arms counts too, mirroring is_an_attacker's own
# inclusive semantics on the offensive side (base game common/
# scripted_triggers/war_triggers.txt confirms both is_a_defender and
# is_an_attacker exist as symmetric war-scoped triggers). is_religious_war
# (same file) reads the war's own casus_belli custom_tags = {
# religious_casus_belli } - it is true for cb_jihad, cb_crusade,
# cb_heretic, deus_vult, crush_hussitism and cb_religious_conformance
# alike, so no hand-maintained CB allow-list is needed here (contrast
# scripted_triggers/z_aqd_fatwa_triggers.txt's aqd_fatwa_core_conditions,
# which has to blacklist specific weak-CB types for an unrelated reason).
#
# NOT = { religion.group = root.religion.group } on attacker_leader is
# what excludes heretic-on-heretic wars (e.g. Shia declaring on Sunni via
# cb_heretic) while still catching a genuine heathen aggressor - and it
# does so for free, with no extra logic: cb_heretic's own declare_enabled
# requires attacker and defender to share the same religion.group, so a
# heretic-CB attacker can never fail this check to begin with. Likewise
# cb_jihad can only ever be declared BY a Muslim, so a Muslim root can
# never legitimately be its defender either - both classes of false
# positive are structurally impossible, not merely filtered out.
aqd_is_defensive_jihad_participant = {
	any_current_war = {
		is_a_defender = root
		is_religious_war = yes
		attacker_leader = {
			NOT = { religion.group = root.religion.group }
		}
	}
}

# Location-scoped, NINE concrete single-madhhab triggers (called inside
# a random_owned_location limit at root's own scope, one call per
# madhhab, from scripted_effects/z_aqd_jihad_fatwa_effects.txt's
# aqd_jihad_fatwa_pick_seat). A location can legitimately hold MULTIPLE
# different madhhab seats at once - building_types/z_aqd_jurist_seats.txt's
# location_potential for each seat_of_<X>_jurists only excludes a
# duplicate of the SAME madhhab, never a different one, and the
# establishment event (events/religion/z_aqd_establishment_events.txt)
# fires independently per madhhab once ITS OWN 3-madrasa threshold is
# met, with no check for another madhhab's seat already standing there.
# Each of these nine checks its own aqd_seat_madhhab_not_heterodox_to_
# root_<X> (scripted_triggers/z_aqd_fatwa_triggers.txt - the single-
# madhhab form, NOT the 9-way umbrella, which would wrongly let a
# co-located seat of a DIFFERENT, acceptable madhhab paper over this
# one's own heterodoxy) UNLESS root is currently a defensive jihad
# participant (aqd_is_defensive_jihad_participant above), in which case a
# heterodox seat is allowed through too - a heathen assault on Islam is
# meant to draw in every school's jurists, not only the ruler's own; an
# offensive call to arms is not. This is the ONLY difference between the
# two participation paths - everything else below applies identically
# regardless of which one is currently true, including the approval roll
# itself: a heterodox seat still scores its full tolerance_heretic
# penalty (script_values/z_aqd_madhhab_approval.txt), so it remains
# naturally less likely to bless than an orthodox one even once eligible.
# Each branch also checks its own single z_aqd_fatwa_displeased_<X> /
# z_aqd_fatwa_of_jihad_<X> keys (NOT aqd_location_has_any_fatwa_ban's
# blanket 9-way OR, for the same reason) - so a Hanafi seat already
# banned or on cooldown never blocks a co-located Maliki seat, and vice
# versa. has_location_modifier = z_aqd_fatwa_pending_pick stays a single
# shared, blanket, LOCATION-wide check on all nine - it is a same-tick,
# stamp-and-immediately-consume marker (scripted_effects/z_aqd_madhhab_
# approval_effects.txt's aqd_stamp_pending_pick), never a standing
# cooldown, so there is no real window where it would wrongly block a
# second seat at the same location.
#
# z_aqd_jihad_declined_<X> (main_menu/common/static_modifiers/z_aqd_
# jihad_declined.txt) is the "one shot per war" gate: stamped by
# scripted_effects/z_aqd_jihad_fatwa_effects.txt's aqd_jihad_fatwa_
# attempt the moment THIS exact seat's approval roll declines, so it
# can never be re-picked for the rest of the current jihad - previously
# a decline cost nothing, so the same silent-miss pulse just tried again
# next month indefinitely, making eventual success for SOME seat a near-
# certainty over a long enough war regardless of how low any individual
# seat's own approval score was. z_aqd_jihad_blessed_<X> (main_menu/
# common/static_modifiers/z_aqd_jihad_blessed.txt) is its SUCCESS-side
# mirror - stamped instead of re-picking a seat that already blessed
# this same war, so its own Fatwa of Jihad event doesn't pop up again
# and again every time z_aqd_fatwa_of_jihad_<X>'s mobilization bonus
# happens to lapse before the war does. Both are cleared realm-wide once
# the jihad episode ends (on_action/00_aqd_game_start.txt's aqd_jihad_
# episode_cleanup_pulse, and defensively again at the start of the next
# episode), so a fresh jihad later gives every seat a clean slate. These
# are SEPARATE, independent gates from z_aqd_fatwa_of_jihad_<X> itself
# (that one is the real post-success throttle, deliberately persisting
# ACROSS wars, untouched by episode boundaries) - all three apply, none
# substitutes for another.
#
# z_aqd_jurist_evicted_<X> (main_menu/common/static_modifiers/z_aqd_
# jurist_evicted.txt, a flat 20 years, scripted_effects/z_aqd_jurist_
# effects.txt's aqd_crackdown_effect) is also excluded - a crackdown-
# persecuted seat has no standing left to bless a jihad either, same
# reasoning aqd_fatwa_seat_eligible_<X> (scripted_triggers/z_aqd_fatwa_
# triggers.txt) and aqd_debate_seat_eligible_<X> (scripted_triggers/
# z_aqd_debate_triggers.txt) both apply for their own features.
aqd_jihad_seat_eligible_hanafi = {
	OR = {
		aqd_seat_madhhab_not_heterodox_to_root_hanafi = yes
		root = { aqd_is_defensive_jihad_participant = yes }
	}
	NOT = { has_location_modifier = z_aqd_fatwa_displeased_hanafi }
	NOT = { has_location_modifier = z_aqd_jurist_evicted_hanafi }
	NOT = { has_location_modifier = z_aqd_fatwa_pending_pick }
	NOT = { has_location_modifier = z_aqd_fatwa_of_jihad_hanafi }
	NOT = { has_location_modifier = z_aqd_jihad_declined_hanafi }
	NOT = { has_location_modifier = z_aqd_jihad_blessed_hanafi }
}
