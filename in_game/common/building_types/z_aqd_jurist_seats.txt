# =====================================================================
# AQIDAH SCHOOLS OF CREED - Ulema Institutions: Jurist Seats.
#
# A seat is the standing juristic authority of a madhhab (a sharia_law
# policy, see laws/z_aqd_sharia_law.txt) in a province - the body that
# trains and appoints its qadis.
#
# Seats are EVENT-ONLY. They are never buildable through the UI
# (allow = { always = no }). They come to exist five ways:
#
#  - the establishment event (events/religion/z_aqd_establishment_events.txt
#    via scripted_effects/z_aqd_establishment_effects.txt): a province
#    that has 3 madrasas of madhhab X (building_types/z_aqd_madrasas.txt)
#    plus at least one mosque, no seat of X yet, and an owned city (or
#    larger) to house it, offers its owner the choice to raise a seat
#    there - the province capital if eligible - or crack down;
#  - the "Foreign Jurists at the Gate" event (events/religion/
#    z_aqd_jurist_events.txt via aqd_jurists_accept): a neighbouring
#    realm's madhhab is invited in and granted a seat in one of our
#    cities directly, bypassing the grassroots chain above;
#  - the "Jurists in Exile" event (events/religion/z_aqd_exile_events.txt
#    via aqd_exile_plant_seat): jurists broken by a crackdown next door
#    are sheltered and given a seat here;
#  - the historically-seeded seats placed at game start for the
#    prominent 1337 juristic centres (scripted_effects/
#    z_aqd_starting_schools.txt's aqd_seed_starting_seats);
#  - conquest of a location that already holds one.
#
# All five place the seat with change_building_level_in_location, which
# bypasses `allow` / `location_potential` / `country_potential`.
#
# country_potential = { always = no } is LOAD-BEARING. A seat is an
# estate_category clergy_estate building with construction_demand /
# build_time set, so any country_potential that resolves TRUE lets the
# clergy estate auto-construct it (allow = { always = no } only hides the
# player's manual build button - it does not stop estate construction or
# the game-start building fill). With the old country_potential = {
# religion = religion:sunni } every Sunni realm sprouted unseeded seats
# of random madhhabs from turn zero - e.g. a Zahiri seat in the Ottoman
# lands. { always = no } is the same value base uses on its own
# event-only buildings (seat_of_cardinal, great_madrasa, everything in
# event_only_buildings.txt); those are placed only by script, exactly
# like these.
#
# location_potential below is now dead weight (every real placement path
# bypasses it) but harmless. remove_if (0.05 sect floor + non-Muslim
# owner) stays as the demographic-cleanup path.
#
# A seat can never be manually demolished (can_destroy = always = no).
# It is removed only by scripted_effects/z_aqd_jurist_effects.txt's
# aqd_crackdown_effect - which razes it with destroy_building_forcefully
# (plain destroy_building is refused by can_destroy = no) - or by
# remove_if. Switching the realm's official madhhab does not remove an
# existing seat.
#
# Counted into script_values/z_aqd_madhhab_counts.txt's per-madhhab
# institution score (weight 4, vs. 1 for a madrasa), which drives the
# share-based tier in auto_modifiers/z_aqd_madhhab_tiers.txt.
#
# ACTIVITY: the seat periodically throws a "ḥalqa" - a province-named
# flavour event (events/religion/z_aqd_halqa_events.txt,
# static_modifiers/z_aqd_halqa.txt) stamping a temporary province-wide
# modifier whose character (state / orthodox / heterodox / tolerated)
# depends on how the realm's aqidah judges the madhhab. One ḥalqa per
# MADHHAB per province at a time, not one overall - a different madhhab's
# seat in the same (or a co-located) province can run its own ḥalqa in
# parallel (scripted_triggers/z_aqd_jurist_triggers.txt's aqd_halqa_
# prov_ready_<X>, checked per madhhab).
#
# ICON: each seat_of_<X>_jurists uses its madhhab's religious-school
# icon, supplied as main_menu/gfx/interface/icons/buildings/
# seat_of_<X>_jurists.dds - the path the engine's GetBuildingIcon()
# resolves by convention (no interface/*.gfx entry needed). The eight
# non-Ibadi seats copy the matching gfx/interface/icons/religious_schools/
# <X>_school.dds; base has no ibadi_school icon, so seat_of_ibadi_jurists
# uses gfx/interface/icons/religion/ibadi.dds instead. Placeholder art -
# real seat icons can drop in at the same paths later.
# =====================================================================

seat_of_hanafi_jurists = {
	estate = clergy_estate
	category = estate_category
	pop_type = clergy
	employment_size = large_religious_building_employment
	max_levels = estate_building_single_level

	rural_settlement = yes  # TESTING: was no - revert before release
	town = yes  # TESTING: was no - revert before release
	city = yes
	megalopolis = yes

	build_time = estate_build_time
	construction_demand = estate_construct_building

	country_potential = { always = no }
	allow = { always = no }
	can_destroy = { always = no }

	location_potential = {
		NOT = { has_building = building_type:seat_of_hanafi_jurists }
	}
	remove_if = {
		location = {
			OR = {
				owner ?= { religion.group != religion_group:muslim }
				"religion_percentage(religion:sunni)" < 0.05
			}
		}
	}

	possible_production_methods = { estate_building_input }
	modifier = {
		global_clergy_max_literacy = 5
	}
}
