# =====================================================================
# AQIDAH SCHOOLS OF CREED - Ulema Institutions: madhhab madrasas.
#
# The madrasa is the buildable grassroots unit of a school of
# jurisprudence (madhhab, a sharia_law policy). It feeds script_values/
# z_aqd_madhhab_counts.txt's per-madhhab institution score (weight 1,
# vs. 4 for a jurist seat), driving the tiered auto_modifiers in
# auto_modifiers/z_aqd_madhhab_tiers.txt.
#
# It is a CROWN religious_category building, mirroring the base
# madrasa/"Maktab" (common/building_types/religion_buildings.txt) - NOT
# an estate building. An `estate = clergy_estate` field would hand
# construction to the clergy estate and take it out of the player's
# build menu; that is why the fiqh madrasa now carries none.
#
# WHO CAN BUILD WHICH (location_potential, evaluated location-scoped;
# `exists = owner` guards it so it never errors on unowned locations):
#  - the nation's OWN current sharia_law madhhab: anywhere the sect
#    population threshold (>= 25%) is met - the `owner has_policy`
#    branch;
#  - any OTHER (foreign) madhhab: only in a location whose AREA already
#    contains a seat_of_<X>_jurists - the `area = { any_location_in_area
#    = { has_building = seat } }` branch. Models a madhhab spreading
#    outward from a juristic centre the realm has (established or
#    conquered).
#
# The foreign-jurist event (events/religion/z_aqd_jurist_events.txt via
# scripted_effects/z_aqd_jurist_effects.txt's
# aqd_build_madrasas_of_madhhab_from) plants 1-3 of these directly with
# change_building_level_in_location, which bypasses location_potential -
# so a realm can be seeded with a foreign madhhab with no prior seat,
# and if that province then reaches 3 madrasas + a mosque the
# establishment event (events/religion/z_aqd_establishment_events.txt)
# can grant it a seat without conquest. The event places only in
# non-rural owned locations of a province that has >= 25% of the sect
# somewhere; the madrasa's remove_if only culls on non-Muslim conquest
# (matching base), so a minority-school madrasa is not instantly pruned.
#
# The modifier is DELIBERATELY identical to the base madrasa/"Maktab"
# (common/building_types/religion_buildings.txt) payload, plus one
# per-madhhab societal-axis nudge - madrasas are not weaker than the
# generic school. The single-level cap and clergy food upkeep are the
# built-in brakes on the AI over-building them.
#
# Removed by scripted_effects/z_aqd_jurist_effects.txt's
# aqd_crackdown_destroy_<X>_madrasa (crackdown), or auto-removed by
# remove_if on demographic collapse / non-Muslim conquest.
# =====================================================================

madrasa_of_hanafi_fiqh = {
	category = religious_category
	pop_type = clergy
	employment_size = large_religious_building_employment
	max_levels = 1

	graphical_tags = { medium_religious }

	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes

	build_time = religious_building_time
	construction_demand = town_building_construction

	country_potential = { religion = religion:sunni }
	location_potential = {
		exists = owner
		"religion_percentage(religion:sunni)" >= 0.25
		NOT = { aqd_location_has_fiqh_madrasa = yes }
		OR = {
			owner = { has_policy = hanafi_policy }
			area = {
				any_location_in_area = {
					has_building = building_type:seat_of_hanafi_jurists
				}
			}
		}
	}
	remove_if = {
		location = {
			owner ?= { religion.group != religion_group:muslim }
		}
	}

	possible_production_methods = { estate_building_input }
	modifier = {
		local_clergy_max_literacy = 5
		local_pop_conversion_speed_modifier = 0.1
		local_clergy_food_consumption = 0.1
		local_tribal_promotion = 0.1
		monthly_towards_mysticism = societal_value_minor_monthly_move
	}
}
