# =====================================================================
# AQIDAH SCHOOLS OF CREED - Ulema Institutions: madhhab madrasas.
#
# The madrasa is the buildable grassroots unit of a school of
# jurisprudence (madhhab, a sharia_law policy). It feeds script_values/
# z_aqd_madhhab_counts.txt's per-madhhab institution score (weight 1,
# vs. 4 for a jurist seat), driving the tiered auto_modifiers in
# auto_modifiers/z_aqd_madhhab_tiers.txt.
#
# It is an ESTATE building (category = estate_category, estate =
# clergy_estate), mirroring vanilla's cathedral/parish_church
# (common/building_types/estate_buildings.txt) rather than the player-
# built base madrasa/"Maktab" (religion_buildings.txt / z_aqd_maktab.txt).
# The clergy estate raises and maintains these on its own initiative,
# not the crown - it is off the player's direct build menu, the same
# way cathedral/parish_church are. This keeps the fiqh madrasa from
# freely stacking with the player-built Maktab as a second, on-demand
# copy of the same literacy/conversion bonus in every eligible city;
# the estate still targets locations per the eligibility rules below,
# just on its own (control/wealth-share-driven) schedule instead of the
# player's. construction_demand is estate_construct_building (no
# physical construction goods), matching cathedral/parish_church - not
# town_building_construction, which assumes a player-funded build.
#
# WHO CAN BUILD WHICH (location_potential, evaluated location-scoped;
# `exists = owner` guards it so it never errors on unowned locations):
#  - the nation's OWN current sharia_law madhhab: anywhere the sect
#    population threshold (>= 25%) is met - the `owner has_policy`
#    branch;
#  - any OTHER (foreign) madhhab: only in a location whose AREA already
#    contains a seat_of_<X>_jurists - the `area = { any_location_in_area
#    = { has_building = seat } }` branch. Models a madhhab spreading
#    outward from a juristic centre the realm has (established or
#    conquered).
#
# The "Foreign Jurists at the Gate" event (events/religion/
# z_aqd_jurist_events.txt) grants the visitors' madhhab a Seat directly
# (aqd_jurists_accept / aqd_jurists_accept_capital), placed with change_
# building_level_in_location, which bypasses location_potential so a
# foreign madhhab with no prior area seat can still be granted one. (The
# dormant aqd_build_madrasas_of_madhhab_from primitive can still seed
# 1-3 of these in a province without a seat, toward an establishment - it
# is currently unreferenced.) These placements target non-rural owned
# locations of a province that has >= 25% of the sect somewhere; the
# madrasa's remove_if only culls on non-Muslim conquest (matching base),
# so a minority-school madrasa is not instantly pruned.
#
# Payload (per level, x goods access): local_clergy_max_literacy 5,
# local_pop_conversion_speed_modifier 0.1, local_clergy_food_consumption
# 0.1 (the cost side), local_cultural_tradition 0.1. Upkeep is
# aqd_madrasa_maintenance (paper 0.1 / cloth 0.2, no output;
# production_methods/z_aqd_pm.txt) - the higher-learning building costs
# real goods to run, unlike the elementary Maktab. The single-level cap,
# clergy food upkeep, and goods upkeep are the brakes on over-building.
#
# Removed by scripted_effects/z_aqd_jurist_effects.txt's
# aqd_crackdown_destroy_<X>_madrasa (crackdown), or auto-removed by
# remove_if on demographic collapse / non-Muslim conquest.
# =====================================================================

madrasa_of_hanafi_fiqh = {
	category = estate_category
	estate = clergy_estate
	pop_type = clergy
	employment_size = large_religious_building_employment
	max_levels = 1

	graphical_tags = { medium_religious }

	rural_settlement = yes  # TESTING: was no - revert before release
	town = yes
	city = yes
	megalopolis = yes

	build_time = religious_building_time
	construction_demand = estate_construct_building

	country_potential = { religion = religion:sunni }
	location_potential = {
		exists = owner
		# TESTING: sect population gate disabled - was "religion_percentage(religion:sunni)" >= 0.25
		NOT = { aqd_location_has_fiqh_madrasa = yes }
		OR = {
			owner = { has_policy = hanafi_policy }
			area = {
				any_location_in_area = {
					has_building = building_type:seat_of_hanafi_jurists
				}
			}
		}
	}
	remove_if = {
		location = {
			owner ?= { religion.group != religion_group:muslim }
		}
	}

	possible_production_methods = { aqd_madrasa_maintenance }
	modifier = {
		local_clergy_max_literacy = 5
		local_pop_conversion_speed_modifier = 0.1
		local_clergy_food_consumption = 0.1
		local_cultural_tradition = 0.1
	}
}
