# Aqidah: Schools of Creed - migration hooks.
#
# IMPORTANT: on_game_start's own `effect` field does NOT merge across
# files - confirmed via jomini_onaction.cpp's own log warning ("There
# is more than one 'effect' defined... using most recent"). Base game's
# common/on_action/_hardcoded.txt:2 also puts a raw `effect` directly
# on on_game_start, and whichever file's `effect` block loads last wins
# outright, discarding the other entirely. Only the `on_actions = {}`
# LIST field actually merges/accumulates across files (confirmed by
# four_yearly_country_pulse and monthly_country_pulse below both
# working correctly) - so game-start hooks must delegate through a
# separate named on_action via `on_actions`, never put `effect`
# directly on on_game_start itself.
on_game_start = {
	on_actions = { aqd_on_game_start }
}
aqd_on_game_start = {
	effect = {
		# Repoint any country left on an evicted religious_school.
		aqd_schools_migrate = yes
		# Then overwrite with the historically-accurate starting aqidah/
		# madhhab for every Muslim country (scripted_effects/
		# z_aqd_starting_schools.txt). Must run after the migrate call
		# above so it has the final say.
		aqd_set_starting_schools = yes
		# Seed jurist seats at the prominent 1337 juristic centres
		# (scripted_effects/z_aqd_starting_schools.txt). After
		# aqd_set_starting_schools so the sharia_law policies are in
		# place. No starting madrasas are seeded - those grow from play.
		aqd_seed_starting_seats = yes
		# Seed the cached tier denominator (var:aqd_realm_jurist_total)
		# for every Muslim sharia realm now that the starting seats
		# exist - otherwise the share bands read a 0 total for the first
		# month. Refreshed monthly by aqd_recount_jurist_total_pulse and
		# after every jurist building placed/razed.
		aqd_recount_all_jurist_totals = yes
	}
}

# Catch stragglers (AI/events that set an evicted school after game
# start, and newly formed countries).
on_new_country_formed = {
	on_actions = { aqd_migrate_new_country }
}
aqd_migrate_new_country = {
	trigger = {
		religion = { has_religious_schools = yes }
		exists = religious_school
	}
	effect = {
		aqd_migrate_one_country = yes
	}
}

four_yearly_country_pulse = {
	on_actions = { aqd_migrate_pulse  aqd_seat_establishment_pulse }
}

# Ulema Institutions - seat establishment. The 9-way `>= 3` check below
# is a cheap raw-footprint prefilter (>=3 madrasas, or >=1 seat, of some
# madhhab) that just avoids the per-province scan for realms with no
# jurist footprint. Under the share-based tier model it is NOT a tier
# threshold and is intentionally decoupled from auto_modifiers/
# z_aqd_madhhab_tiers.txt. The real gate is aqd_province_has_madhhab_density
# (scripted_triggers/z_aqd_jurist_triggers.txt). See also
# scripted_effects/z_aqd_establishment_effects.txt and events/religion/
# z_aqd_establishment_events.txt.
aqd_seat_establishment_pulse = {
	trigger = {
		religion.group = religion_group:muslim
		has_policy = sharia_law_policy
		OR = {
			hanafi_institution_score  >= 3
			hanbali_institution_score >= 3
			maliki_institution_score  >= 3
			shafii_institution_score  >= 3
			zahiri_institution_score  >= 3
			ismaili_institution_score >= 3
			jafari_institution_score  >= 3
			zaidi_institution_score   >= 3
			ibadi_institution_score   >= 3
		}
	}
	effect = {
		aqd_try_seat_establishment = yes
	}
}

# The Jurists' Fatwa Against an Unjust War (events/religion/
# z_aqd_fatwa_events.txt). Cheap prefilters only, mirroring
# aqd_halqa_pulse's own division of labor. This is a plain effect, NOT
# random_events, since stage 1 is now silent-miss: scripted_effects/
# z_aqd_fatwa_effects.txt's aqd_fatwa_attempt_stage1 does the full
# aqd_fatwa_core_conditions check plus the centralized madhhab-approval
# roll (scripted_effects/z_aqd_madhhab_approval_effects.txt) BEFORE any
# event exists, and only calls trigger_event on a YES - a declining
# roll produces no popup at all this month. random = { chance = N ... }
# is the confirmed idiom for an effect-level dice gate (common/on_action/
# country_four_yearly.txt, _hardcoded.txt), replacing random_events'
# own chance_to_happen now that there's no event list to attach it to.
aqd_fatwa_pulse = {
	trigger = {
		religion.group = religion_group:muslim
		NOT = { has_variable = aqd_fatwa_cooldown }
		aqd_country_owns_any_jurist_seat = yes
		war_exhaustion >= 5
		any_current_war = { attacker_leader = root }
	}
	effect = {
		# aqd_fatwa_episode_start_check (scripted_effects/z_aqd_fatwa_
		# effects.txt) runs first every month this trigger passes - a cheap
		# has_variable check once var:aqd_fatwa_episode_active is already
		# set, and a one-time realm-wide sweep of stale z_aqd_fatwa_
		# declined_<X> markers the first month of a fresh unjust-war episode.
		aqd_fatwa_episode_start_check = yes
		random = {
			chance = 15
			aqd_fatwa_attempt_stage1 = yes
		}
	}
}

# Episode-end half of aqd_fatwa_episode_start_check's flag-toggle pair
# (scripted_effects/z_aqd_fatwa_effects.txt) - fires the one month root
# stops qualifying as any_current_war = { attacker_leader = root } while
# still carrying the "episode active" flag from a previous month, clears
# it, and sweep-clears the nine z_aqd_fatwa_declined_<X> "one shot per
# war" markers realm-wide so the NEXT unjust-war episode starts with
# every seat's slate wiped clean rather than inheriting this one's
# declines.
aqd_fatwa_episode_cleanup_pulse = {
	trigger = {
		religion.group = religion_group:muslim
		has_variable = aqd_fatwa_episode_active
		NOT = { any_current_war = { attacker_leader = root } }
	}
	effect = {
		remove_variable = aqd_fatwa_episode_active
		every_owned_location = {
			remove_location_modifier = z_aqd_fatwa_declined_hanafi
			remove_location_modifier = z_aqd_fatwa_declined_hanbali
			remove_location_modifier = z_aqd_fatwa_declined_maliki
			remove_location_modifier = z_aqd_fatwa_declined_shafii
			remove_location_modifier = z_aqd_fatwa_declined_zahiri
			remove_location_modifier = z_aqd_fatwa_declined_ismaili
			remove_location_modifier = z_aqd_fatwa_declined_jafari
			remove_location_modifier = z_aqd_fatwa_declined_zaidi
			remove_location_modifier = z_aqd_fatwa_declined_ibadi
		}
	}
}

# The Fatwa of Jihad (events/religion/z_aqd_jihad_fatwa_events.txt).
# Same silent-miss shape as aqd_fatwa_pulse above: scripted_effects/
# z_aqd_jihad_fatwa_effects.txt's aqd_jihad_fatwa_attempt does the seat
# pick + centralized approval roll before any event exists, and only
# fires the narration event on a YES.
#
# The OR covers BOTH participation paths (scripted_triggers/z_aqd_jihad_
# fatwa_triggers.txt) - aqd_is_jihad_participant (offensive: root fighting
# under cb_jihad or a jihad IO member) and aqd_is_defensive_jihad_
# participant (defensive: a heathen power has declared a religious war on
# root). Both share this same pulse, picker, approval engine, event and
# modifier keys - the only behavioural difference is which seats are
# eligible (aqd_jihad_seat_eligible_<X> allows heterodox seats through
# specifically for the defensive path).
#
# NO country-wide cooldown gate here - the per-seat 4-year z_aqd_fatwa_
# of_jihad_<X> modifier (checked inside aqd_jihad_seat_eligible's own
# per-madhhab branches, scripted_triggers/z_aqd_jihad_fatwa_triggers.txt)
# is the only cooldown, so a country with several eligible seats keeps
# rolling for whichever ones are still off cooldown instead of the whole
# pulse going dark for 5 years after any single success.
aqd_jihad_fatwa_pulse = {
	trigger = {
		religion.group = religion_group:muslim
		OR = {
			aqd_is_jihad_participant = yes
			aqd_is_defensive_jihad_participant = yes
		}
		any_owned_location = { aqd_jihad_seat_eligible = yes }
	}
	effect = {
		# aqd_jihad_episode_start_check (scripted_effects/z_aqd_jihad_
		# fatwa_effects.txt) runs first every month this trigger passes -
		# same reasoning as aqd_fatwa_pulse's own aqd_fatwa_episode_start_
		# check above.
		aqd_jihad_episode_start_check = yes
		random = {
			chance = 20
			aqd_jihad_fatwa_attempt = yes
		}
	}
}

# Episode-end half of aqd_jihad_episode_start_check's flag-toggle pair
# (scripted_effects/z_aqd_jihad_fatwa_effects.txt) - fires the one month
# root stops qualifying as EITHER aqd_is_jihad_participant OR aqd_is_
# defensive_jihad_participant (scripted_triggers/z_aqd_jihad_fatwa_
# triggers.txt) while still carrying the "episode active" flag from a
# previous month, clears it, and sweep-clears the nine z_aqd_jihad_
# declined_<X> "one shot per war" markers AND the nine z_aqd_jihad_
# blessed_<X> "already asked and said yes this war" markers realm-wide so
# the NEXT jihad (offensive or defensive) starts with every seat's slate
# wiped clean. A realm that is simultaneously offensive-jihading in one
# war and defending against a heathen in another keeps the episode open
# until BOTH end, not just one - the OR mirrors the pulse's own gate
# above exactly, so the episode boundary always matches whichever
# condition actually kept the pulse firing. Deliberately does NOT touch
# z_aqd_fatwa_of_jihad_<X> - its own real cross-war throttle keeps
# running on its own duration regardless of this war ending, exactly as
# before this whole feature.
aqd_jihad_episode_cleanup_pulse = {
	trigger = {
		religion.group = religion_group:muslim
		has_variable = aqd_jihad_episode_active
		NOT = {
			OR = {
				aqd_is_jihad_participant = yes
				aqd_is_defensive_jihad_participant = yes
			}
		}
	}
	effect = {
		remove_variable = aqd_jihad_episode_active
		every_owned_location = {
			remove_location_modifier = z_aqd_jihad_declined_hanafi
			remove_location_modifier = z_aqd_jihad_declined_hanbali
			remove_location_modifier = z_aqd_jihad_declined_maliki
			remove_location_modifier = z_aqd_jihad_declined_shafii
			remove_location_modifier = z_aqd_jihad_declined_zahiri
			remove_location_modifier = z_aqd_jihad_declined_ismaili
			remove_location_modifier = z_aqd_jihad_declined_jafari
			remove_location_modifier = z_aqd_jihad_declined_zaidi
			remove_location_modifier = z_aqd_jihad_declined_ibadi
			remove_location_modifier = z_aqd_jihad_blessed_hanafi
			remove_location_modifier = z_aqd_jihad_blessed_hanbali
			remove_location_modifier = z_aqd_jihad_blessed_maliki
			remove_location_modifier = z_aqd_jihad_blessed_shafii
			remove_location_modifier = z_aqd_jihad_blessed_zahiri
			remove_location_modifier = z_aqd_jihad_blessed_ismaili
			remove_location_modifier = z_aqd_jihad_blessed_jafari
			remove_location_modifier = z_aqd_jihad_blessed_zaidi
			remove_location_modifier = z_aqd_jihad_blessed_ibadi
		}
	}
}

# The Jurists' Fiqh Debate (events/religion/z_aqd_debate_events.txt).
# Same silent-miss shape as every other pulse here: scripted_effects/
# z_aqd_debate_effects.txt's aqd_debate_attempt does the full pair pick
# (pivot seat, then a same-location-or-area partner) before any event
# exists, and only fires the event on a confirmed find.
#
# The trigger's own aqd_debate_seat_eligible check is a cheap "does at
# least one eligible seat exist anywhere" prefilter only - it cannot
# cheaply confirm a same-area PARTNER also exists (scripted_triggers/
# z_aqd_debate_triggers.txt's own header explains why), so aqd_debate_
# attempt itself is the real gate; a realm with only one isolated
# eligible seat just silently fails most months. chance = 10 is a
# starting guess, easy to retune like every other pulse here.
aqd_debate_pulse = {
	trigger = {
		religion.group = religion_group:muslim
		has_policy = sharia_law_policy
		aqd_debate_seat_eligible = yes
	}
	effect = {
		random = {
			chance = 10
			aqd_debate_attempt = yes
		}
	}
}

# Ulema Institutions - the Ḥalqa: the periodic "activity" of a jurist
# seat (events/religion/z_aqd_halqa_events.txt). No sharia_law_policy
# requirement - a Muslim realm that only conquered a seat still gets
# ḥalaqāt (orthodox/heterodox only, never "state"). Trigger clauses run
# cheap -> expensive: religion group first, then the cheap "owns any
# seat" prefilter (most Muslim realms own none), then the province scan.
aqd_halqa_pulse = {
	trigger = {
		religion.group = religion_group:muslim
		aqd_country_owns_any_jurist_seat = yes
		aqd_country_has_any_halqa_ready = yes
	}
	random_events = {
		100 = z_aqd_halqa.1

		# TESTING: chance_to_happen raised from 2 to 80 so the pulse fires
		# almost every month instead of ~once per 4 years - revert before
		# release.
		chance_to_happen = 80
	}
}
aqd_migrate_pulse = {
	trigger = {
		religion.group = religion_group:muslim
		exists = religious_school
		OR = {
			religious_school ?= {
				OR = {
					is_sufi_school = yes
					this = religious_school:hanafi_school
					this = religious_school:hanbali_school
					this = religious_school:maliki_school
					this = religious_school:shafii_school
					this = religious_school:zahiri_school
					this = religious_school:jafari_school
					this = religious_school:ibadi_school
					this = religious_school:usuli_school
					this = religious_school:akhbari_school
				}
			}
			# Ibadi-only eviction: these four stay legal for Sunni.
			AND = {
				religion = religion:ibadi
				religious_school ?= {
					OR = {
						this = religious_school:ashari_school
						this = religious_school:maturidi_school
						this = religious_school:athari_school
						this = religious_school:mutazili_school
					}
				}
			}
		}
	}
	effect = {
		aqd_migrate_one_country = yes
	}
}
